scripted_diplomatic_actions = {
	# ----------------------------------------------------------------------
	# 1. Join Request (Non-Member -> Leader)
	# ----------------------------------------------------------------------
	bsm_ea_join_request = {
		icon = GFX_goal_generic_positive_trade_relations
		visible = {
			NOT = { has_war_with = FROM }
			# Target must be an EA leader
			FROM = { has_variable = bsm_ea_is_leader }
			# Must NOT already be in FROM's alliance
			NOT = {
				AND = {
					has_variable = bsm_ea_joined_spheres
					FROM = { has_variable = bsm_ea_sphere }
					is_in_array = {
						array = bsm_ea_joined_spheres
						value = FROM.bsm_ea_sphere
					}
				}
			}
		}
		allowed = {
			ROOT = {
				has_war = no
				NOT = { has_idea = recently_left_economic_alliance }
			}
		}
		selectable = { always = yes }

		complete_effect = {
			# Deduct costs
			add_political_power = -10
			set_temp_variable = { add_value = -25 }
			ucs_add_Unified_Currency = yes

			FROM = {
				country_event = { id = bsm_ea_diplomatic.1 }
			}
		}

		ai_desire = {
			base = 10
			modifier = {
				add = 50
				has_opinion = { target = FROM value > 50 }
			}
		}
	}
	# ----------------------------------------------------------------------
	# 2. Leave (Member -> Leader)
	# ----------------------------------------------------------------------
	bsm_action_leave_economic_alliance = {
		icon = GFX_goal_generic_dangerous_deal
		visible = {
			# Must be in FROM's alliance
			FROM = { has_variable = bsm_ea_is_leader }
			has_variable = bsm_ea_joined_spheres
			FROM = { has_variable = bsm_ea_sphere }
			is_in_array = {
				array = bsm_ea_joined_spheres
				value = FROM.bsm_ea_sphere
			}
		}
		selectable = {
			always = yes
		}
		allowed = {
			ROOT = { has_war = no }
		}

		complete_effect = {
			# Deduct cost
			add_political_power = -50

			# Notify Leader
			FROM = {
				country_event = { id = bsm_ea_diplomatic.2 }
			}
			# Execute Leave Effect
			set_temp_variable = { bsm_ea_token = FROM.bsm_ea_sphere }
			bsm_ea_leave_sphere = yes
		}

		ai_desire = {
			base = 0
			modifier = {
				add = 100
				has_opinion = { target = FROM value < -50 }
			}
		}
	}
	# ----------------------------------------------------------------------
	# 3. Kick (Leader -> Member)
	# ----------------------------------------------------------------------
	bsm_action_kick_from_economic_alliance = {
		icon = GFX_goal_generic_dangerous_deal
		visible = {
			has_variable = bsm_ea_is_leader
			has_variable = bsm_ea_sphere
			# Check if Target is in MY alliance
			FROM = {
				has_variable = bsm_ea_joined_spheres
				is_in_array = {
					array = bsm_ea_joined_spheres
					value = ROOT.bsm_ea_sphere
				}
			}
		}
		allowed = {
			ROOT = {
				FROM = {
					is_in_array = {
						array = bsm_ea_joined_spheres
						value = ROOT.bsm_ea_sphere
					}
				}
			}
		}
		selectable = {
			always = yes
		}

		complete_effect = {
			# Deduct cost
			add_political_power = -100

			# Notify & Kick Target
			FROM = {
				country_event = { id = bsm_ea_diplomatic.3 }
				set_temp_variable = { bsm_ea_token = ROOT.bsm_ea_sphere }
				bsm_ea_leave_sphere = yes
			}
		}

		ai_desire = {
			base = 0
			modifier = {
				add = 100
				has_opinion = { target = FROM value < -75 }
			}
		}
	}
	# ----------------------------------------------------------------------
	# 4. Invite (Leader -> Non-Member)
	# ----------------------------------------------------------------------
	bsm_action_invite_to_economic_alliance = {
		icon = GFX_goal_generic_positive_trade_relations
		visible = {
			has_variable = bsm_ea_is_leader
			has_variable = bsm_ea_sphere
			has_opinion = {
				target = FROM
				value > 0
			}
			# Not already in MY alliance
			NOT = {
				FROM = {
					has_variable = bsm_ea_joined_spheres
					is_in_array = {
						array = bsm_ea_joined_spheres
						value = ROOT.bsm_ea_sphere
					}
				}
			}
		}
		selectable = {
			always = yes
		}
		allowed = {
			always = yes
		}

		complete_effect = {
			# Deduct cost
			add_political_power = -25

			# Send Invite to Target
			FROM = {
				country_event = { id = bsm_ea_diplomatic.4 }
			}
		}

		ai_desire = {
			base = 10
			modifier = {
				add = 50
				has_opinion = { target = FROM value > 50 }
			}
			modifier = {
				add = 25
				is_in_faction_with = FROM
			}
		}
	}
	# ----------------------------------------------------------------------
	# 5. Create (Non-Leader -> Non-Leader)
	# ----------------------------------------------------------------------
	bsm_action_create_economic_alliance = {
		icon = GFX_goal_generic_positive_trade_relations
		visible = {
			# Must NOT be a leader (One leadership per country)
			NOT = { has_variable = bsm_ea_is_leader }
			# Target also not a leader
			FROM = {
				NOT = { has_variable = bsm_ea_is_leader }
			}
			has_opinion = {
				target = FROM
				value > 0
			}
		}
		selectable = {
			always = yes
		}
		allowed = {
			ROOT = { has_war = no }
		}

		complete_effect = {
			# Deduct cost
			add_political_power = -50

			# Send Proposal to Target
			FROM = {
				country_event = { id = bsm_ea_diplomatic.5 }
			}
		}

		ai_desire = {
			base = 5
			modifier = {
				add = 50
				has_opinion = { target = FROM value > 75 }
			}
			modifier = {
				add = 100
				is_in_faction_with = FROM
				is_faction_leader = yes
			}
		}
	}
}
