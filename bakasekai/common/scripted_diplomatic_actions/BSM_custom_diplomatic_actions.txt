scripted_diplomatic_actions = {
	bsm_fabricate_claim_action = {
		# ボタン名/説明はアクションID(bsm_fabricate_claim_action / _desc)をlocキーにして表示される
		icon = "GFX_goal_generic_major_power_challenge"

		visible = {
			# 自分自身は対象にできない
			NOT = { tag = FROM }
			NOT = { has_war_with = FROM }

			custom_trigger_tooltip = {
				tooltip = "BSM_FABRICATE_CLAIM_ENABLED_TOOLTIP"
				bsm_can_initiate_fabricate_claim_diplo_action_trigger = yes
			}
		}

		allowed = {
		}

		cost = 25
		cost_string = "BSM_FABRICATE_CLAIM_DIPLO_COST"

		complete_effect = {
			# 対象国(FROM)を event_target として保存し、自国でイベントを発火する。
			# 無印 country_event だとイベント内 FROM が自国になるため、対象国は event_target で渡す。
			FROM = { save_event_target_as = bsm_fabricate_claim_target }
			country_event = {
				id = bsm_fabricate_claim_events.1
				days = 1
			}
		}


	}
	recall_volunteers = {

		allowed = {
			ROOT = { is_ai = no }
		}
		visible = {
			always = yes
		}
		selectable = {
			# doing this for nicer tooltips
			if = {
				limit = {
					NOT = { has_volunteers_amount_from = { tag = ROOT count > 0 } }
				}
				custom_trigger_tooltip = {
					tooltip = RECALL_VOLUNTEERS_TOOLTIP
					always = no
				}
			}
			else = {
				custom_trigger_tooltip = {
					tooltip = RECALL_VOLUNTEERS_TOOLTIP_G
					always = yes
				}
			}
		}

		cost = 0
		command_power = 0

		show_acceptance_on_action_button = no

		icon = 1

		complete_effect = {
			ROOT = { recall_volunteers_from = PREV }
		}

		ai_acceptance = {
			condition = {
				base = 100
			}
		}
	}
}
