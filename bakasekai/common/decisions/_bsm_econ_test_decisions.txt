# GDP基盤(System A) 検証用ディシジョン
# 既存の bsm_ea_test_category を再利用。bsm_econ_update_gdp を実行し [BSM_GDP] を game.log に出力する。

bsm_ea_test_category = {

	bsm_econ_test_gdp_probe = {
		icon = GFX_decision_generic_political_discourse

		available = { always = yes }
		visible = { always = yes }

		cost = 0
		fire_only_once = no

		complete_effect = {
			bsm_econ_update_gdp = yes
			log = "[BSM_GDP] ([ROOT.GetName]) gdp_real=[?bsm_gdp_real|1] = output[?bsm_gdp_output|2] x state_sum[?bsm_gdp_state_sum|1] /10 | res(別計上)=[?bsm_resource_value|1] growth%=[?bsm_gdp_growth_pct|2] | civ=[?num_of_civilian_factories|0] stab=[?stability|2] cult=[?Cultural_Degree|2] states=[?owned_controlled_states^num|0]"
		}

		ai_will_do = { factor = 0 }
	}

	# 威信(System C) 検証用: GDP更新→威信計算→[BSM_PRESTIGE] を game.log に出力
	bsm_econ_test_prestige_probe = {
		icon = GFX_decision_generic_political_discourse

		available = { always = yes }
		visible = { always = yes }

		cost = 0
		fire_only_once = no

		complete_effect = {
			# Vic3型3フェーズ: 全国の GDP+威信(パワー合算) → 世界平均 → ランク判定 → 自国をログ
			every_country = {
				bsm_econ_update_gdp = yes
				bsm_prestige_compute = yes
			}
			bsm_prestige_update_world = yes
			every_country = { bsm_prestige_assign_rank = yes }
			log = "[BSM_PRESTIGE] ([ROOT.GetName]) #[?bsm_prestige_position|0] [GetBsmPrestigeRankName](rank[?bsm_prestige_rank|0]) prestige=[?bsm_prestige|0] ratio=[?bsm_prestige_ratio|2] world_avg=[?global.bsm_prestige_world_avg|1] | gdp=[?bsm_gdp_real|0] bat=[?num_battalions|0] ships=[?num_ships|0] subj=[?num_subjects|0]"
			log = "[BSM_PRESTIGE] 分布: 超大国[?global.bsm_rank_cnt_7|0] 列強[?global.bsm_rank_cnt_6|0] 大国[?global.bsm_rank_cnt_5|0] 地域大国[?global.bsm_rank_cnt_4|0] 中堅[?global.bsm_rank_cnt_3|0] 小国[?global.bsm_rank_cnt_2|0] 弱小[?global.bsm_rank_cnt_1|0] (全[?global.bsm_prestige_cnt|0]国)"
		}

		ai_will_do = { factor = 0 }
	}

	# 威信 +/- modifier(bsm_prestige_flat/factor)の動作確認: テスト源を自国に付与/解除
	bsm_econ_test_prestige_mod_toggle = {
		icon = GFX_decision_generic_political_discourse

		available = { always = yes }
		visible = { always = yes }

		cost = 0
		fire_only_once = no

		complete_effect = {
			if = {
				limit = { has_dynamic_modifier = { modifier = bsm_prestige_test_boost } }
				remove_dynamic_modifier = { modifier = bsm_prestige_test_boost }
				log = "[BSM_PRESTIGE] test boost 解除。威信probeで戻りを確認"
			}
			else = {
				add_dynamic_modifier = { modifier = bsm_prestige_test_boost }
				log = "[BSM_PRESTIGE] test boost 付与(flat+500 / +20%)。威信probeで上昇を確認"
			}
		}

		ai_will_do = { factor = 0 }
	}
}
