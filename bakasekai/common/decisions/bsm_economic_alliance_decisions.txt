bsm_economic_alliance_decisions = {
	# 従属国への利益分配切り替え
	ucs_toggle_profit_distribution = {
		icon = generic_trade
		allowed = { always = yes }
		available = {
			is_subject = no
			any_subject_country = { always = yes }
			# 経済同盟加盟国のみ
			is_in_bsm_ea_sphere = yes
		}
		visible = {
			is_subject = no
			any_subject_country = { always = yes }
			is_in_bsm_ea_sphere = yes
		}
		modifier = { }
		days_re_enable = 30
		fire_only_once = no
		complete_effect = {
			if = {
				limit = { has_country_flag = ucs_distribute_profit }
				clr_country_flag = ucs_distribute_profit
			}
			else = {
				set_country_flag = ucs_distribute_profit
			}
		}
		ai_will_do = {
			factor = 1
			modifier = {
				factor = 0
				NOT = { has_country_flag = ucs_distribute_profit }
			}
		}
	}

			}
		}

				bsm_ea_can_form_sphere_allowed = yes
			}
		}

		days_re_enable = 30
		cost = 0 # Effect handles cost
		complete_effect = {
			bsm_ea_form_sphere_action = yes
		}
		ai_will_do = { factor = 0 }
	}

	# 汎用経済同盟への加盟
	# ----------------------------------------------------------------------

		icon = generic_diplomacy
		allowed = { always = yes }
		target_array = neighbors

			FROM = {
				is_bsm_ea_leader = yes

			}
		}
		visible = {
			FROM = {
				is_bsm_ea_leader = yes

			}
		}
		available = {
			FROM = {
				bsm_ea_can_join_sphere_allowed = yes
			}
		}
		fire_only_once = no
		days_re_enable = 30
		cost = 0
		complete_effect = {
			bsm_ea_join_sphere_action = yes
		}
		ai_will_do = { factor = 0 }
	}
