	ucs_toggle_profit_distribution = {
		icon = generic_trade
		allowed = { always = yes }
		available = {

			is_in_bsm_ea_sphere = yes
		}
		modifier = { }

		fire_only_once = no
		complete_effect = {
			if = {
				limit = { has_country_flag = ucs_distribute_profit }
				clr_country_flag = ucs_distribute_profit
			}
			else = {
				set_country_flag = ucs_distribute_profit
			}
		}
		ai_will_do = {
			factor = 1
			modifier = {
				factor = 0
				NOT = { has_country_flag = ucs_distribute_profit }
			}
		}
	}

	bsm_form_economic_alliance = {
		icon = generic_research

			}
		}
		visible = {

				NOT = { tag = XXX }
				NOT = { has_variable = bsm_ea_is_leader }
			}
		}
		available = {
			has_political_power > 100

			factor = 1

			modifier = { factor = 0  has_war = yes }

				AND = {
					has_idea = recently_left_economic_alliance
					check_variable = { bsm_ea_recently_left_token = FROM.bsm_ea_sphere }
				}
			}
			has_opinion = {
				target = FROM
				value > 25
			}
		}
		fire_only_once = no
		days_re_enable = 30
		cost = 0
		complete_effect = {
			bsm_ea_join_sphere_action = yes
		}
		ai_will_do = {
			factor = 5
			# 加盟しない方針なら不実行
			modifier = { factor = 0  bsm_ea_ai_isolationist = yes }
			# 拡張志向で積極化
			modifier = { factor = 2  has_country_flag = ea_ai_expansionist }
			modifier = { factor = 2  has_global_flag = ea_ai_global_expansion }
			# 搾取方針の盟主には加盟しにくい
			modifier = {
				factor = 0.3
				FROM = { check_variable = { global.bsm_ea_policy_@var:bsm_ea_sphere = 1 } }
			}
			# 誘致された相手 (盟主FROM向けの加盟バイアス) は加盟しやすい
			modifier = { factor = 1.5  check_variable = { bsm_ea_join_bias@FROM > 0 } }
			modifier = { factor = 2    check_variable = { bsm_ea_join_bias@FROM > 49 } }
			# 妨害された相手 (汎用malus) は加盟しにくい

			modifier = { factor = 0.2  check_variable = { bsm_ea_join_malus > 49 } }
			# 戦争中は控える
			modifier = { factor = 0  has_war = yes }
		}

	# ----------------------------------------------------------------------
	bsm_merge_economic_alliance = {
		icon = generic_research
		allowed = { always = yes }
		# 盟主のみ実行可能。target_array を指定せず全盟主を対象にする
		target_root_trigger = {
			bsm_ea_autonomy_allows = yes
			has_variable = bsm_ea_is_leader
		}
		target_trigger = {
			FROM = {
				exists = yes
				NOT = { tag = XXX }
				has_variable = bsm_ea_is_leader
				NOT = { has_war_with = ROOT }
			}
		}
		visible = {
			has_variable = bsm_ea_is_leader
			NOT = { has_war_with = FROM }
			FROM = {
				exists = yes
				NOT = { tag = XXX }
				has_variable = bsm_ea_is_leader
			}
		}
		available = {
			has_variable = bsm_ea_is_leader
			has_political_power > 200
			check_variable = { Unified_Currency > 200 }
			has_civil_war = no
			NOT = { has_war_with = FROM }
			NOT = { has_government = ruinism }
			NOT = { has_government = anarchism }
			has_stability > 0
			has_opinion = {
				target = FROM
				value > 50
			}
		}
		fire_only_once = no
		days_re_enable = 30
		cost = 0
		complete_effect = {
			bsm_ea_merge_sphere_action = yes
		}
		ai_will_do = {
			factor = 1
			# 統合しない方針なら不実行
			modifier = { factor = 0  bsm_ea_ai_isolationist = yes }
			# 拡張志向で積極化
			modifier = { factor = 2  has_country_flag = ea_ai_expansionist }
			modifier = { factor = 2  has_global_flag = ea_ai_global_expansion }
			# 高好感の相手とのみ統合
			modifier = { factor = 0  FROM = { has_opinion = { target = ROOT value < 75 } } }
			# 戦争中は控える
			modifier = { factor = 0  has_war = yes }
		}
	}
