bsm_economic_alliance_decisions = {
	# 従属国への利益分配切り替え
	ucs_toggle_profit_distribution = {
		icon = generic_trade
		allowed = { always = yes }
		available = {
			is_subject = no
			any_subject_country = { always = yes }
			is_in_bsm_ea_sphere = yes
		}
		visible = {
			is_subject = no
			any_subject_country = { always = yes }
			is_in_bsm_ea_sphere = yes
		}
		modifier = { }
		days_re_enable = 30
		fire_only_once = no
		complete_effect = {
			if = {
				limit = { has_country_flag = ucs_distribute_profit }
				clr_country_flag = ucs_distribute_profit
			}
			else = {
				set_country_flag = ucs_distribute_profit
			}
		}
		ai_will_do = {
			factor = 1
			modifier = {
				factor = 0
				NOT = { has_country_flag = ucs_distribute_profit }
			}
		}
	}
	# ----------------------------------------------------------------------
	# 汎用経済同盟の結成
	# ----------------------------------------------------------------------
	bsm_form_economic_alliance = {
		icon = generic_research
		allowed = { always = yes }
		target_array = neighbors
		target_trigger = {
			FROM = {
				exists = yes
				NOT = { tag = XXX }
				NOT = { has_war_with = ROOT }
			}
		}
		visible = {
			is_subject = no
			has_political_power > 100
			NOT = { has_war_with = FROM }
			FROM = {
				exists = yes
				NOT = { tag = XXX }
			}
		}
		available = {
			has_political_power > 100
			check_variable = { Unified_Currency > 100 }
			has_civil_war = no
			NOT = { has_war_with = FROM }
			NOT = { has_government = ruinism }
			NOT = { has_government = anarchism }
			has_stability > 0
			has_opinion = {
				target = FROM
				value > 25
			}
		}
		fire_only_once = no
		days_re_enable = 30
		cost = 0
		complete_effect = {
			bsm_ea_form_sphere_action = yes
		}
		ai_will_do = {
			factor = 1
			# 結成しない方針なら不実行
			modifier = { factor = 0  bsm_ea_ai_isolationist = yes }
			# 拡張志向で積極化
			modifier = { factor = 4  has_country_flag = ea_ai_expansionist }
			modifier = { factor = 2  has_global_flag = ea_ai_global_expansion }
			# 戦争中は控える
			modifier = { factor = 0  has_war = yes }
		}
	}
	# ----------------------------------------------------------------------
	# 汎用経済同盟への加盟
	# ----------------------------------------------------------------------
	bsm_join_economic_alliance = {
		icon = generic_diplomacy
		allowed = { always = yes }
		target_array = neighbors
		target_trigger = {
			FROM = {
				exists = yes
				NOT = { tag = XXX }
				has_variable = bsm_ea_is_leader
				NOT = { has_war_with = ROOT }
			}
		}
		visible = {
			is_subject = no
			FROM = {
				exists = yes
				has_variable = bsm_ea_is_leader
				NOT = { has_war_with = ROOT }
			}
			NOT = {
				is_in_array = { bsm_ea_joined_spheres = FROM.bsm_ea_sphere }
			}
		}
		available = {
			has_political_power > 10
			check_variable = { Unified_Currency > 25 }
			has_civil_war = no
			NOT = { has_war_with = FROM }
			NOT = { has_government = ruinism }
			NOT = { has_government = anarchism }
			has_stability > 0
			NOT = { has_idea = recently_left_economic_alliance }
			has_opinion = {
				target = FROM
				value > 25
			}
		}
		fire_only_once = no
		days_re_enable = 30
		cost = 0
		complete_effect = {
			bsm_ea_join_sphere_action = yes
		}
		ai_will_do = {
			factor = 5
			# 加盟しない方針なら不実行
			modifier = { factor = 0  bsm_ea_ai_isolationist = yes }
			# 拡張志向で積極化
			modifier = { factor = 2  has_country_flag = ea_ai_expansionist }
			modifier = { factor = 2  has_global_flag = ea_ai_global_expansion }
			# 搾取方針の盟主には加盟しにくい
			modifier = {
				factor = 0.3
				FROM = { check_variable = { global.bsm_ea_policy_@var:bsm_ea_sphere = 1 } }
			}
			# ROOT→対象の加盟意欲バイアス (誘致/妨害)
			modifier = { factor = 1.5  check_variable = { bsm_ea_join_bias@FROM > 0 } }
			modifier = { factor = 2    check_variable = { bsm_ea_join_bias@FROM > 49 } }
			modifier = { factor = 0.5  check_variable = { bsm_ea_join_bias@FROM < 0 } }
			modifier = { factor = 0    check_variable = { bsm_ea_join_bias@FROM < -49 } }
			# 戦争中は控える
			modifier = { factor = 0  has_war = yes }
		}
	}

	# ----------------------------------------------------------------------
	# 加盟誘致 (盟主が対象国の加盟意欲を上げる)
	# ROOT = 盟主, FROM = 対象国
	# ----------------------------------------------------------------------
	bsm_ea_recruit_country = {
		icon = generic_political_discord
		allowed = { always = yes }
		target_array = neighbors
		target_trigger = {
			FROM = {
				exists = yes
				NOT = { tag = XXX }
				is_subject = no
				NOT = { is_in_array = { bsm_ea_joined_spheres = ROOT.bsm_ea_sphere } }
			}
		}
		visible = {
			has_variable = bsm_ea_is_leader
			is_subject = no
			FROM = {
				exists = yes
				NOT = { tag = XXX }
			}
		}
		available = {
			has_political_power > 30
			NOT = { has_war_with = FROM }
		}
		cost = 30
		fire_only_once = no
		days_re_enable = 30
		complete_effect = {
			# ROOT = 盟主, FROM = 対象国
			FROM = {
				add_to_variable = { var = bsm_ea_join_bias@PREV value = 25 }
				clamp_variable = { var = bsm_ea_join_bias@PREV min = -100 max = 100 }
			}
		}
		ai_will_do = {
			factor = 1
			modifier = { factor = 0  bsm_ea_ai_isolationist = yes }
			modifier = { factor = 3  has_country_flag = ea_ai_expansionist }
			modifier = { factor = 2  has_global_flag = ea_ai_global_expansion }
			# 既に好感度が高い相手を優先
			modifier = {
				factor = 2
				FROM = { has_opinion = { target = ROOT value > 40 } }
			}
			modifier = { factor = 0  has_war = yes }
		}
	}

	# ----------------------------------------------------------------------
	# 加盟妨害 (対象国の自陣営への加盟意欲を下げる / ライバル盟主が使用)
	# ROOT = 妨害する盟主, FROM = 対象国
	# ----------------------------------------------------------------------
	bsm_ea_discourage_country = {
		icon = generic_political_discord
		allowed = { always = yes }
		target_array = neighbors
		target_trigger = {
			FROM = {
				exists = yes
				NOT = { tag = XXX }
				is_subject = no
			}
		}
		visible = {
			has_variable = bsm_ea_is_leader
			is_subject = no
			FROM = {
				exists = yes
				NOT = { tag = XXX }
			}
		}
		available = {
			has_political_power > 30
			NOT = { has_war_with = FROM }
		}
		cost = 30
		fire_only_once = no
		days_re_enable = 30
		complete_effect = {
			FROM = {
				add_to_variable = { var = bsm_ea_join_bias@PREV value = -25 }
				clamp_variable = { var = bsm_ea_join_bias@PREV min = -100 max = 100 }
			}
		}
		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------------
	# 経済同盟からの脱退
	# ----------------------------------------------------------------------
	bsm_leave_economic_alliance = {
		icon = generic_volunteer_expedition_bonus
		allowed = { always = yes }
		available = {
			is_in_bsm_ea_sphere = yes
			NOT = { has_idea = recently_left_economic_alliance }
		}
		visible = {
			is_in_bsm_ea_sphere = yes
		}
		cost = 50
		fire_only_once = no
		days_re_enable = 30
		ai_will_do = { factor = 0 }
		complete_effect = {
			bsm_ea_leave_sphere_action = yes
		}
	}
}
