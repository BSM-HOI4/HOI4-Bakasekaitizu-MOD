bsm_economic_alliance_decisions = {
	# 従属国への利益分配切り替え
	ucs_toggle_profit_distribution = {
		icon = generic_trade
		allowed = { always = yes }
		available = {
			is_subject = no
			any_subject_country = { always = yes }
			is_in_bsm_ea_sphere = yes
		}
		visible = {
			is_subject = no
			any_subject_country = { always = yes }
			is_in_bsm_ea_sphere = yes
		}
		modifier = { }
		days_re_enable = 30
		fire_only_once = no
		complete_effect = {
			if = {
				limit = { has_country_flag = ucs_distribute_profit }
				clr_country_flag = ucs_distribute_profit
			}
			else = {
				set_country_flag = ucs_distribute_profit
			}
		}
		ai_will_do = {
			factor = 1
			modifier = {
				factor = 0
				NOT = { has_country_flag = ucs_distribute_profit }
			}
		}
	}
	# ----------------------------------------------------------------------
	# 汎用経済同盟の結成
	# ----------------------------------------------------------------------
	bsm_form_economic_alliance = {
		icon = generic_research
		allowed = { always = yes }
		target_array = neighbors
		target_trigger = {
			FROM = {
				exists = yes
				NOT = { tag = '_' }
				NOT = { has_war_with = ROOT }
			}
		}
		visible = {
			is_subject = no
			has_political_power > 100
			NOT = { has_war_with = FROM }
			FROM = {
				exists = yes
				NOT = { tag = '_' }
			}
		}
		available = {
			has_political_power > 100
			check_variable = { Unified_Currency > 100 }
			has_civil_war = no
			NOT = { has_war_with = FROM }
			NOT = { has_government = ruinism }
			NOT = { has_government = anarchism }
			has_stability > 0
			has_opinion = {
				target = FROM
				value > 25
			}
		}
		fire_only_once = no
		days_re_enable = 30
		cost = 0
		complete_effect = {
			bsm_ea_form_sphere_action = yes
		}
		ai_will_do = { factor = 0 }
	}
	# ----------------------------------------------------------------------
	# 汎用経済同盟への加盟
	# ----------------------------------------------------------------------
	bsm_join_economic_alliance = {
		icon = generic_diplomacy
		allowed = { always = yes }
		target_array = neighbors
		target_trigger = {
			FROM = {
				exists = yes
				NOT = { tag = '_' }
				has_variable = bsm_ea_is_leader
				NOT = { has_war_with = ROOT }
			}
		}
		visible = {
			is_subject = no
			FROM = {
				exists = yes
				has_variable = bsm_ea_is_leader
				NOT = { has_war_with = ROOT }
			}
			NOT = {
				is_in_array = { bsm_ea_joined_spheres = FROM.bsm_ea_sphere }
			}
		}
		available = {
			has_political_power > 10
			check_variable = { Unified_Currency > 25 }
			has_civil_war = no
			NOT = { has_war_with = FROM }
			NOT = { has_government = ruinism }
			NOT = { has_government = anarchism }
			has_stability > 0
			NOT = { has_idea = recently_left_economic_alliance }
			has_opinion = {
				target = FROM
				value > 25
			}
		}
		fire_only_once = no
		days_re_enable = 30
		cost = 0
		complete_effect = {
			bsm_ea_join_sphere_action = yes
		}
		ai_will_do = { factor = 0 }
	}

	# ----------------------------------------------------------------------
	# 経済同盟からの脱退
	# ----------------------------------------------------------------------
	bsm_leave_economic_alliance = {
		icon = generic_volunteer_expedition_bonus
		allowed = { always = yes }
		available = {
			is_in_bsm_ea_sphere = yes
			NOT = { has_idea = recently_left_economic_alliance }
		}
		visible = {
			is_in_bsm_ea_sphere = yes
		}
		cost = 50
		fire_only_once = no
		days_re_enable = 30
		ai_will_do = { factor = 0 }
		complete_effect = {
			bsm_ea_leave_sphere_action = yes
		}
	}
}
