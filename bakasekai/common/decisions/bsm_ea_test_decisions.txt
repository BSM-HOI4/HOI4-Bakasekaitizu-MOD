	# 選択中の加盟国を弱通貨化 (テスト用)
	bsm_ea_test_weaken_selected_currency = {
		icon = GFX_decision_generic_prepare_civil_war

		available = {
			check_variable = { bsm_ea_has_selected_member = 1 }
			has_event_target = bsm_ea_selected_member
		}

		visible = {
			always = yes
		}

		cost = 0
		fire_only_once = no

		complete_effect = {
			event_target:bsm_ea_selected_member = {
				if = {
					limit = { has_variable = bsm_nc_rate }
					set_variable = {
						var = bsm_nc_rate
						value = { value = bsm_nc_base_rate  multiply = 2.0 }
					}
					set_variable = { bsm_nc_stability = 20 }
					set_variable = { bsm_nc_strength_cat = 2 }
					set_variable = { bsm_nc_inflation = 25 }
					log = "[BSM EA TEST] Weakened currency for [THIS.GetName]"
				}
			}
		}

		ai_will_do = {
			factor = 0
		}
