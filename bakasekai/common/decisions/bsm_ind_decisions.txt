	IND_quell_communal_violence = {

		allowed = { original_tag = IND }
		available = {
			original_tag = IND
			has_country_flag = IND_independent
			has_war = no
			has_stability < 0.60
		}
		visible = {
			original_tag = IND
			has_country_flag = IND_independent
			NOT = { has_country_flag = IND_partition_resolved }
		}
		cost = 75
		days_remove = 30
		days_re_enable = 30
		fire_only_once = no
		modifier = {
			stability_weekly = 0.01
			consumer_goods_factor = 0.03
		}
		complete_effect = {
			add_stability = 0.05
			add_war_support = -0.03
		}
		remove_effect = {
			add_stability = 0.02
		}
		ai_will_do = {
			factor = 3
			modifier = {
				factor = 0
