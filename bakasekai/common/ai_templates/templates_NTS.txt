armor_NTS = {
  available_for = {
    NTS
  }
  role = armor
  upgrade_prio = {
    factor = 0
    modifier = {
      factor = 2
      OR = {
        has_country_flag = NTS_purge_accepted
        has_completed_focus = NTS_numbered_divisions
      }
    }
  }

  NTS_numbered_light_armor = {
    upgrade_prio = {
      factor = 0
      modifier = {
        factor = 1
        has_completed_focus = NTS_numbered_divisions
      }
      modifier = {
        factor = 3
        OR = {
          has_tech = basic_light_tank
          has_tech = basic_light_tank_chassis
        }
      }
    }

    target_template = {
      support = {
        engineer = 1
        mot_recon = 1
      }
      regiments = {
        light_armor = 5
        motorized = 4
      }
    }

    replace_at_match = 1.5
    target_min_match = 0.5
  }
}
