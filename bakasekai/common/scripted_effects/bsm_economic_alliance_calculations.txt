# Economic Alliance Power / Influence Calculation
# 影響力 = IC × (1 + modifier@ea_influence_factor)
# 影響力が議席配分と UC/PP バフ配分を駆動する（既存 UC システムと結合）

# ------------------------------------------
# 単一陣営の影響力・議席・表示%を計算（バフは触らない＝GUI/週次共用で安全）
# 入力: bsm_ea_calc_token (temp)
# 各メンバーに bsm_ea_influence / bsm_ea_member_seats / bsm_ea_influence_pct を設定
# 盟主(index 0)に bsm_ea_sphere_total_influence を保存
# ------------------------------------------
bsm_ea_calc_sphere_influence = {
	if = {
		limit = { check_variable = { global.bsm_ea_member_list_@var:bsm_ea_calc_token^num > 0 } }

		# 1. 各メンバーの影響力を算出し合計を取る
		set_temp_variable = { t_total_inf = 0 }
		for_each_loop = {
			array = global.bsm_ea_member_list_@var:bsm_ea_calc_token
			value = v_mem
			var:v_mem = {
				# 基礎国力 = 民需 + 軍需
				# 影響力 = (民需 + 軍需) * clamp(1 + modifier@ea_influence_factor, 0.1+)
				set_variable = {
					var = bsm_ea_influence
					value = {
						value = num_of_civilian_factories
						add = num_of_military_factories
						multiply = {
							value = 1
							add = modifier@ea_influence_factor
							max = 0.1
						}
					}
				}
				# 表示用の既定値（合計0/未計算時のゴミ表示を防ぐ）
				set_variable = { bsm_ea_member_seats = 0 }
				set_variable = { bsm_ea_influence_pct = 0 }
				set_temp_variable = { PREV.t_add_inf = bsm_ea_influence }
			}
			add_to_temp_variable = { t_total_inf = t_add_inf }
		}

		# scope変更先(var:v_mem2)から確実に読めるよう合計をグローバルへ退避
		# (temp変数 PREV.t_total_inf はscope変更後に0と解決されゼロ除算→オーバーフローになる)
		set_variable = { global.bsm_ea_g_total_inf = t_total_inf }

		# 盟主に陣営合計影響力を保存（バフ配分用）
		set_temp_variable = { t_leader = global.bsm_ea_member_list_@var:bsm_ea_calc_token^0 }
		var:t_leader = { set_variable = { bsm_ea_sphere_total_influence = global.bsm_ea_g_total_inf } }

		# 2. 影響力比 → 議席(加盟国数×10) / 表示%
		if = {
			limit = { check_variable = { t_total_inf > 0 } }
			set_temp_variable = {
				var = t_total_seats
				value = {
					value = global.bsm_ea_member_list_@var:bsm_ea_calc_token^num
					multiply = 10
					max = 10
				}
			}
			set_variable = { global.bsm_ea_g_total_seats = t_total_seats }
			for_each_loop = {
				array = global.bsm_ea_member_list_@var:bsm_ea_calc_token
				value = v_mem2
				var:v_mem2 = {
					# 議席数 = round(share × (加盟国数×10))  ※合計はグローバル経由で読む
					set_variable = {
						var = bsm_ea_member_seats
						value = {
							value = bsm_ea_influence
							divide = global.bsm_ea_g_total_inf
							multiply = global.bsm_ea_g_total_seats
							round = yes
							clamp = { min = 0 max = 640 }
						}
					}

					# 表示用 %（整数, 0..100）
					set_variable = {
						var = bsm_ea_influence_pct
						value = {
							value = bsm_ea_influence
							divide = global.bsm_ea_g_total_inf
							multiply = 100
							round = yes
							clamp = { min = 0 max = 100 }
						}
					}
				}
			}
		}
	}
}

# ------------------------------------------
# 週次: 全陣営の影響力再計算 + UC/PP バフ配分
# (on_weekly から呼ばれる)
# ------------------------------------------
bsm_update_economic_alliance_power_share = {
	# 0. ランキング更新
	bsm_ea_sort_sphere_list = yes

	# 1. メンバーのバフ値をリセット（every_country を避ける）
	for_each_loop = {
		array = global.bsm_ea_sphere_list
		value = v_token_reset
		for_each_loop = {
			array = global.bsm_ea_member_list_@var:v_token_reset
			value = v_mem_reset
			var:v_mem_reset = {
				set_variable = { economic_alliance_pp_buff_value = 0 }
			}
		}
	}

	# 2. 各陣営: 影響力・議席を計算 → 影響力比でバフを加算
	for_each_loop = {
		array = global.bsm_ea_sphere_list
		value = v_token

		set_temp_variable = { bsm_ea_calc_token = var:v_token }
		bsm_ea_calc_sphere_influence = yes

		if = {
			limit = { check_variable = { global.bsm_ea_member_list_@var:v_token^num > 0 } }
			set_temp_variable = { t_ldr = global.bsm_ea_member_list_@var:v_token^0 }
			set_temp_variable = { t_total = 0 }
			var:t_ldr = { set_temp_variable = { PREV.t_total = bsm_ea_sphere_total_influence } }

			if = {
				limit = { check_variable = { t_total > 0 } }
				for_each_loop = {
					array = global.bsm_ea_member_list_@var:v_token
					value = v_mem3
					var:v_mem3 = {
						add_to_variable = {
							var = economic_alliance_pp_buff_value
							value = {
								value = bsm_ea_influence
								divide = global.bsm_ea_g_total_inf
								clamp = { min = 0 max = 0.05 }
							}
						}

						# モディファイア再付与（スタック防止）
						if = {
							limit = { has_dynamic_modifier = { modifier = bsm_economic_alliance_pp_buff } }
							remove_dynamic_modifier = { modifier = bsm_economic_alliance_pp_buff }
						}
						add_dynamic_modifier = {
							modifier = bsm_economic_alliance_pp_buff
							days = 15
						}
					}
				}
			}
		}
	}
}
