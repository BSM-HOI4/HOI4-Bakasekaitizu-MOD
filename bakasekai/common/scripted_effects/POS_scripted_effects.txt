    }

    }
}

POS_sell_title_effect_barony = {
    random_other_country = {
        limit = {
            is_major = no
            NOT = { tag = POS }
            NOT = { has_war_with = POS }
        }
        POS = {
            add_opinion_modifier = {
                target = PREV
                modifier = POS_title_sold_barony
            }
        }
        add_opinion_modifier = {
            target = POS
            modifier = POS_title_sold_barony
        }
    }
}

POS_grant_military_title_effect = {
    random_other_country = {
        limit = {
            has_army_size = { size > 0 }
            any_unit_leader = { is_unit_leader = yes }
            NOT = { tag = POS }
            NOT = { has_war_with = POS }
        }
        random_unit_leader = {
            limit = {
                is_unit_leader = yes
            }
            add_unit_leader_trait = desert_fox
        }
        add_opinion_modifier = {
            target = POS
            modifier = POS_title_military_grant
        }
        POS = {
            add_opinion_modifier = {
                target = PREV
                modifier = POS_title_military_grant
            }
        }
    }
}
