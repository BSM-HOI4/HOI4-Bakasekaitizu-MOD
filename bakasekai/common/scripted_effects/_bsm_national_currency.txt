# prefix: bsm_nc_
# ==========================================
#
# 通貨ID一覧:
#  0 = UC直結 (統一通貨)
#  1 = USD (米ドル)         2 = EUR (ユーロ)          3 = GBP (ポンド)
#  4 = JPY (円)             5 = CNY (人民元)          6 = RUB (ルーブル)
#  7 = KRW (ウォン)         8 = INR (ルピー)          9 = TRY (リラ)
# 10 = BRL (レアル)        11 = MXN (メキシコペソ)   12 = ARS (アルゼンチンペソ)
# 13 = SAR (サウジリアル)  14 = EGP (エジプスポンド)  15 = CHF (スイスフラン)
# 16 = SEK (クローナ)      17 = AUD (豪ドル)         18 = IDR (ルピア)
# 19 = THB (バーツ)        20 = NGN (ナイラ)         21 = ZAR (ランド)
# 22 = KES (東アフリカシリング) 23 = COP (コロンビアペソ) 24 = ILS (シェケル)
# 25 = TWD (台湾ドル)      26 = PHP (フィリピンペソ) 27 = MYR (リンギット)
# 28 = MAN (満州国円)      29 = Generic (汎用通貨)

# ==========================================
# 初期化: on_startup から every_country 内で呼び出し
# ==========================================
bsm_nc_init_currency = {
	set_variable = { bsm_nc_id = 0 }
	set_variable = { bsm_nc_base_rate = 1.0 }
	set_variable = { bsm_nc_rate = 1.0 }
	set_variable = { bsm_nc_rate_prev = 1.0 }
	set_variable = { bsm_nc_stability = 100 }
	set_variable = { bsm_nc_inflation = 0 }
	set_variable = { bsm_nc_rate_delta = 0 }
	bsm_nc_assign_currency = yes
	set_variable = { bsm_nc_original_id = bsm_nc_id }
	set_variable = { bsm_nc_rate = bsm_nc_base_rate }
	set_variable = { bsm_nc_rate_prev = bsm_nc_base_rate }
}

# ==========================================
# 月次為替レート計算
# ucs_on_monthly と同じタイミングで呼び出し
# ==========================================
bsm_nc_monthly_update = {
	# UC直結国はスキップ
	if = {
		limit = { check_variable = { bsm_nc_id = 0 } }
	}
	else = {
		# --- 1. 基本ファクター ---
		set_temp_variable = { t_nc_factor = 1.0 }

		# 安定度 (0.0-1.0) → 通貨強度寄与
		set_temp_variable = { t_nc_stab_mod = stability }
		subtract_from_temp_variable = { t_nc_stab_mod = 0.5 }
		multiply_temp_variable = { t_nc_stab_mod = 0.1 }
		add_to_temp_variable = { t_nc_factor = t_nc_stab_mod }

		# --- 2. 戦争ペナルティ ---
		if = {
			limit = { has_war = yes }
			multiply_temp_variable = { t_nc_factor = 0.95 }
			if = {
				limit = { check_variable = { ucs_war_month_counter > 6 } }
				multiply_temp_variable = { t_nc_factor = 0.92 }
			}
			if = {
				limit = { check_variable = { ucs_war_month_counter > 12 } }
				multiply_temp_variable = { t_nc_factor = 0.95 }
			}
		}

		# --- 3. UC残高ボーナス ---
		if = {
			limit = {
				has_variable = ucs_required_ec
				check_variable = { Unified_Currency > ucs_required_ec }
			}
			multiply_temp_variable = { t_nc_factor = 1.02 }
		}
		if = {
			limit = { check_variable = { Unified_Currency < 0 } }
			multiply_temp_variable = { t_nc_factor = 0.93 }
		}

		# --- 4. EA Goal03 通貨同盟ボーナス ---
		if = {
			limit = { has_idea = bsm_ea_goal_03_currency }
			multiply_temp_variable = { t_nc_factor = 1.05 }
		}

		# --- 5. 文明度ボーナス ---
		if = {
			limit = {
				has_variable = Cultural_Degree
				check_variable = { Cultural_Degree > 0.5 }
			}
			multiply_temp_variable = { t_nc_factor = 1.01 }
		}
		if = {
			limit = {
				has_variable = Cultural_Degree
				check_variable = { Cultural_Degree > 0.8 }
			}
			multiply_temp_variable = { t_nc_factor = 1.01 }
		}

		# --- 6. 従属国ペナルティ ---
		if = {
			limit = { is_subject = yes }
			multiply_temp_variable = { t_nc_factor = 0.96 }
		}

		# --- 7. 新レート計算 ---
		# factor > 1.0 → 通貨強 → レート低下 (UCに対して強い)
		# factor < 1.0 → 通貨弱 → レート上昇 (UCに対して弱い)
		set_variable = { bsm_nc_rate_prev = bsm_nc_rate }
		set_variable = { bsm_nc_rate = bsm_nc_base_rate }
		divide_variable = { bsm_nc_rate = t_nc_factor }

		# --- 8. クランプ (基準の0.3倍〜3.0倍) ---
		set_temp_variable = { t_nc_min = bsm_nc_base_rate }
		multiply_temp_variable = { t_nc_min = 0.3 }
		set_temp_variable = { t_nc_max = bsm_nc_base_rate }
		multiply_temp_variable = { t_nc_max = 3.0 }
		clamp_variable = { var = bsm_nc_rate min = t_nc_min max = t_nc_max }

		# --- 9. 前月比デルタ ---
		set_temp_variable = { t_nc_delta = bsm_nc_rate }
		subtract_from_temp_variable = { t_nc_delta = bsm_nc_rate_prev }
		set_variable = { bsm_nc_rate_delta = t_nc_delta }

		# --- 10. インフレ計算 ---
		set_temp_variable = { t_nc_infl = bsm_nc_rate }
		divide_temp_variable = { t_nc_infl = bsm_nc_rate_prev }
		subtract_from_temp_variable = { t_nc_infl = 1.0 }
		multiply_temp_variable = { t_nc_infl = 100.0 }
		set_variable = { bsm_nc_inflation = t_nc_infl }

		# --- 11. 通貨安定度 (0-100) ---
		set_temp_variable = { t_nc_stability = 100 }

		if = {
			limit = { check_variable = { bsm_nc_inflation > 5 } }
			subtract_from_temp_variable = { t_nc_stability = 15 }
		}
		if = {
			limit = { check_variable = { bsm_nc_inflation > 15 } }
			subtract_from_temp_variable = { t_nc_stability = 25 }
		}
		if = {
			limit = { check_variable = { bsm_nc_inflation > 30 } }
			subtract_from_temp_variable = { t_nc_stability = 30 }
		}
		if = {
			limit = { has_war = yes }
			subtract_from_temp_variable = { t_nc_stability = 20 }
		}
		if = {
			limit = { check_variable = { Unified_Currency < 0 } }
			subtract_from_temp_variable = { t_nc_stability = 20 }
		}
		if = {
			limit = { is_subject = yes }
			subtract_from_temp_variable = { t_nc_stability = 10 }
		}
		clamp_temp_variable = { var = t_nc_stability min = 0 max = 100 }
		set_variable = { bsm_nc_stability = t_nc_stability }

		# --- 11b. 通貨強度カテゴリ (0=強い 1=普通 2=弱い) ---
		set_variable = { bsm_nc_strength_cat = 1 }
		set_variable = { bsm_nc_diff_strong = bsm_nc_base_rate }
		multiply_variable = { bsm_nc_diff_strong = 0.9 }
		subtract_from_variable = { bsm_nc_diff_strong = bsm_nc_rate }
		if = {
			limit = { check_variable = { bsm_nc_diff_strong > 0 } }
			set_variable = { bsm_nc_strength_cat = 0 }
		}
		set_variable = { bsm_nc_diff_weak = bsm_nc_base_rate }
		multiply_variable = { bsm_nc_diff_weak = 1.1 }
		subtract_from_variable = { bsm_nc_diff_weak = bsm_nc_rate }
		if = {
			limit = { check_variable = { bsm_nc_diff_weak < 0 } }
			set_variable = { bsm_nc_strength_cat = 2 }
		}

		# --- 12. UC獲得補正 (dynamic_modifier) ---
		if = {
			limit = { check_variable = { bsm_nc_stability < 30 } }
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = bsm_nc_crisis_modifier } } }
				add_dynamic_modifier = { modifier = bsm_nc_crisis_modifier }
			}
		}
		else = {
			if = {
				limit = { has_dynamic_modifier = { modifier = bsm_nc_crisis_modifier } }
				remove_dynamic_modifier = { modifier = bsm_nc_crisis_modifier }
			}
		}

		# --- 13. イベント発火判定 ---
		# 通貨危機
		if = {
			limit = {
				check_variable = { bsm_nc_inflation > 20 }
				NOT = { has_country_flag = nc_crisis_cooldown }
			}
			random_list = {
				20 = { country_event = { id = nc.1 days = 1 } }
				80 = { }
			}
			set_country_flag = { flag = nc_crisis_cooldown days = 180 }
		}
		# デフォルト
		if = {
			limit = {
				check_variable = { Unified_Currency < 0 }
				NOT = { has_country_flag = nc_default_cooldown }
			}
			random_list = {
				30 = { country_event = { id = nc.2 days = 1 } }
				70 = { }
			}
			set_country_flag = { flag = nc_default_cooldown days = 360 }
		}
		# 評価引き上げ
		if = {
			limit = {
				check_variable = { bsm_nc_stability > 90 }
				NOT = { has_country_flag = nc_upgrade_cooldown }
			}
			random_list = {
				15 = { country_event = { id = nc.3 days = 1 } }
				85 = { }
			}
			set_country_flag = { flag = nc_upgrade_cooldown days = 365 }
		}
	}
}

# _bsm_Unified_Currency.txt の temp_factor に加算して使用
# ==========================================
bsm_nc_calc_uc_modifier = {
	set_temp_variable = { t_nc_uc_mod = 0 }

	if = {
		limit = {
			has_variable = bsm_nc_id
			NOT = { check_variable = { bsm_nc_id = 0 } }
		}
		# (base_rate / current_rate - 1.0) * 0.3
		set_temp_variable = { t_nc_uc_mod = bsm_nc_base_rate }
		divide_temp_variable = { t_nc_uc_mod = bsm_nc_rate }
		subtract_from_temp_variable = { t_nc_uc_mod = 1.0 }
		multiply_temp_variable = { t_nc_uc_mod = 0.3 }
	}
}
