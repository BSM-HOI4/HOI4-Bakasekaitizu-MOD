
# ==========================================
# BSM 国家通貨システム (National Currency)
# prefix: bsm_nc_
# ==========================================
#
# 通貨ID一覧:
#  0 = UC直結 (統一通貨)
#  1 = USD (米ドル)         2 = EUR (ユーロ)          3 = GBP (ポンド)
#  4 = JPY (円)             5 = CNY (人民元)          6 = RUB (ルーブル)
#  7 = KRW (ウォン)         8 = INR (ルピー)          9 = TRY (リラ)
# 10 = BRL (レアル)        11 = MXN (メキシコペソ)   12 = ARS (アルゼンチンペソ)
# 13 = SAR (サウジリアル)  14 = EGP (エジプスポンド)  15 = CHF (スイスフラン)
# 16 = SEK (クローナ)      17 = AUD (豪ドル)         18 = IDR (ルピア)
# 19 = THB (バーツ)        20 = NGN (ナイラ)         21 = ZAR (ランド)
# 22 = KES (東アフリカシリング) 23 = COP (コロンビアペソ) 24 = ILS (シェケル)
# 25 = TWD (台湾ドル)      26 = PHP (フィリピンペソ) 27 = MYR (リンギット)
# 28 = MAN (満州国円)
# 30 = 固有国家通貨 (上記に該当しない全国家。表示は bsm_nc_tok=token:nc_<TAG> で nc_<TAG>_name/_sym を解決)

# ==========================================
# 初期化: on_startup から every_country 内で呼び出し
# ==========================================
bsm_nc_init_currency = {
	set_variable = { bsm_nc_id = 0 }
	set_variable = { bsm_nc_tok = 0 }
	set_variable = { bsm_nc_base_rate = 1.0 }
	set_variable = { bsm_nc_rate = 1.0 }
	set_variable = { bsm_nc_rate_prev = 1.0 }
	set_variable = { bsm_nc_stability = 100 }
	set_variable = { bsm_nc_inflation = 0 }
	set_variable = { bsm_nc_rate_delta = 0 }
	set_variable = { bsm_nc_strength_cat = 1 }
	bsm_nc_assign_currency = yes
	set_variable = { bsm_nc_original_id = bsm_nc_id }
	set_variable = { bsm_nc_rate = bsm_nc_base_rate }
	set_variable = { bsm_nc_rate_prev = bsm_nc_base_rate }
}

# ==========================================
# 開始時傀儡国の通貨継承
# 宗主国と同じ通貨を使用させる
# on_startup の通貨初期化(全国)後に every_country 内で呼び出すこと
# ==========================================
bsm_nc_inherit_overlord_currency = {
	if = {
		limit = {
			is_subject = yes
			exists = yes
		}
		set_temp_variable = { t_ov_id = 0 }
		set_temp_variable = { t_ov_base = 1.0 }
		set_temp_variable = { t_ov_tok = 0 }
		overlord = {
			set_temp_variable = { PREV.t_ov_id = bsm_nc_id }
			set_temp_variable = { PREV.t_ov_base = bsm_nc_base_rate }
			set_temp_variable = { PREV.t_ov_tok = bsm_nc_tok }
		}
		set_variable = { bsm_nc_id = t_ov_id }
		set_variable = { bsm_nc_tok = t_ov_tok }
		set_variable = { bsm_nc_base_rate = t_ov_base }
		set_variable = { bsm_nc_original_id = t_ov_id }
		set_variable = { bsm_nc_rate = t_ov_base }
		set_variable = { bsm_nc_rate_prev = t_ov_base }
	}
}

# [退役 2026-06-20] bsm_nc_apply_strongest_for_sphere / bsm_nc_leader_apply_strongest
# 旧仕様は EA陣営内の id=29(generic) メンバーを陣営内最強通貨へ自動吸収していたが、
# 全国家が固有通貨(id=30)を持つ新設計では吸収対象(generic)が存在せず、
# かつ固有通貨を自動で上書きするのは主目的(各国固有通貨)に反し、脱退時の復元も無いため退役。
# 意図的な通貨共有は opt-in の通貨同盟(EA goal_03 / bsm_nc_currency_union_*)に一本化。
# effect 定義と on_actions の呼び出しパス(開始時/月次)も併せて削除済み。
# ==========================================

# ==========================================
# 通貨同盟: 初期化/解除/収斂管理
# 入力(temp): t_nc_union_id, t_nc_union_rate, t_nc_union_base
# ==========================================
bsm_nc_currency_union_start = {
	if = {
		limit = { NOT = { has_variable = bsm_nc_original_id } }
		set_variable = { bsm_nc_original_id = bsm_nc_id }
	}
	clr_country_flag = bsm_nc_currency_union_optout
	set_variable = { bsm_nc_union_active = 1 }
	set_variable = { bsm_nc_union_anchor_id = t_nc_union_id }
	set_variable = { bsm_nc_union_anchor_base = t_nc_union_base }
	set_variable = { bsm_nc_union_anchor_rate = t_nc_union_rate }
	set_variable = { bsm_nc_id = t_nc_union_id }
	set_variable = { bsm_nc_base_rate = t_nc_union_base }
	set_variable = { bsm_nc_rate = t_nc_union_rate }
	set_variable = { bsm_nc_rate_prev = t_nc_union_rate }
	if = {
		limit = { NOT = { has_variable = bsm_nc_union_convergence } }
		set_variable = { bsm_nc_union_convergence = 50 }
	}
	if = {
		limit = { NOT = { has_variable = bsm_nc_union_exit_pressure } }
		set_variable = { bsm_nc_union_exit_pressure = 0 }
	}
}

bsm_nc_currency_union_end = {
	if = {
		limit = { has_variable = bsm_nc_original_id }
		set_variable = { bsm_nc_id = bsm_nc_original_id }
		bsm_nc_assign_currency = yes
		set_variable = { bsm_nc_rate = bsm_nc_base_rate }
		set_variable = { bsm_nc_rate_prev = bsm_nc_base_rate }
		set_variable = { bsm_nc_rate_delta = 0 }
	}
	clear_variable = bsm_nc_union_active
	clear_variable = bsm_nc_union_anchor_id
	clear_variable = bsm_nc_union_anchor_base
	clear_variable = bsm_nc_union_anchor_rate
	clear_variable = bsm_nc_union_convergence
	clear_variable = bsm_nc_union_exit_pressure
	clear_variable = bsm_nc_union_crisis_token
	if = {
		limit = { has_dynamic_modifier = { modifier = bsm_nc_union_converged_modifier } }
		remove_dynamic_modifier = { modifier = bsm_nc_union_converged_modifier }
	}
	if = {
		limit = { has_dynamic_modifier = { modifier = bsm_nc_union_strain_modifier } }
		remove_dynamic_modifier = { modifier = bsm_nc_union_strain_modifier }
	}
}

bsm_nc_currency_union_sync_to_leader = {
	if = {
		limit = {
			has_variable = bsm_ea_sphere
			check_variable = { global.bsm_ea_member_list_@var:bsm_ea_sphere^num > 0 }
		}
		set_temp_variable = { t_nc_union_token = bsm_ea_sphere }
		set_variable = { bsm_nc_union_crisis_token = t_nc_union_token }
		set_temp_variable = { t_nc_union_leader = global.bsm_ea_member_list_@var:t_nc_union_token^0 }
		var:t_nc_union_leader = {
			set_temp_variable = { PREV.t_nc_union_id = bsm_nc_id }
			set_temp_variable = { PREV.t_nc_union_base = bsm_nc_base_rate }
			set_temp_variable = { PREV.t_nc_union_rate = bsm_nc_rate }
		}
		if = {
			limit = {
				NOT = { check_variable = { t_nc_union_id = 0 } }
			}
			set_variable = { bsm_nc_union_anchor_id = t_nc_union_id }
			set_variable = { bsm_nc_union_anchor_base = t_nc_union_base }
			set_variable = { bsm_nc_union_anchor_rate = t_nc_union_rate }
			set_variable = { bsm_nc_id = t_nc_union_id }
			set_variable = { bsm_nc_base_rate = t_nc_union_base }
			if = {
				limit = { check_variable = { bsm_nc_union_convergence > 59 } }
				set_variable = { bsm_nc_rate_prev = bsm_nc_rate }
				set_variable = { bsm_nc_rate = t_nc_union_rate }
				# 前月比デルタ = 新レート − 旧レート  数式版（3op→1op、temp廃止）
				set_variable = {
					var = bsm_nc_rate_delta
					value = {
						value = bsm_nc_rate
						subtract = bsm_nc_rate_prev
					}
				}
			}
		}
	}
}

bsm_nc_currency_union_monthly = {
	if = {
		limit = { has_idea = bsm_ea_goal_03_currency }
		if = {
			limit = { NOT = { has_variable = bsm_nc_union_convergence } }
			set_variable = { bsm_nc_union_convergence = 50 }
		}
		if = {
			limit = { NOT = { has_variable = bsm_nc_union_exit_pressure } }
			set_variable = { bsm_nc_union_exit_pressure = 0 }
		}
		set_variable = { bsm_nc_union_active = 1 }
		bsm_nc_currency_union_sync_to_leader = yes

		if = {
			limit = { check_variable = { bsm_nc_stability > 79 } }
			add_to_variable = { bsm_nc_union_convergence = 8 }
		}
		else_if = {
			limit = { check_variable = { bsm_nc_stability > 59 } }
			add_to_variable = { bsm_nc_union_convergence = 4 }
		}
		else = {
			subtract_from_variable = { bsm_nc_union_convergence = 8 }
		}
		if = {
			limit = { check_variable = { bsm_nc_inflation > 10 } }
			subtract_from_variable = { bsm_nc_union_convergence = 10 }
		}
		else_if = {
			limit = { check_variable = { bsm_nc_inflation < 3 } }
			add_to_variable = { bsm_nc_union_convergence = 4 }
		}
		if = {
			limit = { check_variable = { Unified_Currency < 0 } }
			subtract_from_variable = { bsm_nc_union_convergence = 8 }
		}
		if = {
			limit = { has_war = yes }
			subtract_from_variable = { bsm_nc_union_convergence = 4 }
		}
		clamp_variable = { var = bsm_nc_union_convergence min = 0 max = 100 }

		if = {
			limit = { check_variable = { bsm_nc_union_convergence < 30 } }
			add_to_variable = { bsm_nc_union_exit_pressure = 18 }
		}
		else_if = {
			limit = { check_variable = { bsm_nc_union_convergence < 50 } }
			add_to_variable = { bsm_nc_union_exit_pressure = 8 }
		}
		else = {
			subtract_from_variable = { bsm_nc_union_exit_pressure = 6 }
		}
		if = {
			limit = { check_variable = { bsm_nc_stability < 40 } }
			add_to_variable = { bsm_nc_union_exit_pressure = 10 }
		}
		if = {
			limit = { check_variable = { bsm_nc_inflation > 15 } }
			add_to_variable = { bsm_nc_union_exit_pressure = 12 }
		}
		if = {
			limit = { check_variable = { Unified_Currency < 0 } }
			add_to_variable = { bsm_nc_union_exit_pressure = 8 }
		}
		if = {
			limit = { check_variable = { bsm_nc_union_convergence > 79 } }
			subtract_from_variable = { bsm_nc_union_exit_pressure = 10 }
		}
		clamp_variable = { var = bsm_nc_union_exit_pressure min = 0 max = 100 }

		if = {
			limit = { check_variable = { bsm_nc_union_convergence > 79 } }
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = bsm_nc_union_converged_modifier } } }
				add_dynamic_modifier = { modifier = bsm_nc_union_converged_modifier }
			}
		}
		else = {
			if = {
				limit = { has_dynamic_modifier = { modifier = bsm_nc_union_converged_modifier } }
				remove_dynamic_modifier = { modifier = bsm_nc_union_converged_modifier }
			}
		}
		if = {
			limit = { check_variable = { bsm_nc_union_exit_pressure > 49 } }
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = bsm_nc_union_strain_modifier } } }
				add_dynamic_modifier = { modifier = bsm_nc_union_strain_modifier }
			}
		}
		else = {
			if = {
				limit = { has_dynamic_modifier = { modifier = bsm_nc_union_strain_modifier } }
				remove_dynamic_modifier = { modifier = bsm_nc_union_strain_modifier }
			}
		}
		if = {
			limit = {
				check_variable = { bsm_nc_union_exit_pressure > 79 }
				NOT = { has_country_flag = bsm_nc_union_crisis_recent }
			}
			set_country_flag = { flag = bsm_nc_union_crisis_recent days = 180 }
			country_event = { id = nc.5 days = 1 }
		}
	}
	else_if = {
		limit = { has_variable = bsm_nc_union_active }
		bsm_nc_currency_union_end = yes
	}
}

# ==========================================
# 月次為替レート計算
# ucs_on_monthly と同じタイミングで呼び出し
# ==========================================
bsm_nc_monthly_update = {
	# UC直結国はスキップ
	if = {
		limit = { check_variable = { bsm_nc_id = 0 } }
	}
	else = {
		# --- 1. 基本ファクター ---
		# 安定度 (0.0-1.0) → 通貨強度寄与: 1 + ((stability - 0.5) * 0.1)
		set_temp_variable = {
			var = t_nc_factor
			value = {
				value = 1.0
				add = {
					value = stability
					subtract = 0.5
					multiply = 0.1
				}
			}
		}

		# --- 2. 戦争ペナルティ ---
		if = {
			limit = { has_war = yes }
			multiply_temp_variable = { t_nc_factor = 0.95 }
			if = {
				limit = { check_variable = { ucs_war_month_counter > 6 } }
				multiply_temp_variable = { t_nc_factor = 0.92 }
			}
			if = {
				limit = { check_variable = { ucs_war_month_counter > 12 } }
				multiply_temp_variable = { t_nc_factor = 0.95 }
			}
		}

		# --- 3. UC残高ボーナス ---
		if = {
			limit = {
				has_variable = ucs_required_ec
				check_variable = { Unified_Currency > ucs_required_ec }
			}
			multiply_temp_variable = { t_nc_factor = 1.02 }
		}
		if = {
			limit = { check_variable = { Unified_Currency < 0 } }
			multiply_temp_variable = { t_nc_factor = 0.93 }
		}

		# --- 4. EA Goal03 通貨同盟ボーナス ---
		if = {
			limit = { has_idea = bsm_ea_goal_03_currency }
			multiply_temp_variable = { t_nc_factor = 1.05 }
		}

		# --- 5. 文明度ボーナス ---
		if = {
			limit = {
				has_variable = Cultural_Degree
				check_variable = { Cultural_Degree > 0.5 }
			}
			multiply_temp_variable = { t_nc_factor = 1.01 }
		}
		if = {
			limit = {
				has_variable = Cultural_Degree
				check_variable = { Cultural_Degree > 0.8 }
			}
			multiply_temp_variable = { t_nc_factor = 1.01 }
		}

		# --- 6. 従属国ペナルティ ---
		if = {
			limit = { is_subject = yes }
			multiply_temp_variable = { t_nc_factor = 0.96 }
		}

		# --- 6b. ランダムノイズ (為替流動性) ---
		random_list = {
			25 = { add_to_temp_variable = { t_nc_factor = -0.02 } }
			25 = { add_to_temp_variable = { t_nc_factor = -0.01 } }
			25 = { add_to_temp_variable = { t_nc_factor = 0.01 } }
			25 = { add_to_temp_variable = { t_nc_factor = 0.02 } }
		}

		# --- 7. 新レート計算 ---
		# factor > 1.0 → 通貨強 → レート低下 (UCに対して強い)
		# factor < 1.0 → 通貨弱 → レート上昇 (UCに対して弱い)
		set_variable = { bsm_nc_rate_prev = bsm_nc_rate }
		set_variable = {
			var = bsm_nc_rate
			value = {
				value = bsm_nc_base_rate
				divide = t_nc_factor
				clamp = {
					min = { value = bsm_nc_base_rate  multiply = 0.3 }
					max = { value = bsm_nc_base_rate  multiply = 3.0 }
				}
			}
		}

		# --- 9. 前月比デルタ ---
		set_variable = {
			var = bsm_nc_rate_delta
			value = {
				value = bsm_nc_rate
				subtract = bsm_nc_rate_prev
			}
		}

		# --- 10. インフレ計算 ---
		set_variable = {
			var = bsm_nc_inflation
			value = {
				value = bsm_nc_rate
				divide = bsm_nc_rate_prev
				subtract = 1.0
				multiply = 100.0
			}
		}

		# --- 11. 通貨安定度 (0-100) ---
		set_temp_variable = {
			var = t_nc_stability
			value = {
				value = 100
				subtract = { value = bsm_nc_inflation  greater_than = 5  multiply = 15 }
				subtract = { value = bsm_nc_inflation  greater_than = 15  multiply = 25 }
				subtract = { value = bsm_nc_inflation  greater_than = 30  multiply = 30 }
			}
		}
		if = {
			limit = { has_war = yes }
			subtract_from_temp_variable = { t_nc_stability = 20 }
		}
		if = {
			limit = { check_variable = { Unified_Currency < 0 } }
			subtract_from_temp_variable = { t_nc_stability = 20 }
		}
		if = {
			limit = { is_subject = yes }
			subtract_from_temp_variable = { t_nc_stability = 10 }
		}
		clamp_temp_variable = { var = t_nc_stability min = 0 max = 100 }
		set_variable = { bsm_nc_stability = t_nc_stability }

		# --- 11b. 通貨強度カテゴリ (0=強い 1=普通 2=弱い) ---
		set_variable = { bsm_nc_strength_cat = 1 }
		set_variable = {
			var = bsm_nc_diff_strong
			value = {
				value = bsm_nc_base_rate
				multiply = 0.9
				subtract = bsm_nc_rate
			}
		}
		if = {
			limit = { check_variable = { bsm_nc_diff_strong > 0 } }
			set_variable = { bsm_nc_strength_cat = 0 }
		}
		set_variable = {
			var = bsm_nc_diff_weak
			value = {
				value = bsm_nc_base_rate
				multiply = 1.1
				subtract = bsm_nc_rate
			}
		}
		if = {
			limit = { check_variable = { bsm_nc_diff_weak < 0 } }
			set_variable = { bsm_nc_strength_cat = 2 }
		}

		# --- 12. UC獲得補正 & 通貨弱体化 modifier ---
		if = {
			limit = { check_variable = { bsm_nc_stability < 30 } }
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = bsm_nc_crisis_modifier } } }
				add_dynamic_modifier = { modifier = bsm_nc_crisis_modifier }
			}
		}
		else = {
			if = {
				limit = { has_dynamic_modifier = { modifier = bsm_nc_crisis_modifier } }
				remove_dynamic_modifier = { modifier = bsm_nc_crisis_modifier }
			}
		}
		# --- 12b. 通貨弱体化 modifier (rate > base * 1.5) ---
		set_temp_variable = {
			var = t_nc_vw_threshold
			value = { value = bsm_nc_base_rate  multiply = 1.5 }
		}
		if = {
			limit = { check_variable = { bsm_nc_rate > t_nc_vw_threshold } }
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = bsm_nc_very_weak_modifier } } }
				add_dynamic_modifier = { modifier = bsm_nc_very_weak_modifier }
			}
			if = {
				limit = { has_dynamic_modifier = { modifier = bsm_nc_weak_modifier } }
				remove_dynamic_modifier = { modifier = bsm_nc_weak_modifier }
			}
		}
		else = {
			if = {
				limit = { has_dynamic_modifier = { modifier = bsm_nc_very_weak_modifier } }
				remove_dynamic_modifier = { modifier = bsm_nc_very_weak_modifier }
			}
		}
		# --- 12c. 通貨弱化 modifier (rate > base * 1.1) ---
		set_temp_variable = {
			var = t_nc_w_threshold
			value = { value = bsm_nc_base_rate  multiply = 1.1 }
		}
		if = {
			limit = {
				check_variable = { bsm_nc_rate > t_nc_w_threshold }
				NOT = { has_dynamic_modifier = { modifier = bsm_nc_very_weak_modifier } }
			}
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = bsm_nc_weak_modifier } } }
				add_dynamic_modifier = { modifier = bsm_nc_weak_modifier }
			}
		}
		else = {
			if = {
				limit = { has_dynamic_modifier = { modifier = bsm_nc_weak_modifier } }
				remove_dynamic_modifier = { modifier = bsm_nc_weak_modifier }
			}
		}

	}
}

	if = {
		limit = {
			has_variable = bsm_nc_id
			NOT = { check_variable = { bsm_nc_id = 0 } }
		}
		set_variable = {
			var = bsm_nc_rate
			value = {
				value = bsm_nc_rate
				multiply = 1.3
				clamp = {
					min = { value = bsm_nc_base_rate  multiply = 0.3 }
					max = { value = bsm_nc_base_rate  multiply = 3.0 }
				}
			}
		}
		set_variable = {
			var = bsm_nc_stability
			value = {
				value = bsm_nc_stability
				subtract = 25
				clamp = { min = 0 max = 100 }
			}
		}
	}
}

# ==========================================
# UC獲得補正値を返す (temp変数 t_nc_uc_mod)
# _bsm_Unified_Currency.txt の temp_factor に加算して使用
# ==========================================
bsm_nc_calc_uc_modifier = {
	set_temp_variable = { t_nc_uc_mod = 0 }

	if = {
		limit = {
			has_variable = bsm_nc_id
			NOT = { check_variable = { bsm_nc_id = 0 } }
		}
		# (base_rate / current_rate - 1.0) * 0.3
		set_temp_variable = {
			var = t_nc_uc_mod
			value = {
				value = bsm_nc_base_rate
				divide = bsm_nc_rate
				subtract = 1.0
				multiply = 0.3
			}
		}
	}
}
