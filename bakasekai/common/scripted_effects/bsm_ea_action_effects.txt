# prefix: bsm_ea_

# ------------------------------------------
# ODA（政府開発援助）UC:50
# ------------------------------------------
bsm_ea_action_oda = {
	if = {
		limit = {
			check_variable = { Unified_Currency > 49 }
			check_variable = { bsm_ea_has_selected_member = 1 }
			check_variable = { bsm_ea_cd_oda = 0 }
		}
		add_to_variable = { Unified_Currency = -50 }
		set_variable = { bsm_ea_cd_oda = 90 }
		event_target:bsm_ea_selected_member = {
			add_to_variable = { Unified_Currency = 50 }
			add_stability = 0.01
		}
		add_opinion_modifier = {
			target = event_target:bsm_ea_selected_member
			modifier = bsm_economic_alliance_relation_boost
		}
	}
}

# ------------------------------------------
# 負債肩代わり UC:300 → 対象の政治力+安定度回復
# ------------------------------------------
bsm_ea_action_debt_relief = {
	if = {
		limit = {
			check_variable = { Unified_Currency > 299 }
			check_variable = { bsm_ea_has_selected_member = 1 }
			check_variable = { bsm_ea_cd_debt_relief = 0 }
		}
		add_to_variable = { Unified_Currency = -300 }
		set_variable = { bsm_ea_cd_debt_relief = 365 }
		event_target:bsm_ea_selected_member = {
			add_political_power = 100
			add_stability = 0.03
			if = {
				limit = { has_idea = recently_left_economic_alliance }
				remove_ideas = recently_left_economic_alliance
			}
		}
	}
}
