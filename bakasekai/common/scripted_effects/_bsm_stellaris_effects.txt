# ------------------------------------------
# 初期化
# ------------------------------------------
bsm_stellaris_init = {
  if = {
    limit = { NOT = { has_global_flag = bsm_stellaris_initialized } }
    set_global_flag = bsm_stellaris_initialized

    every_country = {
      bsm_stellaris_country_init = yes
    }
  }
}

bsm_stellaris_country_init = {
  # 7伝統ツリー進捗 (0=未採用 1=採用 2-6=ノード1-5 7=完成)
  set_variable = { bsm_stellaris_tradition_exploration = 0 }
  set_variable = { bsm_stellaris_tradition_expansion = 0 }
  set_variable = { bsm_stellaris_tradition_prosperity = 0 }
  set_variable = { bsm_stellaris_tradition_armaments = 0 }
  set_variable = { bsm_stellaris_tradition_diplomacy = 0 }
  set_variable = { bsm_stellaris_tradition_governance = 0 }
  set_variable = { bsm_stellaris_tradition_harmony = 0 }
  set_variable = { bsm_stellaris_tradition_adopted_count = 0 }
  set_variable = { bsm_stellaris_tradition_cost = 100 }

  set_variable = { bsm_stellaris_ascension_adopted = 0 }
  set_variable = { bsm_stellaris_ascension_slots = 0 }
  set_variable = { bsm_stellaris_ascension_slots_max = 0 }
  set_variable = { bsm_stellaris_ascension_perks_taken = 0 }
  # パーク解放に必要な完成ツリー数（表示用）
  set_variable = { bsm_stellaris_tradition_required = 1 }

  set_variable = { bsm_stellaris_essence_energy = 0 }
  set_variable = { bsm_stellaris_essence_minerals = 0 }
  set_variable = { bsm_stellaris_essence_info = 0 }

  set_variable = { bsm_stellaris_trait_slots = 0 }
  set_variable = { bsm_stellaris_traits_equipped = 0 }

  set_variable = { bsm_stellaris_anomaly_research_bonus = 0 }
  set_variable = { bsm_stellaris_anomaly_research_field = 0 }
  set_variable = { bsm_stellaris_anomaly_difficulty_level = 0 }
  set_variable = { bsm_stellaris_anomaly_stock_count = 0 }

  set_variable = { bsm_stellaris_expedition_count = 0 }
  set_variable = { bsm_stellaris_expedition_active = 0 }

  bsm_stellaris_update_trait_slots = yes
}

# ------------------------------------------
# 特性スロット更新
# ------------------------------------------
bsm_stellaris_update_trait_slots = {
  set_variable = { bsm_stellaris_trait_slots = 0 }

  if = {
    limit = { check_variable = { var = Cultural_Degree value = 30 compare = greater_than_or_equals } }
    set_variable = { bsm_stellaris_trait_slots = 1 }
  }
  if = {
    limit = { check_variable = { var = Cultural_Degree value = 50 compare = greater_than_or_equals } }
    set_variable = { bsm_stellaris_trait_slots = 3 }
  }
  if = {
    limit = { check_variable = { var = Cultural_Degree value = 80 compare = greater_than_or_equals } }
    set_variable = { bsm_stellaris_trait_slots = 5 }
  }
  if = {
    limit = { check_variable = { var = Cultural_Degree value = 100 compare = greater_than_or_equals } }
    set_variable = { bsm_stellaris_trait_slots = 6 }
  }
}

# ------------------------------------------
# 有効伝統コスト算出
#   bsm_stellaris_tradition_cost = 表示用・基本コスト（割引なし）
#   t_cost (temp)               = 実課金額（思想適性なら25%割引）
# bsm_stellaris_tradition_category (1-7) が設定済みであること
# ------------------------------------------
bsm_stellaris_set_effective_tradition_cost = {
  bsm_stellaris_update_tradition_count = yes
  set_temp_variable = { t_cost = bsm_stellaris_tradition_adopted_count }
  multiply_temp_variable = { t_cost = 25 }
  add_to_temp_variable = { t_cost = 100 }
  clamp_temp_variable = { t_cost = 100 max = 600 }
  set_variable = { bsm_stellaris_tradition_cost = t_cost }
  if = {
    limit = { bsm_stellaris_has_current_tradition_affinity = yes }
    multiply_temp_variable = { t_cost = 0.75 }
  }
}

# ------------------------------------------
# 伝統1段取得（統合力消費）
# bsm_stellaris_tradition_category (1=探索 2=拡張 3=繁栄 4=軍備 5=外交 6=統治 7=調和)
# ------------------------------------------
bsm_stellaris_unlock_tradition = {
  if = {
    limit = {
      has_variable = bsm_stellaris_tradition_category
      has_variable = National_Unity_Power
    }
    bsm_stellaris_set_effective_tradition_cost = yes

    if = {
      limit = { check_variable = { var = National_Unity_Power value = t_cost compare = greater_than_or_equals } }

      # --- 1 探索 ---

        limit = {
          check_variable = { var = bsm_stellaris_tradition_category value = 1 compare = equals }
          check_variable = { var = bsm_stellaris_tradition_exploration value = 7 compare = less_than }
        }
        subtract_from_variable = { var = National_Unity_Power value = t_cost }
        add_to_variable = { bsm_stellaris_tradition_exploration = 1 }
        if = { limit = { check_variable = { var = bsm_stellaris_tradition_exploration value = 1 compare = equals } } add_ideas = bsm_stellaris_tradition_exploration_adopt }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_exploration value = 2 compare = equals } } add_ideas = bsm_stellaris_tradition_exploration_1 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_exploration value = 3 compare = equals } } add_ideas = bsm_stellaris_tradition_exploration_2 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_exploration value = 4 compare = equals } } add_ideas = bsm_stellaris_tradition_exploration_3 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_exploration value = 5 compare = equals } } add_ideas = bsm_stellaris_tradition_exploration_4 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_exploration value = 6 compare = equals } } add_ideas = bsm_stellaris_tradition_exploration_5 }
        else = { add_ideas = bsm_stellaris_tradition_exploration_finish }

      # --- 2 拡張 ---
      else_if = {
        limit = {
          check_variable = { var = bsm_stellaris_tradition_category value = 2 compare = equals }
          check_variable = { var = bsm_stellaris_tradition_expansion value = 7 compare = less_than }

        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_expansion value = 2 compare = equals } } add_ideas = bsm_stellaris_tradition_expansion_1 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_expansion value = 3 compare = equals } } add_ideas = bsm_stellaris_tradition_expansion_2 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_expansion value = 4 compare = equals } } add_ideas = bsm_stellaris_tradition_expansion_3 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_expansion value = 5 compare = equals } } add_ideas = bsm_stellaris_tradition_expansion_4 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_expansion value = 6 compare = equals } } add_ideas = bsm_stellaris_tradition_expansion_5 }
        else = { add_ideas = bsm_stellaris_tradition_expansion_finish }
      }
      # --- 3 繁栄 ---
      else_if = {
        limit = {
          check_variable = { var = bsm_stellaris_tradition_category value = 3 compare = equals }
          check_variable = { var = bsm_stellaris_tradition_prosperity value = 7 compare = less_than }
        }
        subtract_from_variable = { var = National_Unity_Power value = t_cost }
        add_to_variable = { bsm_stellaris_tradition_prosperity = 1 }
        if = { limit = { check_variable = { var = bsm_stellaris_tradition_prosperity value = 1 compare = equals } } add_ideas = bsm_stellaris_tradition_prosperity_adopt }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_prosperity value = 2 compare = equals } } add_ideas = bsm_stellaris_tradition_prosperity_1 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_prosperity value = 3 compare = equals } } add_ideas = bsm_stellaris_tradition_prosperity_2 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_prosperity value = 4 compare = equals } } add_ideas = bsm_stellaris_tradition_prosperity_3 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_prosperity value = 5 compare = equals } } add_ideas = bsm_stellaris_tradition_prosperity_4 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_prosperity value = 6 compare = equals } } add_ideas = bsm_stellaris_tradition_prosperity_5 }
        else = { add_ideas = bsm_stellaris_tradition_prosperity_finish }
      }
      # --- 4 軍備 ---
      else_if = {
        limit = {
          check_variable = { var = bsm_stellaris_tradition_category value = 4 compare = equals }
          check_variable = { var = bsm_stellaris_tradition_armaments value = 7 compare = less_than }
        }

        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_armaments value = 2 compare = equals } } add_ideas = bsm_stellaris_tradition_armaments_1 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_armaments value = 3 compare = equals } } add_ideas = bsm_stellaris_tradition_armaments_2 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_armaments value = 4 compare = equals } } add_ideas = bsm_stellaris_tradition_armaments_3 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_armaments value = 5 compare = equals } } add_ideas = bsm_stellaris_tradition_armaments_4 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_armaments value = 6 compare = equals } } add_ideas = bsm_stellaris_tradition_armaments_5 }
        else = { add_ideas = bsm_stellaris_tradition_armaments_finish }
      }
      # --- 5 外交 ---

          check_variable = { var = bsm_stellaris_tradition_category value = 5 compare = equals }
          check_variable = { var = bsm_stellaris_tradition_diplomacy value = 7 compare = less_than }
        }
        subtract_from_variable = { var = National_Unity_Power value = t_cost }
        add_to_variable = { bsm_stellaris_tradition_diplomacy = 1 }
        if = { limit = { check_variable = { var = bsm_stellaris_tradition_diplomacy value = 1 compare = equals } } add_ideas = bsm_stellaris_tradition_diplomacy_adopt }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_diplomacy value = 2 compare = equals } } add_ideas = bsm_stellaris_tradition_diplomacy_1 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_diplomacy value = 3 compare = equals } } add_ideas = bsm_stellaris_tradition_diplomacy_2 }

        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_diplomacy value = 5 compare = equals } } add_ideas = bsm_stellaris_tradition_diplomacy_4 }

        limit = {
          check_variable = { var = bsm_stellaris_tradition_category value = 6 compare = equals }
          check_variable = { var = bsm_stellaris_tradition_governance value = 7 compare = less_than }
        }
        subtract_from_variable = { var = National_Unity_Power value = t_cost }
        add_to_variable = { bsm_stellaris_tradition_governance = 1 }
        if = { limit = { check_variable = { var = bsm_stellaris_tradition_governance value = 1 compare = equals } } add_ideas = bsm_stellaris_tradition_governance_adopt }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_governance value = 2 compare = equals } } add_ideas = bsm_stellaris_tradition_governance_1 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_governance value = 3 compare = equals } } add_ideas = bsm_stellaris_tradition_governance_2 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_governance value = 4 compare = equals } } add_ideas = bsm_stellaris_tradition_governance_3 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_governance value = 5 compare = equals } } add_ideas = bsm_stellaris_tradition_governance_4 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_governance value = 6 compare = equals } } add_ideas = bsm_stellaris_tradition_governance_5 }
        else = { add_ideas = bsm_stellaris_tradition_governance_finish }
      }
      # --- 7 調和 ---
      else_if = {
        limit = {
          check_variable = { var = bsm_stellaris_tradition_category value = 7 compare = equals }
          check_variable = { var = bsm_stellaris_tradition_harmony value = 7 compare = less_than }
        }
        subtract_from_variable = { var = National_Unity_Power value = t_cost }
        add_to_variable = { bsm_stellaris_tradition_harmony = 1 }
        if = { limit = { check_variable = { var = bsm_stellaris_tradition_harmony value = 1 compare = equals } } add_ideas = bsm_stellaris_tradition_harmony_adopt }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_harmony value = 2 compare = equals } } add_ideas = bsm_stellaris_tradition_harmony_1 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_harmony value = 3 compare = equals } } add_ideas = bsm_stellaris_tradition_harmony_2 }

# ------------------------------------------
# 次回伝統コスト更新（表示用・適性は加味しない概算）
# ------------------------------------------
bsm_stellaris_update_tradition_cost = {
  bsm_stellaris_update_tradition_count = yes
  set_variable = { bsm_stellaris_tradition_cost = 100 }
  set_temp_variable = { t_tradition_cost_growth = bsm_stellaris_tradition_adopted_count }
  multiply_temp_variable = { t_tradition_cost_growth = 25 }
  add_to_variable = { bsm_stellaris_tradition_cost = t_tradition_cost_growth }
  clamp_variable = { bsm_stellaris_tradition_cost = 100 max = 600 }
}

# ------------------------------------------
# 伝統取得数・完成数・アセンションスロット更新
# ------------------------------------------
bsm_stellaris_update_tradition_count = {
  set_temp_variable = { t_total = 0 }
  add_to_temp_variable = { t_total = bsm_stellaris_tradition_exploration }
  add_to_temp_variable = { t_total = bsm_stellaris_tradition_expansion }
  add_to_temp_variable = { t_total = bsm_stellaris_tradition_prosperity }
  add_to_temp_variable = { t_total = bsm_stellaris_tradition_armaments }
  add_to_temp_variable = { t_total = bsm_stellaris_tradition_diplomacy }
  add_to_temp_variable = { t_total = bsm_stellaris_tradition_governance }
  add_to_temp_variable = { t_total = bsm_stellaris_tradition_harmony }
  set_variable = { bsm_stellaris_tradition_adopted_count = t_total }
  set_variable = { bsm_stellaris_ascension_adopted = t_total }

  # 完成ツリー数 = アセンションスロット上限
  set_temp_variable = { t_finished = 0 }
  if = { limit = { check_variable = { var = bsm_stellaris_tradition_exploration value = 7 compare = greater_than_or_equals } } add_to_temp_variable = { t_finished = 1 } }
  if = { limit = { check_variable = { var = bsm_stellaris_tradition_expansion value = 7 compare = greater_than_or_equals } } add_to_temp_variable = { t_finished = 1 } }
  if = { limit = { check_variable = { var = bsm_stellaris_tradition_prosperity value = 7 compare = greater_than_or_equals } } add_to_temp_variable = { t_finished = 1 } }
  if = { limit = { check_variable = { var = bsm_stellaris_tradition_armaments value = 7 compare = greater_than_or_equals } } add_to_temp_variable = { t_finished = 1 } }
  if = { limit = { check_variable = { var = bsm_stellaris_tradition_diplomacy value = 7 compare = greater_than_or_equals } } add_to_temp_variable = { t_finished = 1 } }
  if = { limit = { check_variable = { var = bsm_stellaris_tradition_governance value = 7 compare = greater_than_or_equals } } add_to_temp_variable = { t_finished = 1 } }
  if = { limit = { check_variable = { var = bsm_stellaris_tradition_harmony value = 7 compare = greater_than_or_equals } } add_to_temp_variable = { t_finished = 1 } }
  set_variable = { bsm_stellaris_ascension_slots_max = t_finished }

  # 利用可能スロット = 上限 - 取得済みパーク
  set_temp_variable = { t_slots = t_finished }
  subtract_from_temp_variable = { t_slots = bsm_stellaris_ascension_perks_taken }
  clamp_temp_variable = { t_slots = 0 max = 8 }
  set_variable = { bsm_stellaris_ascension_slots = t_slots }
}

# ------------------------------------------
# アセンションパーク取得
# ------------------------------------------
bsm_stellaris_unlock_ascension_park = {
  if = {
    limit = {
      has_variable = bsm_stellaris_ascension_park_token
      has_variable = bsm_stellaris_ascension_slots
      check_variable = { var = bsm_stellaris_ascension_slots value = 0 compare = greater_than }
      NOT = { has_idea = var:bsm_stellaris_ascension_park_token }
    }
    add_ideas = var:bsm_stellaris_ascension_park_token
    add_to_variable = { bsm_stellaris_ascension_perks_taken = 1 }
    bsm_stellaris_update_tradition_count = yes
  }
}

# ------------------------------------------
# 特性装備
# ------------------------------------------
bsm_stellaris_equip_trait = {
  if = {
    limit = {
      has_variable = bsm_stellaris_trait_token
      has_variable = bsm_stellaris_traits_equipped
      has_variable = bsm_stellaris_trait_slots
      check_variable = { var = bsm_stellaris_traits_equipped value = bsm_stellaris_trait_slots compare = less_than }
      NOT = { has_idea = var:bsm_stellaris_trait_token }
    }
    add_ideas = var:bsm_stellaris_trait_token
    add_to_variable = { bsm_stellaris_traits_equipped = 1 }
  }
}

# ------------------------------------------
# 特性解除
# ------------------------------------------
bsm_stellaris_unequip_trait = {
  if = {
    limit = {
      has_variable = bsm_stellaris_trait_token
      has_variable = bsm_stellaris_traits_equipped
      has_idea = var:bsm_stellaris_trait_token
    }
    remove_ideas = var:bsm_stellaris_trait_token
    subtract_from_variable = { bsm_stellaris_traits_equipped = 1 }
  }
}

# ------------------------------------------
# 探検隊作成
# ------------------------------------------
bsm_stellaris_create_expedition = {
  if = {
    limit = {
      has_variable = bsm_stellaris_expedition_type
      has_variable = National_Unity_Power
      check_variable = { var = National_Unity_Power value = 50 compare = greater_than_or_equals }
    }
    subtract_from_variable = { National_Unity_Power = 50 }
    add_to_variable = { bsm_stellaris_expedition_count = 1 }
    add_to_variable = { bsm_stellaris_expedition_active = 1 }
  }
}

# ------------------------------------------
# 探検隊完了
# ------------------------------------------
bsm_stellaris_complete_expedition = {
  if = {
    limit = { check_variable = { var = bsm_stellaris_expedition_active value = 0 compare = greater_than } }
    subtract_from_variable = { bsm_stellaris_expedition_active = 1 }

    if = {
      limit = { check_variable = { var = bsm_stellaris_expedition_type value = 1 compare = equals } }
      random_list = {
        15 = { bsm_stellaris_expedition_great_success_academic = yes }
        50 = { bsm_stellaris_expedition_success_academic = yes }
        20 = { bsm_stellaris_expedition_failure = yes }
        15 = { bsm_stellaris_expedition_disaster_academic = yes }
      }
    }
    else_if = {
      limit = { check_variable = { var = bsm_stellaris_expedition_type value = 2 compare = equals } }
      random_list = {
        10 = { bsm_stellaris_expedition_great_success_military = yes }
        45 = { bsm_stellaris_expedition_success_military = yes }
        35 = { bsm_stellaris_expedition_failure = yes }
        10 = { bsm_stellaris_expedition_disaster_military = yes }
      }
    }
    else_if = {
      limit = { check_variable = { var = bsm_stellaris_expedition_type value = 3 compare = equals } }
      random_list = {
        10 = { bsm_stellaris_expedition_great_success_commercial = yes }
        50 = { bsm_stellaris_expedition_success_commercial = yes }
        30 = { bsm_stellaris_expedition_failure = yes }
        10 = { bsm_stellaris_expedition_disaster_commercial = yes }
      }
    }
  }
}

bsm_stellaris_calculate_anomaly_bonus = {
  set_variable = { bsm_stellaris_anomaly_research_bonus = 0 }

  if = {
    limit = {
      has_variable = bsm_stellaris_tradition_exploration
      check_variable = { var = bsm_stellaris_tradition_exploration value = 0 compare = greater_than }
    }
    add_to_variable = { bsm_stellaris_anomaly_research_bonus = 5 }
  }

  if = {
    limit = {
      has_variable = bsm_stellaris_tradition_prosperity
      check_variable = { var = bsm_stellaris_tradition_prosperity value = 0 compare = greater_than }
    }
    add_to_variable = { bsm_stellaris_anomaly_research_bonus = 3 }
  }

  if = {
    limit = { has_idea = bsm_stellaris_trait_technologically_advanced }
    add_to_variable = { bsm_stellaris_anomaly_research_bonus = 10 }
  }
}
