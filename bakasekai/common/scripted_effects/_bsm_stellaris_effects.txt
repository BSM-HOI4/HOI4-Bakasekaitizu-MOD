# ------------------------------------------
# 初期化
# ------------------------------------------
bsm_stellaris_init = {
  if = {
    limit = { NOT = { has_global_flag = bsm_stellaris_initialized } }
    set_global_flag = bsm_stellaris_initialized

    every_country = {
      bsm_stellaris_country_init = yes
    }
  }
}

bsm_stellaris_country_init = {
  set_variable = { bsm_stellaris_tradition_economy = 0 }
  set_variable = { bsm_stellaris_tradition_military = 0 }
  set_variable = { bsm_stellaris_tradition_society = 0 }
  set_variable = { bsm_stellaris_tradition_civilization = 0 }
  set_variable = { bsm_stellaris_tradition_adopted_count = 0 }
  set_variable = { bsm_stellaris_tradition_cost = 100 }

  set_variable = { bsm_stellaris_ascension_adopted = 0 }
  set_variable = { bsm_stellaris_ascension_slots = 0 }

  set_variable = { bsm_stellaris_essence_energy = 0 }
  set_variable = { bsm_stellaris_essence_minerals = 0 }
  set_variable = { bsm_stellaris_essence_info = 0 }

  set_variable = { bsm_stellaris_trait_slots = 0 }
  set_variable = { bsm_stellaris_traits_equipped = 0 }

  set_variable = { bsm_stellaris_anomaly_research_bonus = 0 }
  set_variable = { bsm_stellaris_anomaly_research_field = 0 }
  set_variable = { bsm_stellaris_anomaly_difficulty_level = 0 }
  set_variable = { bsm_stellaris_anomaly_stock_count = 0 }

  set_variable = { bsm_stellaris_expedition_count = 0 }
  set_variable = { bsm_stellaris_expedition_active = 0 }

  bsm_stellaris_update_trait_slots = yes
}

# ------------------------------------------
# 特性スロット更新
# ------------------------------------------
bsm_stellaris_update_trait_slots = {
  set_variable = { bsm_stellaris_trait_slots = 0 }

  if = {
    limit = { check_variable = { var = Cultural_Degree value = 30 compare = greater_than_or_equals } }
    set_variable = { bsm_stellaris_trait_slots = 1 }
  }
  if = {
    limit = { check_variable = { var = Cultural_Degree value = 50 compare = greater_than_or_equals } }
    set_variable = { bsm_stellaris_trait_slots = 3 }
  }
  if = {
    limit = { check_variable = { var = Cultural_Degree value = 80 compare = greater_than_or_equals } }
    set_variable = { bsm_stellaris_trait_slots = 5 }
  }
  if = {
    limit = { check_variable = { var = Cultural_Degree value = 100 compare = greater_than_or_equals } }
    set_variable = { bsm_stellaris_trait_slots = 6 }
  }
}

# ------------------------------------------
# 伝統取得（統合力消費）
# ------------------------------------------
bsm_stellaris_unlock_tradition = {
  bsm_stellaris_update_tradition_cost = yes

  if = {
    limit = {
      has_variable = bsm_stellaris_tradition_category
      has_variable = bsm_stellaris_tradition_cost
      has_variable = National_Unity_Power
      check_variable = { var = National_Unity_Power value = bsm_stellaris_tradition_cost compare = greater_than_or_equals }
    }

    subtract_from_variable = { var = National_Unity_Power value = bsm_stellaris_tradition_cost }

    if = {
      limit = { check_variable = { var = bsm_stellaris_tradition_category value = 1 compare = equals } }
      if = {
        limit = { check_variable = { var = bsm_stellaris_tradition_economy value = 5 compare = less_than } }
        add_to_variable = { bsm_stellaris_tradition_economy = 1 }
        # FIX(errorlog): 動的idea名連結 _@var: は無効。値(1-5)で実在ideaを選択。
        if = { limit = { check_variable = { var = bsm_stellaris_tradition_economy value = 1 compare = equals } } add_ideas = bsm_stellaris_tradition_economy_1 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_economy value = 2 compare = equals } } add_ideas = bsm_stellaris_tradition_economy_2 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_economy value = 3 compare = equals } } add_ideas = bsm_stellaris_tradition_economy_3 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_economy value = 4 compare = equals } } add_ideas = bsm_stellaris_tradition_economy_4 }
        else = { add_ideas = bsm_stellaris_tradition_economy_5 }
      }
    }
    if = {
      limit = { check_variable = { var = bsm_stellaris_tradition_category value = 2 compare = equals } }
      if = {
        limit = { check_variable = { var = bsm_stellaris_tradition_military value = 5 compare = less_than } }
        add_to_variable = { bsm_stellaris_tradition_military = 1 }
        # FIX(errorlog): 動的idea名連結 _@var: は無効。値(1-5)で実在ideaを選択。
        if = { limit = { check_variable = { var = bsm_stellaris_tradition_military value = 1 compare = equals } } add_ideas = bsm_stellaris_tradition_military_1 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_military value = 2 compare = equals } } add_ideas = bsm_stellaris_tradition_military_2 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_military value = 3 compare = equals } } add_ideas = bsm_stellaris_tradition_military_3 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_military value = 4 compare = equals } } add_ideas = bsm_stellaris_tradition_military_4 }
        else = { add_ideas = bsm_stellaris_tradition_military_5 }
      }
    }
    if = {
      limit = { check_variable = { var = bsm_stellaris_tradition_category value = 3 compare = equals } }
      if = {
        limit = { check_variable = { var = bsm_stellaris_tradition_society value = 5 compare = less_than } }
        add_to_variable = { bsm_stellaris_tradition_society = 1 }
        # FIX(errorlog): 動的idea名連結 _@var: は無効。値(1-5)で実在ideaを選択。
        if = { limit = { check_variable = { var = bsm_stellaris_tradition_society value = 1 compare = equals } } add_ideas = bsm_stellaris_tradition_society_1 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_society value = 2 compare = equals } } add_ideas = bsm_stellaris_tradition_society_2 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_society value = 3 compare = equals } } add_ideas = bsm_stellaris_tradition_society_3 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_society value = 4 compare = equals } } add_ideas = bsm_stellaris_tradition_society_4 }
        else = { add_ideas = bsm_stellaris_tradition_society_5 }
      }
    }
    if = {
      limit = { check_variable = { var = bsm_stellaris_tradition_category value = 4 compare = equals } }
      if = {
        limit = { check_variable = { var = bsm_stellaris_tradition_civilization value = 5 compare = less_than } }
        add_to_variable = { bsm_stellaris_tradition_civilization = 1 }
        # FIX(errorlog): 動的idea名連結 _@var: は無効。値(1-5)で実在ideaを選択。
        if = { limit = { check_variable = { var = bsm_stellaris_tradition_civilization value = 1 compare = equals } } add_ideas = bsm_stellaris_tradition_civilization_1 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_civilization value = 2 compare = equals } } add_ideas = bsm_stellaris_tradition_civilization_2 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_civilization value = 3 compare = equals } } add_ideas = bsm_stellaris_tradition_civilization_3 }
        else_if = { limit = { check_variable = { var = bsm_stellaris_tradition_civilization value = 4 compare = equals } } add_ideas = bsm_stellaris_tradition_civilization_4 }
        else = { add_ideas = bsm_stellaris_tradition_civilization_5 }
      }
    }

    bsm_stellaris_update_tradition_count = yes
    bsm_stellaris_update_tradition_cost = yes
  }
}

# ------------------------------------------
# 次回伝統コスト更新
# ------------------------------------------
bsm_stellaris_update_tradition_cost = {
  if = {
    limit = { NOT = { has_variable = bsm_stellaris_tradition_adopted_count } }
    bsm_stellaris_update_tradition_count = yes
  }

  set_variable = { bsm_stellaris_tradition_cost = 100 }
  set_temp_variable = { t_tradition_cost_growth = bsm_stellaris_tradition_adopted_count }
  multiply_temp_variable = { t_tradition_cost_growth = 25 }
  add_to_variable = { bsm_stellaris_tradition_cost = t_tradition_cost_growth }
  clamp_variable = { bsm_stellaris_tradition_cost = 100 max = 600 }
}

# ------------------------------------------
# 伝統取得数更新
# ------------------------------------------
bsm_stellaris_update_tradition_count = {
  set_variable = { t_total = 0 }
  add_to_variable = { t_total = bsm_stellaris_tradition_economy }
  add_to_variable = { t_total = bsm_stellaris_tradition_military }
  add_to_variable = { t_total = bsm_stellaris_tradition_society }
  add_to_variable = { t_total = bsm_stellaris_tradition_civilization }
  set_variable = { bsm_stellaris_tradition_adopted_count = t_total }
  set_variable = { bsm_stellaris_ascension_adopted = t_total }

  set_variable = { t_slots = 0 }

  if = {
    limit = { check_variable = { var = bsm_stellaris_ascension_adopted value = 10 compare = greater_than_or_equals } }
    add_to_variable = { t_slots = 1 }
  }
  if = {
    limit = { check_variable = { var = bsm_stellaris_ascension_adopted value = 20 compare = greater_than_or_equals } }
    add_to_variable = { t_slots = 1 }
  }

  clamp_variable = { t_slots = 0 max = 8 }
  set_variable = { bsm_stellaris_ascension_slots = t_slots }
}

# ------------------------------------------
# アセンションパーク取得
# ------------------------------------------
bsm_stellaris_unlock_ascension_park = {
  if = {
    limit = {
      has_variable = bsm_stellaris_ascension_park_token
      has_variable = bsm_stellaris_ascension_adopted
      has_variable = bsm_stellaris_ascension_slots
      check_variable = { var = bsm_stellaris_ascension_adopted value = 10 compare = greater_than_or_equals }
      check_variable = { var = bsm_stellaris_ascension_slots value = 0 compare = greater_than }
    }
    add_ideas = var:bsm_stellaris_ascension_park_token
    subtract_from_variable = { bsm_stellaris_ascension_slots = 1 }
  }
}

# ------------------------------------------
# 特性装備
# ------------------------------------------
bsm_stellaris_equip_trait = {
  if = {
    limit = {
      has_variable = bsm_stellaris_trait_token
      has_variable = bsm_stellaris_traits_equipped
      has_variable = bsm_stellaris_trait_slots
      check_variable = { var = bsm_stellaris_traits_equipped value = bsm_stellaris_trait_slots compare = less_than }
      NOT = { has_idea = var:bsm_stellaris_trait_token }
    }
    add_ideas = var:bsm_stellaris_trait_token
    add_to_variable = { bsm_stellaris_traits_equipped = 1 }
  }
}

# ------------------------------------------
# 特性解除
# ------------------------------------------
bsm_stellaris_unequip_trait = {
  if = {
    limit = {
      has_variable = bsm_stellaris_trait_token
      has_variable = bsm_stellaris_traits_equipped
      has_idea = var:bsm_stellaris_trait_token
    }
    remove_ideas = var:bsm_stellaris_trait_token
    subtract_from_variable = { bsm_stellaris_traits_equipped = 1 }
  }
}

# ------------------------------------------
# 探検隊作成
# ------------------------------------------
bsm_stellaris_create_expedition = {
  if = {
    limit = {
      has_variable = bsm_stellaris_expedition_type
      has_variable = National_Unity_Power
      check_variable = { var = National_Unity_Power value = 50 compare = greater_than_or_equals }
    }
    subtract_from_variable = { National_Unity_Power = 50 }
    add_to_variable = { bsm_stellaris_expedition_count = 1 }
    add_to_variable = { bsm_stellaris_expedition_active = 1 }
  }
}

# ------------------------------------------
# 探検隊完了
# ------------------------------------------
bsm_stellaris_complete_expedition = {
  if = {
    limit = { check_variable = { var = bsm_stellaris_expedition_active value = 0 compare = greater_than } }
    subtract_from_variable = { bsm_stellaris_expedition_active = 1 }

    if = {
      limit = { check_variable = { var = bsm_stellaris_expedition_type value = 1 compare = equals } }
      random_list = {
        15 = { bsm_stellaris_expedition_great_success_academic = yes }
        50 = { bsm_stellaris_expedition_success_academic = yes }
        20 = { bsm_stellaris_expedition_failure = yes }
        15 = { bsm_stellaris_expedition_disaster_academic = yes }
      }
    }
    else_if = {
      limit = { check_variable = { var = bsm_stellaris_expedition_type value = 2 compare = equals } }
      random_list = {
        10 = { bsm_stellaris_expedition_great_success_military = yes }
        45 = { bsm_stellaris_expedition_success_military = yes }
        35 = { bsm_stellaris_expedition_failure = yes }
        10 = { bsm_stellaris_expedition_disaster_military = yes }
      }
    }
    else_if = {
      limit = { check_variable = { var = bsm_stellaris_expedition_type value = 3 compare = equals } }
      random_list = {
        10 = { bsm_stellaris_expedition_great_success_commercial = yes }
        50 = { bsm_stellaris_expedition_success_commercial = yes }
        30 = { bsm_stellaris_expedition_failure = yes }
        10 = { bsm_stellaris_expedition_disaster_commercial = yes }
      }
    }
  }
}

    add_to_variable = { bsm_stellaris_anomaly_research_bonus = 5 }

    limit = {
      has_variable = bsm_stellaris_tradition_economy

    add_to_variable = { bsm_stellaris_anomaly_research_bonus = 3 }
  }

  if = {

    add_to_variable = { bsm_stellaris_anomaly_research_bonus = 10 }
  }
