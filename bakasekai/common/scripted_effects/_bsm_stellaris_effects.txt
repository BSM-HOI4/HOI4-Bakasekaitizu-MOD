# ------------------------------------------
# 初期化
# ------------------------------------------
bsm_stellaris_init = {
  if = {
    limit = { NOT = { has_global_flag = bsm_stellaris_initialized } }
    set_global_flag = bsm_stellaris_initialized

    every_country = {
      bsm_stellaris_country_init = yes
    }
  }
}

bsm_stellaris_country_init = {
  set_variable = { bsm_stellaris_tradition_exploration = 0 }
  set_variable = { bsm_stellaris_tradition_expansion = 0 }
  set_variable = { bsm_stellaris_tradition_prosperity = 0 }
  set_variable = { bsm_stellaris_tradition_armaments = 0 }
  set_variable = { bsm_stellaris_tradition_diplomacy = 0 }

  set_variable = { bsm_stellaris_ascension_slots = 0 }

  set_variable = { bsm_stellaris_essence_energy = 0 }
  set_variable = { bsm_stellaris_essence_minerals = 0 }
  set_variable = { bsm_stellaris_essence_info = 0 }

  set_variable = { bsm_stellaris_expedition_count = 0 }

  bsm_stellaris_update_trait_slots = yes

# ------------------------------------------
# 特性スロット更新
# ------------------------------------------
bsm_stellaris_update_trait_slots = {
  set_variable = { bsm_stellaris_trait_slots = 0 }

  if = {
    limit = { check_variable = { var = Cultural_Degree value = 30 compare = greater_than_or_equals } }
    set_variable = { bsm_stellaris_trait_slots = 1 }
  }
  if = {
    limit = { check_variable = { var = Cultural_Degree value = 50 compare = greater_than_or_equals } }
    set_variable = { bsm_stellaris_trait_slots = 3 }
  }
  if = {
    limit = { check_variable = { var = Cultural_Degree value = 80 compare = greater_than_or_equals } }
    set_variable = { bsm_stellaris_trait_slots = 5 }
  }
  if = {
    limit = { check_variable = { var = Cultural_Degree value = 100 compare = greater_than_or_equals } }
    set_variable = { bsm_stellaris_trait_slots = 6 }
  }
}

# ------------------------------------------
# 思想適性と伝統コスト
# ------------------------------------------
bsm_stellaris_get_ideology_tradition_affinity = {
  set_variable = { bsm_stellaris_tradition_has_affinity = 0 }
  set_variable = { bsm_stellaris_tradition_cost = 100 }

  if = {
    limit = { bsm_stellaris_has_current_tradition_affinity = yes }
    set_variable = { bsm_stellaris_tradition_has_affinity = 1 }
    set_variable = { bsm_stellaris_tradition_cost = 75 }
  }
}

# ------------------------------------------
# 伝統取得（採用 + 5伝統 + 完成）
# ------------------------------------------
bsm_stellaris_unlock_tradition = {
  bsm_stellaris_get_ideology_tradition_affinity = yes

  if = {
    limit = {
      has_variable = bsm_stellaris_tradition_category
      has_variable = bsm_stellaris_tradition_cost
      bsm_stellaris_can_unlock_current_tradition = yes
    }

    subtract_from_variable = { National_Unity_Power = bsm_stellaris_tradition_cost }

    if = {
      limit = {
        check_variable = { var = bsm_stellaris_tradition_category value = 1 compare = equals }
        check_variable = { var = bsm_stellaris_tradition_exploration value = 7 compare = less_than }
      }
      add_to_variable = { bsm_stellaris_tradition_exploration = 1 }
      bsm_stellaris_add_exploration_tradition_stage = yes
    }
    if = {
      limit = {
        check_variable = { var = bsm_stellaris_tradition_category value = 2 compare = equals }
        check_variable = { var = bsm_stellaris_tradition_expansion value = 7 compare = less_than }
      }
      add_to_variable = { bsm_stellaris_tradition_expansion = 1 }
      bsm_stellaris_add_expansion_tradition_stage = yes
    }
    if = {
      limit = {
        check_variable = { var = bsm_stellaris_tradition_category value = 3 compare = equals }
        check_variable = { var = bsm_stellaris_tradition_prosperity value = 7 compare = less_than }
      }
      add_to_variable = { bsm_stellaris_tradition_prosperity = 1 }
      bsm_stellaris_add_prosperity_tradition_stage = yes
    }
    if = {
      limit = {
        check_variable = { var = bsm_stellaris_tradition_category value = 4 compare = equals }
        check_variable = { var = bsm_stellaris_tradition_armaments value = 7 compare = less_than }
      }
      add_to_variable = { bsm_stellaris_tradition_armaments = 1 }
      bsm_stellaris_add_armaments_tradition_stage = yes
    }
    if = {
      limit = {
        check_variable = { var = bsm_stellaris_tradition_category value = 5 compare = equals }
        check_variable = { var = bsm_stellaris_tradition_diplomacy value = 7 compare = less_than }
      }
      add_to_variable = { bsm_stellaris_tradition_diplomacy = 1 }
      bsm_stellaris_add_diplomacy_tradition_stage = yes
    }
    if = {
      limit = {
        check_variable = { var = bsm_stellaris_tradition_category value = 6 compare = equals }
        check_variable = { var = bsm_stellaris_tradition_governance value = 7 compare = less_than }
      }
      add_to_variable = { bsm_stellaris_tradition_governance = 1 }
      bsm_stellaris_add_governance_tradition_stage = yes
    }
    if = {
      limit = {
        check_variable = { var = bsm_stellaris_tradition_category value = 7 compare = equals }
        check_variable = { var = bsm_stellaris_tradition_harmony value = 7 compare = less_than }
      }
      add_to_variable = { bsm_stellaris_tradition_harmony = 1 }
      bsm_stellaris_add_harmony_tradition_stage = yes
    }

    bsm_stellaris_update_tradition_count = yes
  }
}

bsm_stellaris_update_tradition_count = {
  set_variable = { t_total = 0 }
  add_to_variable = { t_total = bsm_stellaris_tradition_exploration }
  add_to_variable = { t_total = bsm_stellaris_tradition_expansion }
  add_to_variable = { t_total = bsm_stellaris_tradition_prosperity }
  add_to_variable = { t_total = bsm_stellaris_tradition_armaments }
  add_to_variable = { t_total = bsm_stellaris_tradition_diplomacy }
  add_to_variable = { t_total = bsm_stellaris_tradition_governance }
  add_to_variable = { t_total = bsm_stellaris_tradition_harmony }
  set_variable = { bsm_stellaris_ascension_adopted = t_total }

  set_variable = { t_slots = 0 }
  if = { limit = { check_variable = { var = bsm_stellaris_tradition_exploration value = 7 compare = greater_than_or_equals } } add_to_variable = { t_slots = 1 } }
  if = { limit = { check_variable = { var = bsm_stellaris_tradition_expansion value = 7 compare = greater_than_or_equals } } add_to_variable = { t_slots = 1 } }
  if = { limit = { check_variable = { var = bsm_stellaris_tradition_prosperity value = 7 compare = greater_than_or_equals } } add_to_variable = { t_slots = 1 } }
  if = { limit = { check_variable = { var = bsm_stellaris_tradition_armaments value = 7 compare = greater_than_or_equals } } add_to_variable = { t_slots = 1 } }
  if = { limit = { check_variable = { var = bsm_stellaris_tradition_diplomacy value = 7 compare = greater_than_or_equals } } add_to_variable = { t_slots = 1 } }
  if = { limit = { check_variable = { var = bsm_stellaris_tradition_governance value = 7 compare = greater_than_or_equals } } add_to_variable = { t_slots = 1 } }
  if = { limit = { check_variable = { var = bsm_stellaris_tradition_harmony value = 7 compare = greater_than_or_equals } } add_to_variable = { t_slots = 1 } }
  if = { limit = { check_variable = { var = bsm_stellaris_ascension_adopted value = 49 compare = greater_than_or_equals } } add_to_variable = { t_slots = 1 } }

  subtract_from_variable = { t_slots = bsm_stellaris_ascension_perks_taken }
  clamp_variable = { t_slots = 0 max = 8 }
  set_variable = { bsm_stellaris_ascension_slots = t_slots }
}

  add_ideas = var:bsm_stellaris_trait_token
  add_to_variable = { bsm_stellaris_traits_equipped = 1 }

  remove_ideas = var:bsm_stellaris_trait_token

  if = { limit = { check_variable = { var = bsm_stellaris_traits_equipped value = bsm_stellaris_trait_slots compare = less_than } check_variable = { var = bsm_stellaris_trait_id value = 2 compare = equals } NOT = { has_idea = bsm_stellaris_trait_peaceful } } bsm_stellaris_apply_equip_trait = yes }
  if = { limit = { check_variable = { var = bsm_stellaris_traits_equipped value = bsm_stellaris_trait_slots compare = less_than } check_variable = { var = bsm_stellaris_trait_id value = 3 compare = equals } NOT = { has_idea = bsm_stellaris_trait_expansionist } } bsm_stellaris_apply_equip_trait = yes }
  if = { limit = { check_variable = { var = bsm_stellaris_traits_equipped value = bsm_stellaris_trait_slots compare = less_than } check_variable = { var = bsm_stellaris_trait_id value = 4 compare = equals } NOT = { has_idea = bsm_stellaris_trait_isolationist } } bsm_stellaris_apply_equip_trait = yes }
  if = { limit = { check_variable = { var = bsm_stellaris_traits_equipped value = bsm_stellaris_trait_slots compare = less_than } check_variable = { var = bsm_stellaris_trait_id value = 5 compare = equals } NOT = { has_idea = bsm_stellaris_trait_spiritualist } } bsm_stellaris_apply_equip_trait = yes }
  if = { limit = { check_variable = { var = bsm_stellaris_traits_equipped value = bsm_stellaris_trait_slots compare = less_than } check_variable = { var = bsm_stellaris_trait_id value = 6 compare = equals } NOT = { has_idea = bsm_stellaris_trait_materialist } } bsm_stellaris_apply_equip_trait = yes }

}

  if = {
    limit = {
      has_variable = bsm_stellaris_expedition_type
      has_variable = National_Unity_Power
      check_variable = { var = National_Unity_Power value = 50 compare = greater_than_or_equals }
    }
    subtract_from_variable = { National_Unity_Power = 50 }
    add_to_variable = { bsm_stellaris_expedition_count = 1 }
    add_to_variable = { bsm_stellaris_expedition_active = 1 }
  }
}

  if = {
    limit = { check_variable = { var = bsm_stellaris_expedition_active value = 0 compare = greater_than } }
    subtract_from_variable = { bsm_stellaris_expedition_active = 1 }

    random = { chance = 25 add_to_variable = { bsm_stellaris_essence_energy = 50 } add_to_variable = { bsm_stellaris_essence_minerals = 50 } add_to_variable = { bsm_stellaris_essence_info = 25 } }
    random = { chance = 15 add_to_variable = { National_Unity_Power = 100 } }
    random = { chance = 8 add_ideas = bsm_stellaris_trait_available_technologically_advanced }

    if = { limit = { OR = { has_government = intellectualism has_government = technicalism } } random = { chance = 45 add_to_array = { AN_analyzed_list = token:AS_AN_Knowledge_Intensive_Laboratory } } }
    if = { limit = { OR = { has_government = futurism has_government = ruinism } } random = { chance = 45 add_to_array = { AN_analyzed_list = token:AS_AN_Manhattan_Project_Documents } } }
    if = { limit = { has_government = mythologicalism } random = { chance = 45 add_to_array = { AN_analyzed_list = token:AS_AN_Antarctic_Containment_Breach } } }
    if = { limit = { has_government = philanthropy } random = { chance = 45 add_to_array = { AN_analyzed_list = token:AS_AN_Peaceful_Militarism_Doctrine } } }
    if = { limit = { has_government = transformationism } random = { chance = 45 add_to_array = { AN_analyzed_list = token:AS_AN_Primitive_Regression_Process } } }
    if = { limit = { has_government = longitudinalism } random = { chance = 55 add_to_array = { AN_analyzed_list = token:AS_AN_Longitudinal_Stretching_Device } } }
    if = { limit = { has_government = horizontalism } random = { chance = 55 add_to_array = { AN_analyzed_list = token:AS_AN_Horizontal_Compression_System } } }
    if = { limit = { has_government = hinnulism } random = { chance = 55 add_to_array = { AN_analyzed_list = token:AS_AN_Hinnuism_Secret_Texts } } }
    if = { limit = { has_government = kyonulism } random = { chance = 55 add_to_array = { AN_analyzed_list = token:AS_AN_Kyonuism_Manifesto } } }
    if = { limit = { has_government = stupidism } random = { chance = 70 add_to_array = { AN_analyzed_list = token:AS_AN_Stupid_Education_Reform } } random = { chance = 25 add_to_variable = { bsm_stellaris_anomaly_accident_risk = 1 } } }
    if = { limit = { NOT = { has_government = stupidism } } random = { chance = 30 add_to_array = { AN_analyzed_list = token:AS_AN_International_Oxygen_Documents } } }
