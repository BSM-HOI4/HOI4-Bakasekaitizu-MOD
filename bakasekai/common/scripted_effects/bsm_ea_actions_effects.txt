# ------------------------------------------
# クールダウン初期化 (新規加盟)
# ------------------------------------------
bsm_ea_init_member_cooldowns = {
  set_variable = { bsm_ea_cd_oda = 0 }
  set_variable = { bsm_ea_cd_loan_long = 0 }
  set_variable = { bsm_ea_cd_loan_short = 0 }
  set_variable = { bsm_ea_cd_debt_relief = 0 }
  set_variable = { bsm_ea_cd_infra = 0 }
  set_variable = { bsm_ea_cd_tech = 0 }
}

# ------------------------------------------
# クールダウン減少 (月次)
# ------------------------------------------
bsm_ea_update_cooldowns = {
  if = { limit = { check_variable = { bsm_ea_cd_oda > 0 } } subtract_from_variable = { bsm_ea_cd_oda = 1 } }
  if = { limit = { check_variable = { bsm_ea_cd_loan_long > 0 } } subtract_from_variable = { bsm_ea_cd_loan_long = 1 } }
  if = { limit = { check_variable = { bsm_ea_cd_loan_short > 0 } } subtract_from_variable = { bsm_ea_cd_loan_short = 1 } }
  if = { limit = { check_variable = { bsm_ea_cd_debt_relief > 0 } } subtract_from_variable = { bsm_ea_cd_debt_relief = 1 } }
  if = { limit = { check_variable = { bsm_ea_cd_infra > 0 } } subtract_from_variable = { bsm_ea_cd_infra = 1 } }
  if = { limit = { check_variable = { bsm_ea_cd_tech > 0 } } subtract_from_variable = { bsm_ea_cd_tech = 1 } }
}

# ------------------------------------------
# ODA
# ------------------------------------------
bsm_ea_action_oda = {
  if = {
    limit = {
      bsm_ea_action_target_valid = yes
      check_variable = { Unified_Currency > 49 }
      check_variable = { bsm_ea_cd_oda = 0 }
    }
    add_to_variable = { Unified_Currency = -50 }
    event_target:bsm_ea_selected_member = {
      add_timed_idea = { idea = bsm_ea_action_oda_buff days = 90 }
      add_political_power = 50
      add_opinion_modifier = { target = ROOT modifier = bsm_ea_oda_sent }
    }
    set_variable = { bsm_ea_cd_oda = 3 }
  }
}

# ------------------------------------------
# 負債肩代わり
# ------------------------------------------
bsm_ea_action_debt_relief = {
  if = {
    limit = {
      bsm_ea_action_target_valid = yes
      check_variable = { Unified_Currency > 299 }
      check_variable = { bsm_ea_cd_debt_relief = 0 }
    }
    add_to_variable = { Unified_Currency = -300 }
    event_target:bsm_ea_selected_member = {
      if = {
        limit = { has_country_flag = ucs_low_currency_penalty }
        clr_country_flag = ucs_low_currency_penalty
        remove_dynamic_modifier = { modifier = ucs_low_currency_penalty_modifier }
      }
      add_opinion_modifier = { target = ROOT modifier = bsm_ea_debt_relief_sent }
    }
    set_variable = { bsm_ea_cd_debt_relief = 12 }
  }
}
