# 複数のready flagがある場合の優先順: party > poets > court > plains
NTS_settle_name_by_ready_path = {
  if = {
    limit = {
      has_country_flag = NTS_ready_party
    }
    NTS_settle_name_party = yes
  }
  else_if = {
    limit = {
      has_country_flag = NTS_ready_poets
    }
    NTS_settle_name_poets = yes
  }
  else_if = {
    limit = {
      has_country_flag = NTS_ready_court
    }
    NTS_settle_name_court = yes
  }
  else_if = {
    limit = {
      has_country_flag = NTS_ready_plains
    }
    NTS_settle_name_plains = yes
  }
}
