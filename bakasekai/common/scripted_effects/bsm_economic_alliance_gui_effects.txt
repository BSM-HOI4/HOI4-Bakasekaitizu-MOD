bsm_update_ea_gui_data = {
	# 1. Initialize System if needed
	bsm_ea_init_system = yes
	
	# 2. Sort Ranking (Stub)
	bsm_ea_sort_sphere_list = yes
	
	# 3. Determine Default View (My Sphere)
	# If Player is in a sphere, select it.
	if = {
		limit = { has_variable = bsm_ea_sphere }
		set_variable = { bsm_ea_selected_token = bsm_ea_sphere }
	}
	else = {
		# Fallback: Select the first ranked sphere if available, or none?
		# For now, if no sphere, clear token (list will be empty)
		if = {
			limit = { 
				check_variable = { global.bsm_ea_sphere_list^num > 0 }
			}
			set_variable = { bsm_ea_selected_token = global.bsm_ea_sphere_list^0 }
		}
		else = {
			set_variable = { bsm_ea_selected_token = 0 }
		}
	}
	
	# 4. Calculate Data for Selected Sphere (for Title/Members view)
	if = {
		limit = { 
			# Check if token is valid/set
			NOT = { check_variable = { bsm_ea_selected_token = 0 } }
		}
		bsm_ea_select_sphere_for_gui = yes
	}
}
