# ESP 帰還の四証・対FRA rapport 系scripted effects
# ESP_FRA_rapport の正本は ESP 国変数。FRA 側からは ESP = { ESP_rapport_* = yes } で動かす。

ESP_add_proof_fleet = {
  add_to_variable = {
    var = ESP_proof_fleet
    value = 1
  }
  clamp_variable = {
    var = ESP_proof_fleet
    min = 0
    max = 3
  }
  if = {
    limit = {
      check_variable = {
        var = ESP_proof_fleet
        value = 1
        compare = greater_than_or_equals
      }
      NOT = { has_idea = ESP_proof_fleet_idea }
    }
    add_ideas = ESP_proof_fleet_idea
  }
}

    var = ESP_proof_fleet
    value = 0
  }
  set_variable = {
    var = ESP_proof_bridgehead
    value = 0
  }

  }

  }

  }
}
