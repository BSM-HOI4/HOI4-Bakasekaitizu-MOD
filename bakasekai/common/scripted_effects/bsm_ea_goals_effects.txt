bsm_ea_calculate_trust = {
	set_temp_variable = { t_trust = 0 }

	# 文明度 (0~1) × 50
	if = {
		limit = { has_variable = Cultural_Degree }
		set_temp_variable = { t_cd = Cultural_Degree }
		multiply_temp_variable = { t_cd = 50 }
		add_to_temp_variable = { t_trust = t_cd }
	}

	# 統合力 (0~1000+) × 0.05
	if = {
		limit = { has_variable = National_Unity_Power }
		set_temp_variable = { t_po = National_Unity_Power }
		multiply_temp_variable = { t_po = 0.05 }
		add_to_temp_variable = { t_trust = t_po }
	}

	# UC貯蓄 ÷ 100
	if = {
		limit = { has_variable = Unified_Currency }
		set_temp_variable = { t_uc = Unified_Currency }
		divide_temp_variable = { t_uc = 100 }
		add_to_temp_variable = { t_trust = t_uc }
	}

	clamp_temp_variable = { var = t_trust min = 0 max = 100 }
	set_variable = { bsm_ea_trust = t_trust }
}

# ------------------------------------------
# 陣営指標更新
# 入力: bsm_ea_token
# ------------------------------------------
bsm_ea_update_sphere_metrics = {
	if = {
		limit = {
			check_variable = { global.bsm_ea_member_list_@var:bsm_ea_token^num > 0 }
		}
		set_temp_variable = { t_member_count = global.bsm_ea_member_list_@var:bsm_ea_token^num }
		set_temp_variable = { t_high_trust = 0 }
		set_temp_variable = { t_very_high_trust = 0 }

		for_each_loop = {
			array = global.bsm_ea_member_list_@var:bsm_ea_token
			value = v_member
			var:v_member = {
				if = {
					limit = {
						has_variable = bsm_ea_trust
						check_variable = { bsm_ea_trust > 29 }
					}
					add_to_temp_variable = { PREV.t_high_trust = 1 }
				}
				if = {
					limit = {
						has_variable = bsm_ea_trust
						check_variable = { bsm_ea_trust > 49 }
					}
					add_to_temp_variable = { PREV.t_very_high_trust = 1 }
				}
			}
		}

		# 成熟度計算
		set_temp_variable = { t_maturity = 1 }
		if = {
			limit = {
				OR = {
					check_variable = { t_member_count > 7 }
					check_variable = { t_high_trust > 4 }
				}
			}
			set_temp_variable = { t_maturity = 5 }
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { t_member_count > 5 }
					check_variable = { t_high_trust > 2 }
				}
			}
			set_temp_variable = { t_maturity = 4 }
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { t_member_count > 3 }
					check_variable = { t_high_trust > 1 }
				}
			}
			set_temp_variable = { t_maturity = 3 }
		}
		else_if = {
			limit = {
				OR = {
					check_variable = { t_member_count > 1 }
					check_variable = { t_high_trust > 0 }
				}
			}
			set_temp_variable = { t_maturity = 2 }
		}

		# 盟主に保存
		set_temp_variable = { t_leader = global.bsm_ea_member_list_@var:bsm_ea_token^0 }
		var:t_leader = {
			if = {
				limit = { NOT = { has_variable = bsm_ea_governance_mode } }
				set_variable = { bsm_ea_governance_mode = 1 }
			}
			set_variable = { bsm_ea_maturity = PREV.t_maturity }
			set_variable = { bsm_ea_member_count = PREV.t_member_count }
			set_variable = { bsm_ea_high_trust_count = PREV.t_high_trust }
		}
	}
}

# 目標追加/削除
# 入力: bsm_ea_token, bsm_ea_goal_token

bsm_ea_add_goal_to_sphere = {
	if = {
		limit = {
			check_variable = { global.bsm_ea_member_list_@var:bsm_ea_token^num > 0 }
		}
		# 重複チェック
		if = {
			limit = {
				NOT = { is_in_array = { global.bsm_ea_goal_list_@var:bsm_ea_token = var:bsm_ea_goal_token } }
			}
			add_to_array = { global.bsm_ea_goal_list_@var:bsm_ea_token = var:bsm_ea_goal_token }
			# 全メンバーにアイデアを付与 + Tech Sharing ならグループ追加
			set_temp_variable = { t_goal_for_tech = var:bsm_ea_goal_token }
			set_temp_variable = { t_token_for_tech = var:bsm_ea_token }
			for_each_loop = {
				array = global.bsm_ea_member_list_@var:bsm_ea_token
				value = v_member
				var:v_member = {
					add_ideas = var:t_goal_for_tech
					# Tech Sharing Goal の場合はグループにも追加
					if = {
						limit = {
							check_variable = { t_goal_for_tech = token:bsm_ea_goal_06_tech_sharing }
						}
						# t_token_for_tech は temp 変数のためスコープ横断でアクセス可能
						set_temp_variable = { bsm_ea_token = t_token_for_tech }
						bsm_ea_join_tech_group = yes
					}
				}
			}
		}
	}
}

bsm_ea_remove_goal_from_sphere = {
	if = {
		limit = {
			check_variable = { global.bsm_ea_member_list_@var:bsm_ea_token^num > 0 }
		}
		remove_from_array = { global.bsm_ea_goal_list_@var:bsm_ea_token = var:bsm_ea_goal_token }
		set_temp_variable = { t_goal_for_tech = var:bsm_ea_goal_token }
		set_temp_variable = { t_token_for_tech = var:bsm_ea_token }
		for_each_loop = {
			array = global.bsm_ea_member_list_@var:bsm_ea_token
			value = v_member
			var:v_member = {
				remove_ideas = var:t_goal_for_tech
				# Tech Sharing Goal の場合はグループからも削除
				if = {
					limit = {
						check_variable = { t_goal_for_tech = token:bsm_ea_goal_06_tech_sharing }
					}
					set_temp_variable = { bsm_ea_token = t_token_for_tech }
					bsm_ea_leave_tech_group = yes
				}
			}
		}
	}
}

# ------------------------------------------
# 目標設定 (盟主/投票)
# 入力: bsm_ea_selected_token, bsm_ea_goal_token
# ------------------------------------------
bsm_ea_set_goal_or_vote = {
	if = {
		limit = {
			has_variable = bsm_ea_selected_token
			check_variable = { bsm_ea_is_selected_leader = 1 }
			check_variable = { global.bsm_ea_member_list_@var:bsm_ea_selected_token^num > 0 }
		}
		set_temp_variable = { t_leader = global.bsm_ea_member_list_@var:bsm_ea_selected_token^0 }
		# 保存して子スコープに渡す
		set_temp_variable = { t_sel_token = var:bsm_ea_selected_token }
		set_temp_variable = { t_goal_token = var:bsm_ea_goal_token }

		var:t_leader = {
			if = {
				limit = {
					OR = {
						NOT = { has_variable = bsm_ea_vote_active }
						check_variable = { bsm_ea_vote_active = 0 }
					}
				}
				if = {
					limit = { check_variable = { bsm_ea_governance_mode = 1 } }
					# 盟主独裁 → 即時追加
					set_temp_variable = { bsm_ea_token = PREV.t_sel_token }
					set_temp_variable = { bsm_ea_goal_token = PREV.t_goal_token }
					bsm_ea_add_goal_to_sphere = yes
				}
				else_if = {
					limit = { check_variable = { bsm_ea_governance_mode = 2 } }
					# 民主制 → 投票開始
					set_temp_variable = { bsm_ea_token = PREV.t_sel_token }
					set_temp_variable = { bsm_ea_goal_token = PREV.t_goal_token }
					bsm_ea_start_goal_vote = yes
				}
			}
		}
	}
}

# ------------------------------------------
# 投票開始
# 入力: bsm_ea_token, bsm_ea_goal_token
# ------------------------------------------
bsm_ea_start_goal_vote = {
	if = {
		limit = {
			check_variable = { global.bsm_ea_member_list_@var:bsm_ea_token^num > 0 }
		}
		set_temp_variable = { t_member_count = global.bsm_ea_member_list_@var:bsm_ea_token^num }

		# 投票状態初期化
		for_each_loop = {
			array = global.bsm_ea_member_list_@var:bsm_ea_token
			value = v_member
			var:v_member = {
				set_variable = { bsm_ea_vote_choice = 0 }
			}
		}

		# 盟主に投票状態を保存
		set_temp_variable = { t_saved_goal = var:bsm_ea_goal_token }
		set_temp_variable = { t_leader = global.bsm_ea_member_list_@var:bsm_ea_token^0 }
		var:t_leader = {
			set_variable = { bsm_ea_vote_active = 1 }
			set_variable = { bsm_ea_vote_goal_token = PREV.t_saved_goal }
			set_variable = { bsm_ea_vote_days = 30 }
			set_variable = { bsm_ea_vote_yes = 0 }
			set_variable = { bsm_ea_vote_no = 0 }
			set_variable = { bsm_ea_vote_abstain = PREV.t_member_count }
		}

		# AI投票（盟主以外）
		set_temp_variable = { t_leader_ref = global.bsm_ea_member_list_@var:bsm_ea_token^0 }
		var:t_leader_ref = { save_event_target_as = bsm_ea_vote_leader }
		for_each_loop = {
			array = global.bsm_ea_member_list_@var:bsm_ea_token
			value = v_member
			index = v_idx
			# index 0 は盟主なのでスキップ
			if = {
				limit = { check_variable = { v_idx > 0 } }
				var:v_member = {
					if = {
						limit = { is_ai = yes }
						if = {
							limit = {
								has_opinion = {
									target = event_target:bsm_ea_vote_leader
									value > 30
								}
							}
							set_variable = { bsm_ea_vote_choice = 1 }
							var:t_leader_ref = {
								add_to_variable = { bsm_ea_vote_yes = 1 }
								subtract_from_variable = { bsm_ea_vote_abstain = 1 }
							}
						}
						else_if = {
							limit = {
								has_opinion = {
									target = event_target:bsm_ea_vote_leader
									value < -10
								}
							}
							set_variable = { bsm_ea_vote_choice = 2 }
							var:t_leader_ref = {
								add_to_variable = { bsm_ea_vote_no = 1 }
								subtract_from_variable = { bsm_ea_vote_abstain = 1 }
							}
						}
					}
				}
			}
		}
	}
}

# ------------------------------------------
bsm_ea_vote_weekly_update = {
	if = {
		limit = {
			check_variable = { global.bsm_ea_member_list_@var:bsm_ea_token^num > 0 }
		}
		set_temp_variable = { t_leader = global.bsm_ea_member_list_@var:bsm_ea_token^0 }
		var:t_leader = {
			if = {
				limit = { check_variable = { bsm_ea_vote_active = 1 } }
				subtract_from_variable = { bsm_ea_vote_days = 7 }
				if = {
					limit = { check_variable = { bsm_ea_vote_days < 1 } }
					# 投票期間終了 → 結果処理
					bsm_ea_finalize_vote = yes
				}
			}
		}
	}
}

# ------------------------------------------
# 投票最終処理
# スコープ: 盟主国（global.bsm_ea_member_list^0 内から呼ばれる）
# ------------------------------------------
bsm_ea_finalize_vote = {
	# THIS = 盟主国
	# bsm_ea_token は親スコープから継承
	set_temp_variable = { t_total = global.bsm_ea_member_list_@var:bsm_ea_token^num }
	set_temp_variable = { t_half = t_total }
	divide_temp_variable = { t_half = 2 }

	# 投票終了
	set_variable = { bsm_ea_vote_active = 0 }

	# 過半数チェック
	if = {
		limit = {
			check_variable = { bsm_ea_vote_yes > bsm_ea_vote_no }
			check_variable = { bsm_ea_vote_yes > t_half }
		}
		# 可決: 目標を追加
		set_temp_variable = { bsm_ea_goal_token = bsm_ea_vote_goal_token }
		bsm_ea_add_goal_to_sphere = yes
	}

	# 投票状態リセット
	set_variable = { bsm_ea_vote_goal_token = 0 }
	set_variable = { bsm_ea_vote_yes = 0 }
	set_variable = { bsm_ea_vote_no = 0 }
	set_variable = { bsm_ea_vote_abstain = 0 }
}

bsm_ea_vote_reset_root = {
	if = {
		limit = {
			has_variable = bsm_ea_selected_token
			check_variable = { global.bsm_ea_member_list_@var:bsm_ea_selected_token^num > 0 }
		}
		set_temp_variable = { t_leader = global.bsm_ea_member_list_@var:bsm_ea_selected_token^0 }
		if = {
			limit = { check_variable = { bsm_ea_vote_choice = 1 } }
			var:t_leader = {
				add_to_variable = { bsm_ea_vote_abstain = 1 }
				subtract_from_variable = { bsm_ea_vote_yes = 1 }
			}
		}
		else_if = {
			limit = { check_variable = { bsm_ea_vote_choice = 2 } }
			var:t_leader = {
				add_to_variable = { bsm_ea_vote_abstain = 1 }
				subtract_from_variable = { bsm_ea_vote_no = 1 }
			}
		}
		# choice = 3 (棄権) は変更不要
		set_variable = { bsm_ea_vote_choice = 0 }
	}
}
