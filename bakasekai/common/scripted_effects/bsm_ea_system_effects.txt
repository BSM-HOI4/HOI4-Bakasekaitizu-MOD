# ==========================================
# BSM 経済陣営システム Effect
# prefix: bsm_ea_
# ==========================================
# ------------------------------------------
# 初期化
# ------------------------------------------
bsm_ea_init_system = {
	if = {
		limit = {
			NOT = {
				has_global_flag = bsm_ea_system_initialized
			}
		}
		set_global_flag = bsm_ea_system_initialized
		clear_array = global.bsm_ea_sphere_list
		bsm_ea_init_goal_system = yes
	}
}

# ------------------------------------------
# 経済陣営管理
# ------------------------------------------
# 新しい経済陣営を作成
# 入力:
#   bsm_ea_token (一時変数, 文字列/トークン)
#   THIS = 盟主国
bsm_ea_create_sphere = {
	# Check if already a leader
	if = {
		limit = { has_variable = bsm_ea_is_leader }
		log = "[BSM EA] Error: [THIS.GetName] already a leader, cannot create new sphere."
	}
	else = {
		log = "[BSM EA] Creating sphere with token [?bsm_ea_token] for [THIS.GetName]"
		# グローバルリストに追加
		add_to_array = {
			global.bsm_ea_sphere_list = var:bsm_ea_token
		}
		# 盟主の初期設定
		set_variable = {
			bsm_ea_is_leader = yes
		}
		# Store sphere token on leader for future reference
		set_variable = {
			bsm_ea_sphere = var:bsm_ea_token
		}
		# 配列に追加 (Multiple Support)
		add_to_array = {
			bsm_ea_joined_spheres = var:bsm_ea_token
		}
		# この経済陣営のメンバーリストを初期化
		clear_array = global.bsm_ea_member_list_@var:bsm_ea_token
		add_to_array = {
			global.bsm_ea_member_list_@var:bsm_ea_token = THIS
		}
		# 目標リスト初期化
		clear_array = global.bsm_ea_goal_list_@var:bsm_ea_token
		# 運営モード初期値
		set_variable = { bsm_ea_governance_mode = 1 }
		set_variable = { bsm_ea_maturity = 1 }
		# 盟主のクールダウン初期化
		bsm_ea_init_member_cooldowns = yes
		log = "[BSM EA] Sphere [?bsm_ea_token] created successfully. Global list size: [?global.bsm_ea_sphere_list^num]"
		log = "[BSM EA] Member list size: [?global.bsm_ea_member_list_@var:bsm_ea_token^num]"
	}
}



# 経済陣営に加盟
# 入力:
#   bsm_ea_token (一時変数)
#   THIS = 加盟国
bsm_ea_join_sphere = {
	log = "[BSM EA] [THIS.GetName] attempting to join sphere [?bsm_ea_token]"
	log = "[BSM EA] Current joined spheres: [?bsm_ea_joined_spheres^num]"
	log = "[BSM EA] Member list before join: [?global.bsm_ea_member_list_@var:bsm_ea_token^num]"

	# Check for duplicate join
	if = {
		limit = {
			NOT = {
				is_in_array = { bsm_ea_joined_spheres = var:bsm_ea_token }
			}
		}
		log = "[BSM EA] [THIS.GetName] not already in sphere, adding now"
		# 変数を設定 (Array)
		add_to_array = {
			bsm_ea_joined_spheres = var:bsm_ea_token
		}
		# メンバーリストに追加
		add_to_array = {
			global.bsm_ea_member_list_@var:bsm_ea_token = THIS
		}
		# 目標を付与
		for_each_loop = {
			array = global.bsm_ea_goal_list_@var:bsm_ea_token
			value = v_goal
			add_ideas = temp_var:v_goal
		}
		# 新規加盟クールダウン
		bsm_ea_init_member_cooldowns = yes
		log = "[BSM EA] [THIS.GetName] added to sphere [?bsm_ea_token]"
		log = "[BSM EA] New joined spheres count: [?bsm_ea_joined_spheres^num]"
		log = "[BSM EA] New member list size: [?global.bsm_ea_member_list_@var:bsm_ea_token^num]"
	}
	else = {
		log = "[BSM EA] [THIS.GetName] already in sphere [?bsm_ea_token], skipping"
	}
}

# 現在の経済陣営から離脱
# 入力:
#   bsm_ea_token (一時変数) - 必須
#   THIS = 離脱国
bsm_ea_leave_sphere = {
	if = {
		limit = {
			is_in_array = { bsm_ea_joined_spheres = var:bsm_ea_token }
		}
		# メンバーリストから削除
		remove_from_array = {
			global.bsm_ea_member_list_@var:bsm_ea_token = THIS
		}
		# 国変数をクリーンアップ (Remove from array)
		remove_from_array = {
			bsm_ea_joined_spheres = var:bsm_ea_token
		}

		# Leader Logic Cleanup
		# If this country was leader of THIS sphere?
		# Currently `bsm_ea_is_leader` is a boolean flag, doesn't say WHICH sphere.
		# Ideally we check if we were the leader of THIS sphere.
		# (Leader is usually index 0 of member list).
		# Since we just removed ourselves, if we were 0, index 0 is now someone else or empty.

		# For now, simplistic cleanup:
		# If joined_spheres is empty, remove leader flag?

		# Cleanup bsm_ea_sphere if we are no longer in any sphere (or if we need to update to another one)
		if = {
			limit = { check_variable = { bsm_ea_joined_spheres^num = 0 } }
			clear_variable = bsm_ea_sphere
			clear_variable = bsm_ea_is_leader
		}
		else = {
			# If we are still in other spheres, update bsm_ea_sphere to the first one in list
			# to ensure GUI has something to show
			set_variable = {
				bsm_ea_sphere = bsm_ea_joined_spheres^0
			}
		}
	}
}

# 経済陣営リストを国力順にソート
# global.bsm_ea_sphere_list をソートする

# ヘルパー: 新しいトークン生成
# ------------------------------------------
bsm_ea_generate_new_token = {
	if = {
		limit = {
			NOT = {
				check_variable = {
					global.next_alliance_id > 0
				}
			}
		}
		set_variable = {
			global.next_alliance_id = 10
		}
	}
	set_temp_variable = {
		bsm_ea_token = global.next_alliance_id
	}
	add_to_variable = {
		global.next_alliance_id = 1
	}
}

# ------------------------------------------
# アクション: 経済陣営の結成
# 入力: FROM = 招待国
# ------------------------------------------
bsm_ea_form_sphere_action = {
	# 1. Generate Token
	bsm_ea_generate_new_token = yes
	# 2. Pay Costs
	add_political_power = -100
	add_to_variable = {
		var = Unified_Currency
		value = -100
	}
	# 3. Create Sphere (Leader init)
	bsm_ea_create_sphere = yes
	# 4. Add Member (FROM)
	FROM = {
		bsm_ea_join_sphere = yes
	}
	# 5. Legacy Flags / Relations Support
	set_country_flag = is_economic_alliance_leader
	set_country_flag = is_economic_alliance_member
	FROM = {
		set_country_flag = is_economic_alliance_member
	}
	# Relations
	add_opinion_modifier = {
		target = FROM
		modifier = bsm_economic_alliance_relation_boost
	}
	FROM = {
		add_opinion_modifier = {
			target = ROOT
			modifier = bsm_economic_alliance_relation_boost
		}
	}
}

# ------------------------------------------
# アクション: 経済陣営への加盟
# 入力: FROM = 盟主国 (Leader)
# ------------------------------------------
bsm_ea_join_sphere_action = {
	# Pre-requisite: FROM must have a sphere token
	# 1. Pay Costs
	add_political_power = -10
	add_to_variable = {
		var = Unified_Currency
		value = -25
	}
	# 2. Get Token from Leader
	set_temp_variable = {
		bsm_ea_token = FROM.bsm_ea_sphere
	}
	# 3. Join Sphere
	bsm_ea_join_sphere = yes
	# 4. Legacy Flags / Relations Support
	set_country_flag = is_economic_alliance_member
	# Leader Relations
	add_opinion_modifier = {
		target = FROM
		modifier = bsm_economic_alliance_relation_boost
	}
	FROM = {
		add_opinion_modifier = {
			target = ROOT
			modifier = bsm_economic_alliance_relation_boost
		}
	}
	# Member Relations: 新規加盟国(ROOT)と既存メンバー全員の間にオピニオン修正値を付与
	# (盟主FROMとは上で既に処理済みのため除外)
	for_each_loop = {
		array = global.bsm_ea_member_list_@var:bsm_ea_token
		value = v_member
		if = {
			limit = {
				NOT = { var:v_member = { tag = FROM } }
				NOT = { var:v_member = { tag = ROOT } }
			}
			var:v_member = {
				save_event_target_as = bsm_ea_temp_existing_member
			}
			event_target:bsm_ea_temp_existing_member = {
				add_opinion_modifier = {
					target = ROOT
					modifier = bsm_economic_alliance_relation_boost
				}
			}
			ROOT = {
				add_opinion_modifier = {
					target = event_target:bsm_ea_temp_existing_member
					modifier = bsm_economic_alliance_relation_boost
				}
			}
		}
	}
}

# 計算ヘルパー: 特定の陣営の国力を計算
# 入力: bsm_ea_calc_token
bsm_ea_calc_sphere_power = {
	log = "[BSM EA] Calculating power for sphere token [?bsm_ea_calc_token]"
	set_variable = {
		t_total_power = 0
	}
	# Guard: メンバーリストが空の場合はスキップ
	if = {
		limit = {
			NOT = { check_variable = { global.bsm_ea_member_list_@var:bsm_ea_calc_token^num > 0 } }
		}
		log = "[BSM EA] Warning: sphere [?bsm_ea_calc_token] has no members, skipping calc"
		break = yes
	}
	# Get leader
	set_temp_variable = {
		t_leader = global.bsm_ea_member_list_@var:bsm_ea_calc_token^0
	}
	# Calculate member count BEFORE entering scope
	set_temp_variable = {
		t_mem_count = global.bsm_ea_member_list_@var:bsm_ea_calc_token^num
	}
	log = "[BSM EA] Member list size for token [?bsm_ea_calc_token]: [?t_mem_count]"

	for_each_loop = {
		array = global.bsm_ea_member_list_@var:bsm_ea_calc_token
		value = v_mem
		var:v_mem = {
			# Calculate power based on available variables
			set_temp_variable = {
				t_p = 0
			}

			# Check if Unified_Currency_Last_Month_gain exists AND is greater than 0
			if = {
				limit = {
					has_variable = Unified_Currency_Last_Month_gain
					check_variable = { Unified_Currency_Last_Month_gain > 0 }
				}
				set_temp_variable = {
					t_p = Unified_Currency_Last_Month_gain
				}
				divide_temp_variable = {
					t_p = 10
				}
				log = "[BSM EA] [THIS.GetName] using Unified_Currency: [?t_p]"
			}
			else = {
				# Fallback: Use factory count as power measure
				set_temp_variable = {
					t_civ = num_of_civilian_factories
				}
				set_temp_variable = {
					t_mil = num_of_military_factories
				}
				set_temp_variable = {
					t_p = t_civ
				}
				add_to_temp_variable = {
					t_p = t_mil
				}
				log = "[BSM EA] [THIS.GetName] using factory count (civ: [?t_civ], mil: [?t_mil], total: [?t_p])"
			}

			# Store individual power for GUI
			set_variable = {
				bsm_ea_member_power = t_p
			}
			# Pass value to parent scope for total calculation
			set_temp_variable = {
				PREV.t_member_p = t_p
			}
			log = "[BSM EA] Member [THIS.GetName] final power: [?t_p]"
		}
		add_to_variable = {
			t_total_power = t_member_p
		}
	}
	# Store power and member count on Leader
	# Calculate member count in current scope first
	set_temp_variable = {
		final_count = global.bsm_ea_member_list_@var:bsm_ea_calc_token^num
	}
	set_temp_variable = {
		final_power = t_total_power
	}
	log = "[BSM EA] About to store on leader - count: [?final_count], power: [?final_power]"

	var:t_leader = {
		set_variable = {
			bsm_ea_total_power = PREV.final_power
		}
		set_variable = {
			bsm_ea_mem_count = PREV.final_count
		}
		log = "[BSM EA] [THIS.GetName] - Total power: [?bsm_ea_total_power], Member count: [?bsm_ea_mem_count]"
	}
}

# すべての経済陣営を再計算するヘルパー
bsm_ea_recalc_all_spheres = {
	for_each_loop = {
		array = global.bsm_ea_sphere_list
		value = v_token
		set_temp_variable = {
			bsm_ea_calc_token = var:v_token
		}
		bsm_ea_calc_sphere_power = yes
	}
}

# 経済陣営リストを国力順にソート
# global.bsm_ea_sphere_list をソートし、global.bsm_ea_sorted_sphere_list に格納
bsm_ea_sort_sphere_list = {
	log = "[BSM EA] === Starting sphere list sort ==="
	log = "[BSM EA] Input sphere list size: [?global.bsm_ea_sphere_list^num]"
	# 1. Recalculate all powers
	bsm_ea_recalc_all_spheres = yes
	# 2. Prepare lists
	clear_array = global.bsm_ea_sorted_sphere_list
	# Copy to temp list for processing
	clear_array = global.bsm_ea_temp_sort_list
	for_each_loop = {
		array = global.bsm_ea_sphere_list
		value = v_t
		add_to_array = {
			global.bsm_ea_temp_sort_list = var:v_t
		}
	}
	# 3. Selection Sort (Descending)
	set_temp_variable = {
		t_rank_counter = 1
	}
	set_temp_variable = {
		t_safety_counter = 0
	}
	while_loop_effect = {
		limit = {
			check_variable = {
				global.bsm_ea_temp_sort_list^num > 0
			}
			check_variable = { t_safety_counter < 100 }
		}
		# Find Max
		set_temp_variable = {
			t_max_power = -1
		}
		set_temp_variable = {
			t_best_token = -1
		}
		set_temp_variable = {
			t_best_index = -1
		}
		for_each_loop = {
			array = global.bsm_ea_temp_sort_list
			value = v_check_token
			index = i_index
			# Get Power (stored on Leader)
			set_temp_variable = {
				t_ldr = global.bsm_ea_member_list_@var:v_check_token^0
			}
			set_temp_variable = {
				t_current_power = 0
			}
			var:t_ldr = {
				set_temp_variable = {
					t_p = bsm_ea_total_power
				}
				# Pass value to parent scope
				set_temp_variable = {
					PREV.t_current_power = t_p
				}
			}
			if = {
				limit = {
					check_variable = {
						t_current_power > t_max_power
					}
				}
				set_temp_variable = {
					t_max_power = t_current_power
				}
				set_temp_variable = {
					t_best_token = var:v_check_token
				}
				set_temp_variable = {
					t_best_index = i_index
				}
				# Track the leader scope
				set_temp_variable = {
					t_best_leader = var:t_ldr
				}
			}
		}
		# Set Ranking on Leader
		var:t_best_leader = {
			set_variable = {
				bsm_ea_rank = PREV.t_rank_counter
			}
		}
		add_to_variable = {
			t_rank_counter = 1
		}
		# Add best to sorted list (Add Leader Scope)
		add_to_array = {
			global.bsm_ea_sorted_sphere_list = var:t_best_leader
		}
		# Note: 'var:t_best_leader' is a scope reference. 'countr:' might be safer if it's a tag,
		# but if it's a scope variable, just var:t_best_leader should work if add_to_array supports it.
		# If t_best_leader is a Scope Object, add_to_array works.
		# Let's try direct var usage.
		# Actually, 'add_to_array = { array = ... value = var:t_best_leader }'
		# Retrying with simple var.
		# Remove best from temp list
		remove_from_array = {
			global.bsm_ea_temp_sort_list = var:t_best_token
		}
		add_to_variable = { t_safety_counter = 1 }
	}
	log = "[BSM EA] === Sort completed. Sorted list size: [?global.bsm_ea_sorted_sphere_list^num] ==="
}

# GUI表示/計算用に経済陣営を選択
# 入力: bsm_ea_selected_token
# スコープ: プレイヤー/ROOT
bsm_ea_select_sphere_for_gui = {
	# Verify Token Validity
	if = {
		limit = {
			OR = {
				check_variable = { bsm_ea_selected_token = 0 }
				# Check if array exists/has size (indirect check via token in main list?)
				# Or just check if member list 0 exists
				# NOT = { check_variable = { global.bsm_ea_member_list_@var:bsm_ea_selected_token^num > 0 } }
			}
		}
		# Invalid Token
		set_variable = { bsm_ea_total_power = 0 }
		clear_variable = bsm_ea_selected_leader
		clear_array = global.bsm_ea_gui_display_list
	}
	else = {
		# Calculate Power
		set_temp_variable = {
			bsm_ea_calc_token = var:bsm_ea_selected_token
		}
		bsm_ea_calc_sphere_power = yes

		# Retrieve calculated power
		set_temp_variable = {
			t_leader = global.bsm_ea_member_list_@var:bsm_ea_selected_token^0
		}
		set_temp_variable = {
			t_sphere_power = 0
		}
		var:t_leader = {
			set_temp_variable = {
				PREV.t_sphere_power = bsm_ea_total_power
			}
		}
		set_variable = {
			bsm_ea_total_power = t_sphere_power
		}
		# 盟主設定
		set_variable = {
			bsm_ea_selected_leader = var:t_leader
		}

		# GUI用リストの作成 (静的配列へのコピー)
		clear_array = global.bsm_ea_gui_display_list
		for_each_loop = {
			array = global.bsm_ea_member_list_@var:bsm_ea_selected_token
			value = v_mem
			add_to_array = {
				global.bsm_ea_gui_display_list = var:v_mem
			}
			var:v_mem = {
				log = "[BSM EA] Added member to display list: [THIS.GetName]"
			}
		}
		log = "[BSM EA] Display list size: [?global.bsm_ea_gui_display_list^num]"
	}
}
