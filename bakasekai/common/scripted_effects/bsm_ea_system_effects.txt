# ==========================================
# BSM 経済陣営システム Effect
# prefix: bsm_ea_
# ==========================================
# ------------------------------------------
# 初期化
# ------------------------------------------
bsm_ea_init_system = {
	if = {
		limit = {
			NOT = {
				has_global_flag = bsm_ea_system_initialized
			}
		}
		set_global_flag = bsm_ea_system_initialized
		clear_array = global.bsm_ea_sphere_list
		bsm_ea_init_goal_system = yes
	}
}

# ------------------------------------------
# 加盟ニュース発火
# 入力: bsm_ea_token (temp) = 加盟先トークン, THIS = 加盟国
# 起動時の一括生成中 (bsm_ea_news_suppressed) は発火しない
# ------------------------------------------
bsm_ea_fire_join_news = {
	if = {
		limit = {
			NOT = { has_global_flag = bsm_ea_news_suppressed }
			check_variable = { global.bsm_ea_member_list_@var:bsm_ea_token^num > 0 }
		}
		save_global_event_target_as = bsm_ea_news_joiner
		set_temp_variable = { t_news_leader = global.bsm_ea_member_list_@var:bsm_ea_token^0 }
		var:t_news_leader = { save_global_event_target_as = bsm_ea_news_leader }
		# 陣営メンバー全員へニュースを送る
		for_each_loop = {
			array = global.bsm_ea_member_list_@var:bsm_ea_token
			value = v_news_member
			var:v_news_member = {
				news_event = bsm_ea_news.1
			}
		}
	}
}

# ------------------------------------------
# 経済陣営管理
# ------------------------------------------
# 新しい経済陣営を作成
# 入力:
#   bsm_ea_token (一時変数, 文字列/トークン)
#   THIS = 盟主国
bsm_ea_create_sphere = {
	# Check if already a leader
	if = {
		limit = { has_variable = bsm_ea_is_leader }
		log = "[BSM EA] Error: [THIS.GetName] already a leader, cannot create new sphere."
	}
	else = {
		log = "[BSM EA] Creating sphere with token [?bsm_ea_token] for [THIS.GetName]"
		# グローバルリストに追加
		add_to_array = {
			global.bsm_ea_sphere_list = var:bsm_ea_token
		}
		# 盟主の初期設定
		set_variable = { bsm_ea_is_leader = 1 }
		set_variable = { bsm_ea_leader_of = var:bsm_ea_token }
		# Store sphere token on leader for future reference
		set_variable = { bsm_ea_sphere = var:bsm_ea_token }
		# 配列に追加 (Multiple Support)
		add_to_array = {
			bsm_ea_joined_spheres = var:bsm_ea_token
		}
		# この経済陣営のメンバーリストを初期化
		clear_array = global.bsm_ea_member_list_@var:bsm_ea_token
		add_to_array = {
			global.bsm_ea_member_list_@var:bsm_ea_token = THIS
		}
		# 目標リスト初期化
		clear_array = global.bsm_ea_goal_list_@var:bsm_ea_token
		# 運営モード初期値
		set_variable = { bsm_ea_governance_mode = 1 }
		set_variable = { bsm_ea_maturity = 1 }
		# 全体方針 初期値: 互恵(0)
		set_variable = { global.bsm_ea_policy_@var:bsm_ea_token = 0 }
		# 議長(盟主)役職アイデアを付与
		if = {
			limit = { NOT = { has_idea = bsm_ea_role_chair } }
			add_ideas = bsm_ea_role_chair
		}
		# 盟主のクールダウン初期化
		bsm_ea_init_member_cooldowns = yes
		log = "[BSM EA] Sphere [?bsm_ea_token] created successfully. Global list size: [?global.bsm_ea_sphere_list^num]"
	}
}

# 経済陣営に加盟
# 入力:
#   bsm_ea_token (一時変数)
#   THIS = 加盟国
bsm_ea_join_sphere = {
	log = "[BSM EA] [THIS.GetName] attempting to join sphere [?bsm_ea_token]"

	# Check for duplicate join
	if = {
		limit = {
			NOT = {
				is_in_array = { bsm_ea_joined_spheres = var:bsm_ea_token }
			}
		}
		# 変数を設定 (Array)
		add_to_array = {
			bsm_ea_joined_spheres = var:bsm_ea_token
		}
		# bsm_ea_sphere を最初の所属陣営に設定（未設定の場合）
		if = {
			limit = { NOT = { has_variable = bsm_ea_sphere } }
			set_variable = { bsm_ea_sphere = var:bsm_ea_token }
		}
		# メンバーリストに追加
		add_to_array = {
			global.bsm_ea_member_list_@var:bsm_ea_token = THIS
		}
		# 目標を付与（配列内のトークンから）
		set_temp_variable = { t_join_token = var:bsm_ea_token }
		for_each_loop = {
			array = global.bsm_ea_goal_list_@var:bsm_ea_token
			value = v_goal
			add_ideas = var:v_goal
			# Tech Sharing グループが有効なら追加
			if = {
				limit = {
					check_variable = { v_goal = token:bsm_ea_goal_06_tech_sharing }
				}
				set_temp_variable = { bsm_ea_token = t_join_token }
				bsm_ea_join_tech_group = yes
			}
			if = {
				limit = {
					check_variable = { v_goal = token:bsm_ea_goal_03_currency }
					check_variable = { global.bsm_ea_member_list_@var:t_join_token^num > 0 }
				}
				set_temp_variable = { t_nc_join_leader = global.bsm_ea_member_list_@var:t_join_token^0 }
				var:t_nc_join_leader = {
					set_temp_variable = { PREV.t_nc_union_id = bsm_nc_id }
					set_temp_variable = { PREV.t_nc_union_rate = bsm_nc_rate }
					set_temp_variable = { PREV.t_nc_union_base = bsm_nc_base_rate }
				}
				bsm_nc_currency_union_start = yes
			}
		}
		# 新規加盟クールダウン
		bsm_ea_init_member_cooldowns = yes
		# 現行の全体方針アイデアを陣営全体へ再適用（新規加盟国にも付与）
		set_temp_variable = { bsm_ea_token = t_join_token }
		bsm_ea_apply_policy = yes
		# 加盟ニュースを発火（起動時の一括生成中は抑制）
		set_temp_variable = { bsm_ea_token = t_join_token }
		bsm_ea_fire_join_news = yes
		log = "[BSM EA] [THIS.GetName] added to sphere [?bsm_ea_token]"
	}
	else = {
		log = "[BSM EA] [THIS.GetName] already in sphere [?bsm_ea_token], skipping"
	}
}

# 現在の経済陣営から離脱
# 入力:
#   bsm_ea_token (一時変数) - 必須
#   THIS = 離脱国
bsm_ea_leave_sphere = {
	if = {
		limit = {
			is_in_array = { bsm_ea_joined_spheres = var:bsm_ea_token }
		}
		if = {
			limit = { has_idea = bsm_ea_goal_03_currency }
			remove_ideas = bsm_ea_goal_03_currency
			bsm_nc_currency_union_end = yes
		}
		# Tech Sharing グループが有効なら離脱
		if = {
			limit = {
				is_in_array = { global.bsm_ea_goal_list_@var:bsm_ea_token = token:bsm_ea_goal_06_tech_sharing }
			}
			bsm_ea_leave_tech_group = yes
		}
		# メンバーリストから削除
		remove_from_array = {
			global.bsm_ea_member_list_@var:bsm_ea_token = THIS
		}
		# 国変数をクリーンアップ (Remove from array)
		remove_from_array = {
			bsm_ea_joined_spheres = var:bsm_ea_token
		}

		# --- 盟主離脱時のリーダー移譲 ---
		if = {
			limit = {
				has_variable = bsm_ea_is_leader
				check_variable = { bsm_ea_leader_of = var:bsm_ea_token }
			}
			clear_variable = bsm_ea_is_leader
			clear_variable = bsm_ea_leader_of
			# メンバーが残っている場合、新盟主を設定
			if = {
				limit = {
					check_variable = { global.bsm_ea_member_list_@var:bsm_ea_token^num > 0 }
				}
				# index 0 の国を新盟主にする
				set_temp_variable = { t_new_leader = global.bsm_ea_member_list_@var:bsm_ea_token^0 }
				var:t_new_leader = {
					set_variable = { bsm_ea_is_leader = 1 }
					set_variable = { bsm_ea_leader_of = PREV.bsm_ea_token }
					# 運営モードを引き継ぎ
					if = {
						limit = { NOT = { has_variable = bsm_ea_governance_mode } }
						set_variable = { bsm_ea_governance_mode = 1 }
					}
					# 議長(盟主)役職アイデアを継承
					if = {
						limit = { NOT = { has_idea = bsm_ea_role_chair } }
						add_ideas = bsm_ea_role_chair
					}
					log = "[BSM EA] New leader: [THIS.GetName] for sphere [?PREV.bsm_ea_token]"
				}
			}
			else = {
				# メンバーゼロ → 陣営解散
				remove_from_array = {
					global.bsm_ea_sphere_list = var:bsm_ea_token
				}
				clear_array = global.bsm_ea_goal_list_@var:bsm_ea_token
				log = "[BSM EA] Sphere [?bsm_ea_token] dissolved (no members)"
			}
		}

		# bsm_ea_sphere の更新
		if = {
			limit = { check_variable = { bsm_ea_joined_spheres^num = 0 } }
			clear_variable = bsm_ea_sphere
			clear_variable = bsm_ea_is_leader
			clear_variable = bsm_ea_leader_of
		}
		else = {
			# 残っている最初の陣営を設定
			set_variable = { bsm_ea_sphere = bsm_ea_joined_spheres^0 }
		}
		log = "[BSM EA] [THIS.GetName] left sphere [?bsm_ea_token]"
	}
}

# ------------------------------------------
# アクション: 離脱
# 入力: THIS = 離脱国
# 選択済みトークン優先、フォールバックで bsm_ea_sphere
# ------------------------------------------
bsm_ea_leave_sphere_action = {
	# 離脱対象トークンの決定
	set_temp_variable = { bsm_ea_token = 0 }
	if = {
		limit = {
			has_variable = bsm_ea_selected_token
			NOT = { check_variable = { bsm_ea_selected_token = 0 } }
			is_in_array = { bsm_ea_joined_spheres = var:bsm_ea_selected_token }
		}
		set_temp_variable = { bsm_ea_token = var:bsm_ea_selected_token }
	}
	else_if = {
		limit = {
			has_variable = bsm_ea_sphere
			is_in_array = { bsm_ea_joined_spheres = var:bsm_ea_sphere }
		}
		set_temp_variable = { bsm_ea_token = bsm_ea_sphere }
	}

	if = {
		limit = { NOT = { check_variable = { bsm_ea_token = 0 } } }

		# リレーション削除
		for_each_loop = {
			array = global.bsm_ea_member_list_@var:bsm_ea_token
			value = v_member
			if = {
				limit = {
					NOT = { var:v_member = { tag = ROOT } }
				}
				var:v_member = {
					save_event_target_as = bsm_ea_temp_member
					remove_opinion_modifier = { target = ROOT modifier = bsm_economic_alliance_relation_boost }
				}
				remove_opinion_modifier = { target = event_target:bsm_ea_temp_member modifier = bsm_economic_alliance_relation_boost }
			}
		}

		# コア離脱処理
		bsm_ea_leave_sphere = yes

		# 役職・方針アイデアのクリーンアップ
		bsm_ea_cleanup_my_roles = yes

		# フラグ・クールダウン
		clr_country_flag = is_economic_alliance_member
		if = {
			limit = { has_country_flag = is_economic_alliance_leader }
			clr_country_flag = is_economic_alliance_leader
		}
		add_timed_idea = { idea = recently_left_economic_alliance days = 365 }
		add_political_power = -50
	}
}

# ------------------------------------------
# アクション: 経済陣営の結成
# 入力: FROM = 招待国
# ------------------------------------------
bsm_ea_form_sphere_action = {
	# 1. Generate Token
	bsm_ea_generate_new_token = yes
	# 2. Pay Costs
	add_political_power = -100
	add_to_variable = { Unified_Currency = -100 }
	# 3. Create Sphere (Leader init)
	bsm_ea_create_sphere = yes
	# 4. トークンを保存してFROMに渡す
	set_temp_variable = { t_saved_token = var:bsm_ea_token }
	FROM = {
		set_temp_variable = { bsm_ea_token = PREV.t_saved_token }
		bsm_ea_join_sphere = yes
	}
	# 5. Legacy Flags / Relations Support
	set_country_flag = is_economic_alliance_leader
	set_country_flag = is_economic_alliance_member
	FROM = {
		set_country_flag = is_economic_alliance_member
	}
	# Relations
	add_opinion_modifier = {
		target = FROM
		modifier = bsm_economic_alliance_relation_boost
	}
	FROM = {
		add_opinion_modifier = {
			target = ROOT
			modifier = bsm_economic_alliance_relation_boost
		}
	}
}

# ------------------------------------------
# アクション: 経済陣営への加盟
# 入力: FROM = 盟主国 (Leader)
# ------------------------------------------
bsm_ea_join_sphere_action = {
	# 1. Pay Costs
	add_political_power = -10
	add_to_variable = { Unified_Currency = -25 }
	# 2. Get Token from Leader
	set_temp_variable = { bsm_ea_token = FROM.bsm_ea_sphere }
	# 3. Join Sphere
	bsm_ea_join_sphere = yes
	# 4. Legacy Flags / Relations Support
	set_country_flag = is_economic_alliance_member
	# Leader Relations
	add_opinion_modifier = {
		target = FROM
		modifier = bsm_economic_alliance_relation_boost
	}
	FROM = {
		add_opinion_modifier = {
			target = ROOT
			modifier = bsm_economic_alliance_relation_boost
		}
	}
	# Member Relations: 新規加盟国(ROOT)と既存メンバー全員の間にオピニオン修正値を付与
	for_each_loop = {
		array = global.bsm_ea_member_list_@var:bsm_ea_token
		value = v_member
		if = {
			limit = {
				NOT = { var:v_member = { tag = FROM } }
				NOT = { var:v_member = { tag = ROOT } }
			}
			var:v_member = {
				save_event_target_as = bsm_ea_temp_member
				add_opinion_modifier = {
					target = ROOT
					modifier = bsm_economic_alliance_relation_boost
				}
			}
			add_opinion_modifier = {
				target = event_target:bsm_ea_temp_member
				modifier = bsm_economic_alliance_relation_boost
			}
		}
	}
}

# 計算ヘルパー: 特定の陣営の国力を計算
# 入力: bsm_ea_calc_token
bsm_ea_calc_sphere_power = {
	set_variable = { t_total_power = 0 }

	# Guard: メンバーリストが空の場合はスキップ
	if = {
		limit = {
			check_variable = { global.bsm_ea_member_list_@var:bsm_ea_calc_token^num > 0 }
		}
		# Get leader
		set_temp_variable = {
			t_leader = global.bsm_ea_member_list_@var:bsm_ea_calc_token^0
		}
		set_temp_variable = {
			t_mem_count = global.bsm_ea_member_list_@var:bsm_ea_calc_token^num
		}

		for_each_loop = {
			array = global.bsm_ea_member_list_@var:bsm_ea_calc_token
			value = v_mem
			var:v_mem = {
				set_temp_variable = { t_p = 0 }

				# Check if Unified_Currency_Last_Month_gain exists AND is greater than 0
				if = {
					limit = {
						has_variable = Unified_Currency_Last_Month_gain
						check_variable = { Unified_Currency_Last_Month_gain > 0 }
					}
					set_temp_variable = { t_p = Unified_Currency_Last_Month_gain }
					divide_temp_variable = { t_p = 10 }
				}
				else = {
					# Fallback: Use factory count as power measure
					set_temp_variable = { t_civ = num_of_civilian_factories }
					set_temp_variable = { t_mil = num_of_military_factories }
					set_temp_variable = { t_p = t_civ }
					add_to_temp_variable = { t_p = t_mil }
				}

				# Store individual power for GUI
				set_variable = { bsm_ea_member_power = t_p }
				# Pass value to parent scope for total calculation
				set_temp_variable = { PREV.t_member_p = t_p }
			}
			add_to_variable = { t_total_power = t_member_p }
		}

		# Store power and member count on Leader
		# Use set_variable (persistent) so PREV access works reliably
		set_variable = { bsm_ea_temp_final_power = t_total_power }

		var:t_leader = {
			set_variable = { bsm_ea_total_power = PREV.bsm_ea_temp_final_power }
			set_variable = { bsm_ea_mem_count = global.bsm_ea_member_list_@var:bsm_ea_sphere^num }
		}
	}
}

# すべての経済陣営を再計算するヘルパー
bsm_ea_recalc_all_spheres = {
	for_each_loop = {
		array = global.bsm_ea_sphere_list
		value = v_token
		set_temp_variable = { bsm_ea_calc_token = var:v_token }
		bsm_ea_calc_sphere_power = yes
	}
}

# 経済陣営リストを国力順にソート
# global.bsm_ea_sphere_list をソートし、global.bsm_ea_sorted_sphere_list に格納
# ソート結果は盟主スコープの配列（dynamic_list用）
bsm_ea_sort_sphere_list = {
	# 1. Recalculate all powers
	bsm_ea_recalc_all_spheres = yes
	# 2. Prepare lists
	clear_array = global.bsm_ea_sorted_sphere_list
	clear_temp_array = temp_sort_tokens
	clear_temp_array = temp_sort_powers

	for_each_loop = {
		array = global.bsm_ea_sphere_list
		value = v_t
		add_to_temp_array = { temp_sort_tokens = var:v_t }
		# メンバーリストが空でないかチェック
		if = {
			limit = { check_variable = { global.bsm_ea_member_list_@var:v_t^num > 0 } }
			set_temp_variable = { t_ldr = global.bsm_ea_member_list_@var:v_t^0 }
			set_temp_variable = { t_pw = 0 }
			var:t_ldr = {
				set_temp_variable = { PREV.t_pw = bsm_ea_total_power }
			}
			add_to_temp_array = { temp_sort_powers = t_pw }
		}
		else = {
			add_to_temp_array = { temp_sort_powers = 0 }
		}
	}

	# 3. Selection Sort (Descending) using find_highest_in_array
	set_temp_variable = { t_rank_counter = 1 }
	set_temp_variable = { t_sort_size = temp_sort_tokens^num }

	while_loop_effect = {
		limit = {
			check_variable = { temp_sort_tokens^num > 0 }
			check_variable = { t_rank_counter < 100 }
		}
		find_highest_in_array = {
			array = temp_sort_powers
			index = t_best_index
			value = t_best_power
		}

		# 該当トークンを取得
		set_temp_variable = { t_best_token = temp_sort_tokens^t_best_index }

		# 盟主を取得してランクを設定 & sorted_sphere_list に追加
		if = {
			limit = { check_variable = { global.bsm_ea_member_list_@var:t_best_token^num > 0 } }
			set_temp_variable = { t_best_leader = global.bsm_ea_member_list_@var:t_best_token^0 }
			var:t_best_leader = {
				set_variable = { bsm_ea_rank = PREV.t_rank_counter }
			}
			add_to_array = { global.bsm_ea_sorted_sphere_list = var:t_best_leader }
		}

		# temp配列から削除（インデックス有効性チェック）
		if = {
			limit = {
				check_variable = { t_best_index < temp_sort_tokens^num }
				check_variable = { t_best_index < temp_sort_powers^num }
			}
			remove_from_temp_array = { array = temp_sort_tokens index = t_best_index }
			remove_from_temp_array = { array = temp_sort_powers index = t_best_index }
		}
		add_to_temp_variable = { t_rank_counter = 1 }
	}
}

# GUI用: 議席グリッドの色フレームを構築する
# 入力: bsm_ea_selected_token
bsm_ea_build_seat_diagram_arrays = {
	clear_array = bsm_ea_seat_frame

	set_temp_variable = { t_target_seat_total = global.bsm_ea_member_list_@var:bsm_ea_selected_token^num }
	multiply_temp_variable = { t_target_seat_total = 10 }
	clamp_temp_variable = { var = t_target_seat_total min = 10 max = 640 }

	set_temp_variable = { t_total_seats_added = 0 }
	for_each_loop = {
		array = global.bsm_ea_member_list_@var:bsm_ea_selected_token
		value = v_mem
		index = v_idx
		set_temp_variable = { t_member_frame = v_idx }
		add_to_temp_variable = { t_member_frame = 1 }
		clamp_temp_variable = { var = t_member_frame min = 1 max = 64 }
		var:v_mem = {
			set_variable = { bsm_ea_legend_frame = PREV.t_member_frame }
			set_temp_variable = { PREV.t_member_seats = bsm_ea_member_seats }
		}
		clamp_temp_variable = { var = t_member_seats min = 0 max = 640 }
		while_loop_effect = {
			limit = {
				check_variable = { t_member_seats > 0 }
				check_variable = { t_total_seats_added < t_target_seat_total }
			}
			add_to_array = { bsm_ea_seat_frame = t_member_frame }
			add_to_temp_variable = { t_member_seats = -1 }
			add_to_temp_variable = { t_total_seats_added = 1 }
		}
	}
	while_loop_effect = {
		limit = { check_variable = { t_total_seats_added < t_target_seat_total } }
		add_to_array = { bsm_ea_seat_frame = 64 }
		add_to_temp_variable = { t_total_seats_added = 1 }
	}
}

# GUI表示/計算用に経済陣営を選択
# 入力: bsm_ea_selected_token
# スコープ: プレイヤー/ROOT
bsm_ea_select_sphere_for_gui = {
	if = {
		limit = {
			has_variable = bsm_ea_selected_token
			NOT = { check_variable = { bsm_ea_selected_token = 0 } }
			check_variable = { global.bsm_ea_member_list_@var:bsm_ea_selected_token^num > 0 }
		}
		# Calculate Power
		set_temp_variable = { bsm_ea_calc_token = var:bsm_ea_selected_token }
		bsm_ea_calc_sphere_power = yes
		# Calculate Influence / Seats (display)
		set_temp_variable = { bsm_ea_calc_token = var:bsm_ea_selected_token }
		bsm_ea_calc_sphere_influence = yes

		# Retrieve calculated power
		set_temp_variable = { t_leader = global.bsm_ea_member_list_@var:bsm_ea_selected_token^0 }
		set_temp_variable = { t_sphere_power = 0 }
		var:t_leader = {
			set_temp_variable = { PREV.t_sphere_power = bsm_ea_total_power }
		}
		set_variable = { bsm_ea_total_power = t_sphere_power }
		# 盟主設定
		set_variable = { bsm_ea_selected_leader = var:t_leader }

		# GUI用リストの作成 (静的配列へのコピー)
		clear_array = global.bsm_ea_gui_display_list
		for_each_loop = {
			array = global.bsm_ea_member_list_@var:bsm_ea_selected_token
			value = v_mem
			add_to_array = {
				global.bsm_ea_gui_display_list = var:v_mem
			}
		}
		bsm_ea_build_seat_diagram_arrays = yes
	}
	else = {
		# Invalid Token or empty
			set_variable = { bsm_ea_total_power = 0 }
			clear_variable = bsm_ea_selected_leader
			clear_array = global.bsm_ea_gui_display_list
			clear_array = bsm_ea_seat_frame
		}
	}
