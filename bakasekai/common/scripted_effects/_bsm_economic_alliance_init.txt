# ----------------------------------------------------------------------
# 初期化: 経済同盟リーダー設定
# ----------------------------------------------------------------------
bsm_initialize_economic_alliances = {
	# 次の同盟IDを初期化
	if = {
		limit = {
			NOT = {
				check_variable = {
					global.next_alliance_id >= 10
				}
			}
		}
		set_global_variable = {
			next_alliance_id = 10
		}
	}
	# EU (ID: 1) -> EUR
	if = {
		limit = {
			country_exists = EUR
		}
		EUR = {
			set_global_variable = {
				var = economic_alliance_leader
				index = 1
				value = id
			}
			set_country_flag = is_economic_alliance_leader
			set_country_flag = in_economic_alliance_EU
			add_to_array = {
				economic_alliance_ids = 1
			}
			add_to_array = {
				economic_alliance_members = THIS
			}
		}
	}
	# Allies (ID: 2) -> GBR
	if = {
		limit = {
			country_exists = GBR
		}
		GBR = {
			set_global_variable = {
				var = economic_alliance_leader
				index = 2
				value = id
			}
			set_country_flag = is_economic_alliance_leader
			set_country_flag = in_economic_alliance_Allies
			add_to_array = {
				economic_alliance_ids = 2
			}
			add_to_array = {
				economic_alliance_members = THIS
			}
		}
	}
	# GEACPS (ID: 3) -> JPN
	if = {
		limit = {
			country_exists = JPN
		}
		JPN = {
			set_global_variable = {
				var = economic_alliance_leader
				index = 3
				value = id
			}
			set_country_flag = is_economic_alliance_leader
			set_country_flag = in_economic_alliance_GEACPS
			add_to_array = {
				economic_alliance_ids = 3
			}
			add_to_array = {
				economic_alliance_members = THIS
			}
		}
	}
}
