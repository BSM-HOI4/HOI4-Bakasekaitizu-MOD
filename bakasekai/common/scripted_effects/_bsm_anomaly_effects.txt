# ------------------------------------------
# 研究開始（ID指定）
# ------------------------------------------
AN_research_start_id_1 = {
	if = {
		limit = {
			has_variable = National_Unity_Power
			check_variable = { var = National_Unity_Power value = 100 compare = greater_than_or_equals }
			NOT = { has_idea = AS_AN_International_Oxygen_Documents }
		}
		subtract_from_variable = { var = National_Unity_Power value = 100 }

		set_variable = { AN_research_progress_1 = 0 }
		set_variable = { AN_research_start_date_1 = current_date }
		set_variable = { AN_research_expected_days_1 = 150 }
		
		country_event = { id = AN_research.4 }
	}
	else = {
		country_event = { id = AN_research.3 }
	}
}

AN_research_start_id_2 = {
	if = {
		limit = {
			has_variable = National_Unity_Power
			check_variable = { var = National_Unity_Power value = 100 compare = greater_than_or_equals }
			NOT = { has_idea = AS_AN_Shiga_Conference_Records }
		}
		subtract_from_variable = { var = National_Unity_Power value = 100 }

		set_variable = { AN_research_progress_2 = 0 }
		set_variable = { AN_research_start_date_2 = current_date }
		set_variable = { AN_research_expected_days_2 = 275 }
		
		country_event = { id = AN_research.4 }
	}
	else = {
		country_event = { id = AN_research.3 }
	}
}

# ------------------------------------------
# 研究進行（週次）
# ------------------------------------------
AN_research_weekly = {
	if = {
		limit = { has_variable = AN_researching_list }
		
		if = {
			limit = { has_variable = AN_research_progress_1 }
			set_temp_variable = { t_progress = AN_research_progress_1 }
			set_temp_variable = { t_days = AN_research_expected_days_1 }
			
			set_temp_variable = { t_daily_progress = 100 }
			divide_temp_variable = { t_daily_progress = t_days }
			multiply_temp_variable = { t_daily_progress = 7 }
			
			set_temp_variable = { t_bonus = bsm_stellaris_anomaly_research_bonus }
			add_to_temp_variable = { var = t_bonus value = 1 }
			multiply_temp_variable = { t_daily_progress = t_bonus }
			
			add_to_variable = { AN_research_progress_1 = t_daily_progress }
			
			if = {
				limit = { check_variable = { var = AN_research_progress_1 value = 100 compare = greater_than_or_equals } }
					add_ideas = AS_AN_International_Oxygen_Documents
					bsm_stellaris_anomaly_reward = yes
					
					clear_variable = AN_research_progress_1
					clear_variable = AN_research_start_date_1
					clear_variable = AN_research_expected_days_1
					
					country_event = { id = AN_research.1 }
			}
		}
		
		if = {
			limit = { has_variable = AN_research_progress_2 }
			set_temp_variable = { t_progress = AN_research_progress_2 }
			set_temp_variable = { t_days = AN_research_expected_days_2 }
			
			set_temp_variable = { t_daily_progress = 100 }
			divide_temp_variable = { t_daily_progress = t_days }
			multiply_temp_variable = { t_daily_progress = 7 }
			
			set_temp_variable = { t_bonus = bsm_stellaris_anomaly_research_bonus }
			add_to_temp_variable = { var = t_bonus value = 1 }
			multiply_temp_variable = { t_daily_progress = t_bonus }
			
			add_to_variable = { AN_research_progress_2 = t_daily_progress }
			
			if = {
				limit = { check_variable = { var = AN_research_progress_2 value = 100 compare = greater_than_or_equals } }
					add_ideas = AS_AN_Shiga_Conference_Records
					bsm_stellaris_anomaly_reward = yes
					
					clear_variable = AN_research_progress_2
					clear_variable = AN_research_start_date_2
					clear_variable = AN_research_expected_days_2
					
					country_event = { id = AN_research.1 }
			}
		}
	}
}

# ------------------------------------------
# 研究中断
# ------------------------------------------
AN_research_cancel = {
	if = {
		limit = { has_variable = AN_research_progress_1 }
		clear_variable = AN_research_progress_1
		clear_variable = AN_research_start_date_1
		clear_variable = AN_research_expected_days_1
	}

	if = {
		limit = { has_variable = AN_research_progress_2 }
		clear_variable = AN_research_progress_2
		clear_variable = AN_research_start_date_2
		clear_variable = AN_research_expected_days_2
	}
}

# ------------------------------------------
# 研究完了後処理（AN_research.1 イベントオプションから呼出）
# ------------------------------------------
AN_research_complete = {
	# 全スロットが完了していれば研究リストをクリア
	if = {
		limit = {
			NOT = { has_variable = AN_research_progress_1 }
			NOT = { has_variable = AN_research_progress_2 }
		}
		clear_variable = AN_researching_list
	}
}
