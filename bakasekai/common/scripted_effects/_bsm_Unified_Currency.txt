
#Unified Currency System (統一通貨システム)

#[[ SUM{ 人口(m) × (インフラ×1.25 + 飛行場×1.1 + 港湾×0.8) } × 民需出力 × (安定度+文明) ] - (徴兵(m)×ステート数) ] / 100
# × 係数(基本1 + UC_gain_factor + 生産効率×0.5 + 消費財×0.5 + 戦時(-1.5%/月×戦争支持軽減) + 同盟(+10%/陣営,cap+30%) + 従属補正)

#初期設定
ucs_first_setting = {
	# log = "first_setting"

	if = {
		limit = { NOT = { has_country_flag = stop_Unified_Currency_gain } }
		set_variable = { var = Unified_Currency value = 0 } #リセット
		set_variable = { var = Unified_Currency_Last_Month_gain value = 0 } #リセット
		set_variable = { var = Unified_Currency_state_loss value = 0 } #リセット
		set_variable = { var = ucs_war_month_counter value = 0 } #戦時カウンターリセット
		set_temp_variable = { var = multiply_value value = 6 } #初期値は6か月分
		# log = "[?multiply_value]月分==========================="
		ucs_add_Unified_Currency_month = yes

	}
}

#指定の月数分の統一通貨を追加する  temp multiply_value  が必要
ucs_add_Unified_Currency_month = {
	if = {
		limit = { has_variable = multiply_value }

		custom_effect_tooltip = tt_ucs_add_Unified_Currency_month
		set_temp_variable = {
			var = temp_sum_stabiluty_civilization
			value = {
				value = stability
				add = Cultural_Degree
			}
		} #安定度+文明

		# 民需出力 = (民需工場 + 1) × (安定度+文明)  数式版（3op→1op）
		set_temp_variable = {
			var = temp_add_amount
			value = {
				value = num_of_civilian_factories
				add = 1
				multiply = temp_sum_stabiluty_civilization
			}
		}

		set_temp_variable = { temp_state_sum = 0 }
		for_each_scope_loop = {
			array = owned_controlled_states
			# SUM{ 人口(m) × (インフラ×1.25 + 飛行場×1.1 + 港湾×0.8) }
			set_temp_variable = {
				var = temp_state
				value = {
					value = state_population_k
					divide = 1000
					multiply = {
						value = infrastructure_level
						multiply = 1.25
						add = { value = building_level@air_base  multiply = 1.1 }
						add = { value = building_level@naval_base  multiply = 0.8 }
						max = 1.0
					}
				}
			}
			# multiply_temp_variable = { temp_state = 種族係数 } #（未実装）
			add_to_temp_variable = { PREV.temp_state_sum = temp_state }
		}
		set_temp_variable = {
			var = temp_add_amount
			value = {
				value = temp_add_amount
				multiply = temp_state_sum
				subtract = {
					value = num_owned_controlled_states^num
					multiply = deployed_army_manpower_k
					divide = 1000
				}
				divide = 100
			}
		} # [SUM × 民需出力 × (安定度+文明) - (徴兵(m)×ステート数)] / 100
		#係数
		set_temp_variable = {
			var = temp_factor
			value = {
				value = 1
				add = modifier@Unified_Currency_gain_factor
				add = { value = modifier@production_factory_max_efficiency_factor  multiply = 0.5 }
				add = { value = modifier@consumer_goods_factor  multiply = 0.5 }
			}
		}

		# 戦時下（一月ごと悪化、戦争支持率で最大50%軽減）
		if = {
			limit = { has_variable = ucs_war_month_counter }
			# penalty × (1 - war_support×0.5): war_support=1.0 で50%軽減
			set_temp_variable = {
				var = temp_war_penalty
				value = {
					value = ucs_war_month_counter
					multiply = -0.015
					multiply = {
						value = 1
						subtract = { value = war_support  multiply = 0.5 }
					}
				}
			}
			add_to_temp_variable = { temp_factor = temp_war_penalty }
		}

		# 経済同盟ボーナス（1陣営+10%、上限+30%）
		if = {
			limit = { check_variable = { economic_alliance_count > 0 } }
			set_temp_variable = {
				var = temp_alliance_bonus
				value = {
					value = economic_alliance_count
					multiply = 0.10
					clamp = { min = 0 max = 0.30 }
				}
			} # 同盟1つにつき+10%、上限+30%
			add_to_temp_variable = { temp_factor = temp_alliance_bonus }
		}

		# 宗主国・傀儡従属国 (経済同盟関連)
		# 自治度Tier判定は共有effect(_bsm_autonomy_tier.txt)に委譲し、
		# 通貨システム固有の利益分配(ucs_distribute_profit)補正のみここで適用する。
		bsm_calc_autonomy_penalty = yes

		# ----------------------------------------------------------------
		# 従属国側の処理
		# ----------------------------------------------------------------
		set_temp_variable = { temp_subject_penalty = temp_autonomy_penalty }
		if = {
			limit = { is_subject = yes }

			# 宗主国が利益分配フラグを持つ場合、ペナルティを軽減
			overlord = {
				if = {
					limit = { has_country_flag = ucs_distribute_profit }
					PREV = {
						add_to_temp_variable = { temp_subject_penalty = 0.15 }
					}
				}
			}

			add_to_temp_variable = { temp_factor = temp_subject_penalty }
		}

		# ----------------------------------------------------------------
		# 宗主国側の処理
		# ----------------------------------------------------------------
		set_temp_variable = { temp_subject_bonus_total = temp_autonomy_bonus_total }

		# 利益分配フラグを持つ宗主国は搾取分を半減
		# (各subjectの寄与を個別に0.5倍してから合算するのと数学的に等価: 0.5*(a+b+c) = 0.5a+0.5b+0.5c)
		if = {
			limit = { has_country_flag = ucs_distribute_profit }
			multiply_temp_variable = { temp_subject_bonus_total = 0.5 }
		}

		add_to_temp_variable = { temp_factor = temp_subject_bonus_total }
		clamp_temp_variable = { var = temp_factor min = 0.1 } #最低10%保証（長期戦争・多重ペナルティ対策）

		# 国家通貨 為替レート補正
		bsm_nc_calc_uc_modifier = yes
		add_to_temp_variable = { temp_factor = t_nc_uc_mod }
		clamp_temp_variable = { var = temp_factor min = 0.05 }

		set_variable = {
			var = temp_last_month
			value = {
				value = temp_add_amount
				multiply = temp_factor
			}
		} #先月の獲得量の登録
		set_temp_variable = {
			var = temp_add_amount
			value = {
				value = temp_last_month
				subtract = Unified_Currency_additional_gain
				multiply = multiply_value
			}
		} #月数の倍率
		add_to_variable = { Unified_Currency = temp_add_amount } #追加
		clamp_variable = { var = Unified_Currency max = 100000 min = -100000 }
		# log = "[?THIS.GetName]:通貨追加([?multiply_value|0]ヶ月分):[?temp_add_amount]"
		log = "・[?THIS.GetName],[?temp_add_amount],[?temp_state_sum],[?num_of_factories],[?num_owned_controlled_states^num]"

	}
	else = { log = "wow... No temp variable !!" }

}
#指定分分の統一通貨を追加する  temp add_value  が必要
ucs_add_Unified_Currency = {
	if = {
		limit = { has_variable = add_value }
		custom_effect_tooltip = tt_ucs_add_Unified_Currency
		add_to_variable = { Unified_Currency = add_value } #追加
		clamp_variable = { var = Unified_Currency max = 10000 min = -10000 }
	}
	else = { log = "wow... No temp variable !!" }
}
#月毎の計算
ucs_on_monthly = {
	if = {
		limit = { NOT = { has_country_flag = stop_Unified_Currency_gain } }
		set_temp_variable = { multiply_value = 1 }

		# 戦時カウンター更新
		if = {
			limit = { has_war = yes }
			add_to_variable = { var = ucs_war_month_counter value = 1 }
		}
		else = {
			set_variable = { var = ucs_war_month_counter value = 0 }
		}
		ucs_add_Unified_Currency_month = yes
		set_variable = { Unified_Currency_Last_Month_gain = temp_last_month } #先月の獲得量の登録
		set_variable = { var = Unified_Currency_state_loss value = 0 } #リセット

		# 最低保持EC量 (IC基準)
		# 基準: 民需 + 軍需 + 造船所 * 50 (仮)
		set_variable = {
			var = ucs_required_ec
			value = {
				value = num_of_factories
				multiply = 50
			}
		}

		# EC不足ペナルティ判定
		if = {
			limit = { check_variable = { Unified_Currency < ucs_required_ec } }
			set_country_flag = ucs_low_currency_penalty
			set_variable = {
				var = ucs_ec_deficit
				value = {
					value = ucs_required_ec
					subtract = Unified_Currency
				}
			}
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = ucs_low_currency_penalty_modifier } } }
				add_dynamic_modifier = { modifier = ucs_low_currency_penalty_modifier }
			}
		}
		else = {
			clr_country_flag = ucs_low_currency_penalty
			set_variable = { var = ucs_ec_deficit value = 0 }
			if = {
				limit = { has_dynamic_modifier = { modifier = ucs_low_currency_penalty_modifier } }
				remove_dynamic_modifier = { modifier = ucs_low_currency_penalty_modifier }
			}
		}

		# ステート拡大限界 (EC量基準)
		# 基準: 5 + (EC / 1000)
		set_variable = {
			var = ucs_state_limit
			value = {
				value = Unified_Currency
				divide = 1000
				add = 5
				round = yes
			}
		}

		# ステート超過ペナルティ判定
		if = {
			limit = { check_variable = { num_owned_states > ucs_state_limit } }
			set_country_flag = ucs_over_state_limit_penalty
			set_variable = {
				var = ucs_state_overage
				value = {
					value = num_owned_states
					subtract = ucs_state_limit
				}
			}
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = ucs_over_state_limit_penalty_modifier } } }
				add_dynamic_modifier = { modifier = ucs_over_state_limit_penalty_modifier }
			}
		}
		else = {
			clr_country_flag = ucs_over_state_limit_penalty
			set_variable = { var = ucs_state_overage value = 0 }
			if = {
				limit = { has_dynamic_modifier = { modifier = ucs_over_state_limit_penalty_modifier } }
				remove_dynamic_modifier = { modifier = ucs_over_state_limit_penalty_modifier }
			}
		}

		# 経済同盟の統合経済力計算
		# 経済同盟の統合経済力計算 (重複なし・単一所属)
		set_variable = { var = unified_economic_power_bonus value = 0 }

		# 陣営に所属している場合
		if = {
			limit = { has_variable = bsm_ea_sphere }
			# 統合経済力計算呼び出し
			ucs_accumulate_alliance_power = yes
		}

		# ボーナス適用
		if = {
			limit = { check_variable = { unified_economic_power_bonus > 0 } }
			if = {
				limit = { NOT = { has_dynamic_modifier = { modifier = ucs_economic_power_bonus_modifier } } }
				add_dynamic_modifier = { modifier = ucs_economic_power_bonus_modifier }
			}
		}
		else = {
			if = {
				limit = { has_dynamic_modifier = { modifier = ucs_economic_power_bonus_modifier } }
				remove_dynamic_modifier = { modifier = ucs_economic_power_bonus_modifier }
			}
		}


	}
}

# 経済同盟の統合経済力計算 (加算用)
# 経済同盟の統合経済力計算 (加算用)
# 使用法:
# set_temp_variable = { temp_alliance_id = 1 } # 1:EU, 2:Allies, 3:GEACPS
# ucs_accumulate_alliance_power = yes
ucs_accumulate_alliance_power = {
	set_temp_variable = { var = temp_power value = 0 }

	# 統一ロジック: 現在の陣営メンバーを取得
	if = {
		limit = { has_variable = bsm_ea_sphere }

		set_temp_variable = { t_token = bsm_ea_sphere }

		for_each_loop = {
			array = global.bsm_ea_member_list_@var:t_token
			value = v_mem
			var:v_mem = {
				add_to_variable = { PREV.temp_power = num_of_factories }
			}
		}
	}

	# ボーナス計算 (例: 50工場につき1% -> 0.01)
	# temp_power / 5000
	divide_temp_variable = { var = temp_power value = 5000 }
	add_to_variable = { unified_economic_power_bonus = temp_power }
}

# 汎用経済同盟の統合経済力計算 (リーダー用)
ucs_accumulate_generic_alliance_power = {
	set_temp_variable = { temp_power = num_of_factories } # 自分の工場

	# メンバーの工場を加算
	for_each_scope_loop = {
		array = economic_alliance_members
		add_to_temp_variable = { PREV.temp_power = num_of_factories }
	}

	# グローバル変数に保存 (メンバーが参照するため)
	# global.generic_alliance_power^MY_ID = temp_power
	set_temp_variable = { my_id = id }
	set_variable = {
		var = global.generic_alliance_power
		index = my_id
		value = temp_power
	}

	# ボーナス計算 (例: 5000工場につき1 -> 0.01)
	divide_temp_variable = { var = temp_power value = 5000 }
	add_to_variable = { unified_economic_power_bonus = temp_power }
}
