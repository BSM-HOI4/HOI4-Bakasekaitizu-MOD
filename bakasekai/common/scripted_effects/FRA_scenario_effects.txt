# FRA 二つの王冠・対ESP rapport 系scripted effects
# ESP_FRA_rapport の正本は ESP 国変数。FRA からは ESP = { ESP_rapport_* = yes } で動かす。
# PAL_crown_consent の正本は PAL 国変数。FRA からは PAL = { FRA_pal_consent_* = yes } で動かす。

FRA_init_crown_seat = {
  set_variable = {
    var = FRA_crown_seat
    value = 0
  }
}

# crown_seat: 負=パリ宮廷優位 / 正=リスボン宮廷優位
FRA_crown_toward_paris = {
  add_to_variable = {
    var = FRA_crown_seat
    value = -10
  }
  clamp_variable = {
    var = FRA_crown_seat
    min = -100
    max = 100
  }
}

# PAL 側受容度（PAL scope で呼ぶ）
FRA_pal_consent_add = {
  add_to_variable = {
    var = PAL_crown_consent
    value = 10
  }
  clamp_variable = {
    var = PAL_crown_consent
    min = 0
    max = 100
  }
}

# イベリア解放タグをイベリア総督府として従属化（FRA scope）
FRA_release_cat_governate = {
  if = {
    limit = {
      CAT = { exists = no }
    }
    release = CAT
  }
  if = {
    limit = {
      CAT = { exists = yes }
    }
    set_autonomy = {
      target = CAT
      autonomy_state = autonomy_ibrgovernate
    }
  }
}
