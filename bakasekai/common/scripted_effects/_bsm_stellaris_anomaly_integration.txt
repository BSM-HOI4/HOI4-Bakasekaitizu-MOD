# ------------------------------------------
# アノマリー研究速度計算（更新版）
# ------------------------------------------
bsm_stellaris_anomaly_research_speed_calc = {
	set_temp_variable = { t_cd = Cultural_Degree }
	divide_temp_variable = { t_cd = 100 }

	set_variable = { bsm_stellaris_anomaly_research_bonus = t_cd }

	set_temp_variable = { t_exploration = bsm_stellaris_tradition_exploration }
	multiply_temp_variable = { t_exploration = 0.06 }
	add_to_variable = { bsm_stellaris_anomaly_research_bonus = t_exploration }

	set_temp_variable = { t_harmony = bsm_stellaris_tradition_harmony }
	multiply_temp_variable = { t_harmony = 0.04 }
	add_to_variable = { bsm_stellaris_anomaly_research_bonus = t_harmony }

	set_temp_variable = { t_governance = bsm_stellaris_tradition_governance }
	multiply_temp_variable = { t_governance = 0.03 }
	add_to_variable = { bsm_stellaris_anomaly_research_bonus = t_governance }

	set_temp_variable = { t_prosperity = bsm_stellaris_tradition_prosperity }
	multiply_temp_variable = { t_prosperity = 0.03 }
	add_to_variable = { bsm_stellaris_anomaly_research_bonus = t_prosperity }

	clamp_variable = { bsm_stellaris_anomaly_research_bonus = 0.01 max = 0.5 }
}
