bsm_special_research_init = {
  if = {
    limit = { NOT = { has_variable = bsm_special_research_initialized } }
    set_variable = { bsm_special_research_initialized = 1 }
  }
  if = {
    limit = { has_idea = bsm_special_research_bureau }
    bsm_sr_refresh_all_cards = yes
  }
}
