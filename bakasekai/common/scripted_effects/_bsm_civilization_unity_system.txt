
# ==========================================
# BSM 文明度・統合力システム (Cultural Degree / National Unity Power)
# prefix: bsm_cd_ / bsm_nup_
# 週次補正(宗教・宗主国・戦時下)を反映した Cultural_Degree / National_Unity_Power の増加処理。
# 宗主国ファクターは _bsm_autonomy_tier.txt の bsm_calc_autonomy_penalty を共用する。
# ==========================================

# 週次: 文明度の上昇（宗教・宗主国補正込み）
# 補正は既存の weekly_Cultural_Degree の"倍率"として掛かる(加算ではない)。
# Cultural_Degree は[-1,1]の狭いレンジで長期間累積されるため、加算式だと
# ファクターが暴走しやすい。倍率をclampすることで安全域を保つ。
bsm_calc_cultural_degree_weekly = {
	set_temp_variable = {
		var = temp_cd_rate_factor
		value = {
			value = 1
			add = modifier@Cultural_Degree_religion_factor
		}
	}

	# 宗主国ファクター（共有effect）
	bsm_calc_autonomy_penalty = yes
	add_to_temp_variable = { temp_cd_rate_factor = temp_autonomy_penalty }

	clamp_temp_variable = { var = temp_cd_rate_factor min = 0.1 max = 2.0 }

	set_temp_variable = {
		var = temp_cd_gain
		value = {
			value = modifier@weekly_Cultural_Degree
			multiply = temp_cd_rate_factor
		}
	}

	set_variable = { Cultural_Degree_Last_Week_gain = temp_cd_gain }
	add_to_variable = { Cultural_Degree = temp_cd_gain }
	clamp_variable = { var = Cultural_Degree min = -1 max = 1 }
}

# 週次: 統合力の上昇（文明度からの基礎貢献 + 戦時/宗教/宗主国補正込みdaily gain）
# daily_National_Unity_Power_gain / National_Unity_Power_gain_factor は既存ideaが
# 付与しているが、これまでどこからも読まれていなかった("宙に浮いたフック")。
# このeffectで初めて配線する。該当ideaを持たない国はdaily gain=0のため無影響(後方互換)。
bsm_calc_national_unity_power_weekly = {
	# 既存: 文明度からの基礎貢献（維持）
	set_variable = { National_Unity_Power_Last_Week_gain = Cultural_Degree }
	add_to_variable = { National_Unity_Power = Cultural_Degree }

	# 新規: 戦時下・宗教・宗主国補正込みのdaily gain(週7日分)
	set_temp_variable = {
		var = temp_nup_factor
		value = {
			value = 1
			add = modifier@National_Unity_Power_gain_factor
			add = modifier@National_Unity_Power_religion_factor
			add = modifier@National_Unity_Power_war_factor
		}
	}

	# 宗主国ファクター（共有effect）
	bsm_calc_autonomy_penalty = yes
	add_to_temp_variable = { temp_nup_factor = temp_autonomy_penalty }
	add_to_temp_variable = { temp_nup_factor = temp_autonomy_bonus_total }

	clamp_temp_variable = { var = temp_nup_factor min = 0.1 }

	set_temp_variable = {
		var = temp_nup_gain
		value = {
			value = modifier@daily_National_Unity_Power_gain
			multiply = 7
			multiply = temp_nup_factor
		}
	}

	add_to_variable = { National_Unity_Power = temp_nup_gain }
	add_to_variable = { National_Unity_Power_Last_Week_gain = temp_nup_gain }
	clamp_variable = { var = National_Unity_Power min = -10000 max = 10000 }
}
