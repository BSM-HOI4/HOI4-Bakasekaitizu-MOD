# NSI 潮位／古層深度 のscripted triggers

NSI_tide_low = {
  check_variable = {
    var = NSI_tide_gauge
    value = 3
    compare = less_than_or_equals
  }
}

NSI_tide_mid = {
  check_variable = {
    var = NSI_tide_gauge
    value = 4
    compare = greater_than_or_equals
  }
}

NSI_tide_high = {
  check_variable = {
    var = NSI_tide_gauge
    value = 7
    compare = greater_than_or_equals
  }
}

NSI_depth_mid = {
  check_variable = {
    var = NSI_depth
    value = 4
    compare = greater_than_or_equals
  }
}

NSI_depth_high = {
  check_variable = {
    var = NSI_depth
    value = 7
    compare = greater_than_or_equals
  }
}
