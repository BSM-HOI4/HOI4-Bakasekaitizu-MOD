# rapport 閾値。+60友好確定 / -60敵対確定 / その間は冷たい共存
ESP_rapport_friendly = {
  ESP = {
    check_variable = {
      var = ESP_FRA_rapport
      value = 60
      compare = greater_than_or_equals
    }
  }
}

ESP_rapport_hostile = {
  ESP = {
    check_variable = {
      var = ESP_FRA_rapport
      value = -60
      compare = less_than_or_equals
    }
  }
}

ESP_rapport_cold = {
  ESP = {
    check_variable = {
      var = ESP_FRA_rapport
      value = -60
      compare = greater_than
    }
    check_variable = {
      var = ESP_FRA_rapport
      value = 60
      compare = less_than
    }
  }
}
