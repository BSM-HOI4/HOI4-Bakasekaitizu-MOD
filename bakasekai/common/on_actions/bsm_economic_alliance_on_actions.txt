on_actions = {

	on_startup = {
		effect = {
			# システムタグ _ から初期化を呼び出す
			# ※ _ 自体はどの陣営にも加盟しない
			XXX = {
				log = "========== ON_STARTUP TRIGGERED =========="
				log = "[BSM EA] About to call bsm_ea_init_initial_factions"
				bsm_ea_init_system = yes
				bsm_ea_init_initial_factions = yes
				log = "[BSM EA] Returned from bsm_ea_init_initial_factions"
			}
		}
	}

	on_weekly = {
		effect = {
			# パワーシェア計算（ランキング更新含む）
			bsm_update_economic_alliance_power_share = yes
			# 投票の週次更新
			for_each_loop = {
				array = global.bsm_ea_sphere_list
				value = v_token
				set_temp_variable = { bsm_ea_token = var:v_token }
				bsm_ea_vote_weekly_update = yes
			}
		}
	}

	on_monthly = {
		effect = {
			bsm_ea_monthly_update = yes
		}
	}
}
