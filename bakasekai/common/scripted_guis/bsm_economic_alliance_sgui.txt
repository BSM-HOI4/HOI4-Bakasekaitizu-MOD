scripted_gui = {
	# ボタン側の設定
	bsm_economic_alliance_sgui = {
		context_type = player_context
		window_name = "bsm_ea_button_container"
		parent_window_token = politics_tab
		visible = {
			always = yes
		}
		effects = {
			bsm_open_ea_window_button_click = {
				if = {
					limit = {
						check_variable = {
							bsm_ea_window_open = 1
						}
					}
					# 開いているなら閉じる
					set_variable = {
						bsm_ea_window_open = 0
					}
					log = "[BSM] EA Window Closed"
				}
				else = {
					# 閉じているなら開く
					bsm_update_ea_gui_data = yes
					set_variable = {
						bsm_ea_window_open = 1
					}
					log = "[BSM] EA Window Opened"
				}
			}
		}
	}
	# ウィンドウ本体の設定
	bsm_economic_alliance_window_sgui = {
		context_type = player_context
		window_name = "bsm_economic_alliance_window"
		visible = {
			check_variable = {
				bsm_ea_window_open = 1
			}
		}
		effects = {
			bsm_close_ea_window_button_click = {
				set_variable = {
					bsm_ea_window_open = 0
				}
				log = "[BSM] EA Window Closed via Close Button"
			}
		}
		triggers = {
			bsm_ea_window_visible = {
				check_variable = {
					bsm_ea_window_open = 1
				}
			}
		}
		dynamic_lists = {
			# List of Members in the *Selected* Sphere
			economic_alliance_ids = {
				# We expect 'bsm_ea_selected_token' to be set when opening the window or clicking a sphere
				array = global.bsm_ea_member_list_@var:bsm_ea_selected_token
				change_scope = yes # Array contains Country Scopes
				entry_container = "economic_alliance_ids_entry"
			}

			# List of All Spheres (Ranking)
			# We assume 'global.bsm_ea_sphere_list' contains tokens.
			# If tokens are NOT scopes, 'change_scope=yes' will fail.
			# However, if we assume tokens ARE Leader Scopes (or we only add Leaders to this list), it works.
			# Let's assume tokens are effectively Leader Scopes for now.
			# List of All Spheres (Ranking)
			# We assume 'global.bsm_ea_sphere_list' contains tokens.
			# If tokens are NOT scopes, 'change_scope=yes' will fail.
			# However, if we assume tokens ARE Leader Scopes (or we only add Leaders to this list), it works.
			# Let's assume tokens are effectively Leader Scopes for now.
			economic_alliance_ranking_list = {
				array = global.bsm_ea_sorted_sphere_list
				change_scope = yes
				entry_container = "economic_alliance_ranking_entry"
			}
		}
	}
}
