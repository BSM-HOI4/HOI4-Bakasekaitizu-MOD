scripted_gui = {
	# ボタン側の設定（countrypoliticsview.gui内に直接配置）
	bsm_economic_alliance_sgui = {
		context_type = player_context
		parent_window_name = "countrypoliticsview"
		window_name = "bsm_ea_button_container"
		visible = {
			always = yes
		}
		effects = {
			bsm_open_ea_window_button_click = {
				if = {
					limit = {
						check_variable = { bsm_ea_window_open = 1 }
					}
					# 開いているなら閉じる
					set_variable = { bsm_ea_window_open = 0 }
				}
				else = {
					# 閉じているなら開く
					bsm_update_ea_gui_data = yes
					set_variable = { bsm_ea_window_open = 1 }
				}
			}
		}
	}
	# ウィンドウ本体の設定（政治タブとは独立してフローティング表示）
	bsm_economic_alliance_window_sgui = {
		context_type = player_context
		window_name = "bsm_economic_alliance_window"
		visible = {
			check_variable = { bsm_ea_window_open = 1 }
		}
		effects = {
			bsm_close_ea_window_button_click = {
				set_variable = { bsm_ea_window_open = 0 }
			}
			bsm_ea_next_button_click = {
				bsm_ea_switch_view_next = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_prev_button_click = {
				bsm_ea_switch_view_prev = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_player_entry_button_click = {
				# dynamic_list の行スコープから、クリックした同盟盟主のトークンを取得する

				var:bsm_ea_player_entry_scope = {
					set_temp_variable = { PREV.t_bsm_ea_selected_token = bsm_ea_sphere }
				}
				ROOT = {
					set_variable = { bsm_ea_selected_token = PREV.t_bsm_ea_selected_token }
					bsm_ea_select_sphere_for_gui = yes
					bsm_ea_refresh_selection_data = yes
				}
			}
			bsm_ea_target_entry_button_click = {
				set_variable = { bsm_ea_has_selected_member = 1 }
				var:bsm_ea_target_entry_scope = {
					save_global_event_target_as = bsm_ea_selected_member
				}
				ROOT = {
					set_variable = { bsm_ea_has_selected_member = 1 }
				}
			}
			bsm_ea_governance_leader_button_click = {
				bsm_ea_set_governance_leader = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_governance_democratic_button_click = {
				bsm_ea_set_governance_democratic = yes
				bsm_ea_refresh_selection_data = yes
			}
			# 目標ボタン
			bsm_ea_goal_01_button_click = {
				set_temp_variable = { bsm_ea_goal_token = token:bsm_ea_goal_01_cooperation }
				bsm_ea_set_goal_or_vote = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_goal_02_button_click = {
				set_temp_variable = { bsm_ea_goal_token = token:bsm_ea_goal_02_tariff }
				bsm_ea_set_goal_or_vote = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_goal_03_button_click = {
				set_temp_variable = { bsm_ea_goal_token = token:bsm_ea_goal_03_currency }
				bsm_ea_set_goal_or_vote = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_goal_04_button_click = {

				bsm_ea_refresh_selection_data = yes
			}

				bsm_ea_set_goal_or_vote = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_goal_06_button_click = {
				set_temp_variable = { bsm_ea_goal_token = token:bsm_ea_goal_06_tech_sharing }
				bsm_ea_set_goal_or_vote = yes
				bsm_ea_refresh_selection_data = yes
			}
			# 投票ボタン
			bsm_ea_vote_yes_button_click = {
				bsm_ea_vote_yes = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_vote_no_button_click = {
				bsm_ea_vote_no = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_vote_abstain_button_click = {
				bsm_ea_vote_abstain = yes
				bsm_ea_refresh_selection_data = yes
			}
			# アクションボタン
			bsm_ea_action_oda_button_click = {

				bsm_ea_action_debt_relief = yes
				bsm_ea_refresh_selection_data = yes
			}

				bsm_ea_action_infra_invest = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_action_tech_button_click = {
				bsm_ea_action_tech_support = yes
				bsm_ea_refresh_selection_data = yes
			}

			bsm_ea_leave_button_click = {
				bsm_ea_leave_sphere_action = yes
				bsm_update_ea_gui_data = yes
			}
		}

			bsm_ea_nav_container_visible = {
				check_variable = { bsm_ea_joined_spheres^num > 1 }
			}
			# 非加盟テキスト: 未加盟時のみ表示
			no_alliance_text_visible = {
				NOT = { check_variable = { bsm_ea_joined_spheres^num > 0 } }

			}

			}

			actions_header_visible = {
				check_variable = { bsm_ea_joined_spheres^num > 0 }
			}

				check_variable = { bsm_ea_joined_spheres^num > 0 }
			}
			action_target_value_visible = {
				check_variable = { bsm_ea_joined_spheres^num > 0 }
			}

				check_variable = { bsm_ea_joined_spheres^num > 0 }
			}

			bsm_ea_action_oda_button_enabled = {

				check_variable = { bsm_ea_cd_loan_long = 0 }
			}
			bsm_ea_action_loan_short_button_enabled = {

				check_variable = { bsm_ea_cd_loan_short = 0 }
			}
			bsm_ea_action_debt_relief_button_enabled = {
				check_variable = { bsm_ea_has_selected_member = 1 }
				check_variable = { Unified_Currency > 299 }
				check_variable = { bsm_ea_cd_debt_relief = 0 }
			}
			bsm_ea_action_infra_button_enabled = {
				check_variable = { bsm_ea_has_selected_member = 1 }
				check_variable = { Unified_Currency > 149 }
				check_variable = { bsm_ea_cd_infra = 0 }
			}
			bsm_ea_action_tech_button_enabled = {
				check_variable = { bsm_ea_has_selected_member = 1 }
				check_variable = { Unified_Currency > 249 }
				check_variable = { bsm_ea_cd_tech = 0 }
			}
			# 離脱ボタン: 加盟中のみ表示
			bsm_ea_leave_button_visible = {
				check_variable = { bsm_ea_joined_spheres^num > 0 }
			}

			economic_alliance_player_list = {

				change_scope = yes
				country_scope_entry_container = "economic_alliance_ranking_entry"
			}
