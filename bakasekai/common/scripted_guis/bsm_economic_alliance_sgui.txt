scripted_gui = {
	# ボタン側の設定
	bsm_economic_alliance_sgui = {
		context_type = player_context
		window_name = "bsm_ea_button_container"
		parent_window_token = politics_tab
		visible = {
			always = yes
		}
		effects = {
			bsm_open_ea_window_button_click = {
				if = {
					limit = {
						check_variable = {
							bsm_ea_window_open = 1
						}
					}
					# 開いているなら閉じる
					set_variable = {
						bsm_ea_window_open = 0
					}
					log = "[BSM] EA Window Closed"
				}
				else = {
					# 閉じているなら開く
					bsm_update_ea_gui_data = yes
					set_variable = {
						bsm_ea_window_open = 1
					}
					log = "[BSM] EA Window Opened"
				}
			}
		}
	}
	# ウィンドウ本体の設定
	bsm_economic_alliance_window_sgui = {
		context_type = player_context
		window_name = "bsm_economic_alliance_window"
		visible = {
			check_variable = {
				bsm_ea_window_open = 1
			}
		}
		effects = {
			bsm_close_ea_window_button_click = {
				set_variable = {
					bsm_ea_window_open = 0
				}
				log = "[BSM] EA Window Closed via Close Button"
			}
			bsm_ea_next_button_click = {
				bsm_ea_switch_view_next = yes
			}
			bsm_ea_prev_button_click = {
				bsm_ea_switch_view_prev = yes
			}
			bsm_ea_player_entry_button_click = {
				set_variable = { bsm_ea_selected_token = FROM.bsm_ea_sphere }
				bsm_ea_select_sphere_for_gui = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_target_entry_button_click = {
				set_variable = { bsm_ea_has_selected_member = 1 }
				FROM = { save_event_target_as = bsm_ea_selected_member }
			}
			bsm_ea_governance_leader_button_click = {
				bsm_ea_set_governance_leader = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_governance_democratic_button_click = {
				bsm_ea_set_governance_democratic = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_goal_01_button_click = {
				set_temp_variable = { bsm_ea_goal_token = token:bsm_ea_goal_01_cooperation }
				bsm_ea_set_goal_or_vote = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_goal_02_button_click = {
				set_temp_variable = { bsm_ea_goal_token = token:bsm_ea_goal_02_tariff }
				bsm_ea_set_goal_or_vote = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_goal_03_button_click = {
				set_temp_variable = { bsm_ea_goal_token = token:bsm_ea_goal_03_currency }
				bsm_ea_set_goal_or_vote = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_goal_04_button_click = {
				set_temp_variable = { bsm_ea_goal_token = token:bsm_ea_goal_04_market }
				bsm_ea_set_goal_or_vote = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_goal_05_button_click = {
				set_temp_variable = { bsm_ea_goal_token = token:bsm_ea_goal_05_union }
				bsm_ea_set_goal_or_vote = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_vote_yes_button_click = {
				bsm_ea_vote_yes = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_vote_no_button_click = {
				bsm_ea_vote_no = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_vote_abstain_button_click = {
				bsm_ea_vote_abstain = yes
				bsm_ea_refresh_selection_data = yes
			}
			bsm_ea_action_oda_button_click = { bsm_ea_action_oda = yes }
			bsm_ea_action_loan_long_button_click = { bsm_ea_action_loan_long = yes }
			bsm_ea_action_loan_short_button_click = { bsm_ea_action_loan_short = yes }
			bsm_ea_action_debt_relief_button_click = { bsm_ea_action_debt_relief = yes }
			bsm_ea_action_infra_button_click = { bsm_ea_action_infra_invest = yes }
			bsm_ea_action_tech_button_click = { bsm_ea_action_tech_support = yes }
		}
		triggers = {
			bsm_ea_window_visible = {
				check_variable = {
					bsm_ea_window_open = 1
				}
			}
			# We can use triggers here to hide nav buttons via SGUI visible_when?
			# No, button visibility is usually in .gui file triggers.
			# But we can expose a trigger for the GUI to check if multiple spheres exist.
			bsm_ea_has_multiple_spheres = {
				check_variable = { bsm_ea_joined_spheres^num > 1 }
			}
		}
		dynamic_lists = {
			# List of Player's Alliances (alliances the player is a member of)
			economic_alliance_player_list = {
				array = global.bsm_ea_player_alliance_list
				change_scope = yes # Array contains Country Scopes (alliance leaders)
				entry_container = "economic_alliance_player_entry"
			}

			# List of Members in the *Selected* Sphere (displayed in "Member List" section)
			economic_alliance_member_list = {
				# We use a static global array populated by 'bsm_ea_select_sphere_for_gui'
				# to avoid scope issues with dynamic path interpolation in SGUI.
				array = global.bsm_ea_gui_display_list
				change_scope = yes # Array contains Country Scopes
				entry_container = "economic_alliance_member_entry"
			}

			# Target selection list for actions
			economic_alliance_target_list = {
				array = global.bsm_ea_gui_display_list
				change_scope = yes
				entry_container = "economic_alliance_target_entry"
			}

			# List of All Spheres (Ranking)
			economic_alliance_ranking_list = {
				array = global.bsm_ea_sorted_sphere_list
				change_scope = yes
				entry_container = "economic_alliance_ranking_entry"
			}
		}
	}
}
