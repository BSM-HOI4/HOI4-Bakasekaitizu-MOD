scripted_gui = {
	bsm_topbar = {
		window_name = "bsm_topbar"
		context_type = player_context
		
		effects = {
			bsm_OE_GUI_open_button_click = {
				if = {
					limit = { has_variable = OE_page }
					clear_variable = OE_page
				}
				else = {
					set_variable = { OE_page = 1 }
				}
			}
			anomaries_button_click = {
				log = "bsm_anomaly_button_click"
				if = {
					limit = { check_variable = { bsm_stellaris_tab = 6 } }
					clear_variable = bsm_stellaris_tab
				}
				else = {
					set_variable = { bsm_stellaris_tab = 6 }
				}
			}
			bsm_stellaris_open_button_click = {
				if = {
					limit = { has_variable = bsm_stellaris_tab }
					clear_variable = bsm_stellaris_tab
				}
				else = {
					set_variable = { bsm_stellaris_tab = 1 }
				}
			}
		}
	}
}
