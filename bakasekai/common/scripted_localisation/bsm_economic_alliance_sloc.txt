# Tooltip Member List Helpers
# Logic: Checks if variable bsm_ea_mem_X exists/is set, returns its name.

defined_text = {
	name = GetEAMember0
	text = {
		trigger = { check_variable = { bsm_ea_mem_0.is_setup = yes } } # simplified check: just check if var is set?
		# "var:x" returns scope. If not set?
		# check_variable = { bsm_ea_mem_count > 0 } is safer
		trigger = { check_variable = { bsm_ea_mem_count > 0 } }
		localization_key = BSM_EA_MEM_0_LOC
	}
}
