# --- 段階表示 (強/普通/弱) ---
defined_text = {
	name = GetNCTier
	text = {
		trigger = {
			set_temp_variable = { t_nc_thresh = bsm_nc_base_rate }
			multiply_temp_variable = { t_nc_thresh = 0.9 }
			check_variable = { bsm_nc_rate < t_nc_thresh }
		}
		localization_key = nc_tier_strong
	}
	text = {
		trigger = {
			set_temp_variable = { t_nc_high = bsm_nc_base_rate }
			multiply_temp_variable = { t_nc_high = 1.1 }
			check_variable = { bsm_nc_rate > t_nc_high }
		}
		localization_key = nc_tier_weak
	}
	text = { trigger = { always = yes } localization_key = nc_tier_normal }
}
