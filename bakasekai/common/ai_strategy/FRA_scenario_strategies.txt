# FRA 対ESP・PAL・イベリアAI戦略

FRA_esp_friend = {
  enable = {
    original_tag = FRA
    ESP = { exists = yes }
    OR = {
      has_completed_focus = FRA_spain_handshake
      has_country_flag = FRA_chose_friendship
      ESP = {
        check_variable = {
          var = ESP_FRA_rapport
          value = 60
          compare = greater_than_or_equals
        }
      }
    }
  }
  abort_when_not_enabled = yes

  ai_strategy = {
    type = befriend
    id = "ESP"
    value = 120
  }
  ai_strategy = {
    type = antagonize
    id = "ESP"
    value = -80
  }
}
