ESP_fra_friend = {
  enable = {
    original_tag = ESP
    FRA = { exists = yes }
    OR = {
      has_completed_focus = ESP_fra_handshake
      has_country_flag = ESP_chose_friendship
      check_variable = {

        value = 60
        compare = greater_than_equals

  }

    type = befriend

    value = 150
  }

    type = alliance
    id = "FRA"
    value = 50
  }
  ai_strategy = {
    type = antagonize
    id = "FRA"
    value = -100
  }
}
