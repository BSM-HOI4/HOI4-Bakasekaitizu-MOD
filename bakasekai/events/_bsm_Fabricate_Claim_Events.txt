# Hearts of Iron IV スクリプトファイル
# events/BSM_Fabricate_Claim_Events.txt

add_namespace = bsm_fabricate_claim_events

# 請求権の捏造イベント
# 外交アクション bsm_fabricate_claim_action の complete_effect から発火する。
# 発火元 (complete_effect) で対象国を event_target:bsm_fabricate_claim_target として保存しているため、
# このイベントでは ROOT = 自国(実行国)、event_target:bsm_fabricate_claim_target = 対象国 となる。
# （無印 country_event で発火するため FROM は自国になってしまう点に注意。対象国は必ず event_target で参照する）
country_event = {
    id = bsm_fabricate_claim_events.1
    title = "BSM_FABRICATE_CLAIM_EVENT_1_TITLE" # "[対象国]に対する請求権の捏造"
    desc = "BSM_FABRICATE_CLAIM_EVENT_1_DESC"  # どの州に捏造するかは、このdescで動的に表示
    picture = GFX_report_event_generic_contract_sign # 仮のイベント画像

    is_triggered_only = yes

    # 即時効果: 対象となるランダムな州を決定し、捏造コストを算出する
    immediate = {
        event_target:bsm_fabricate_claim_target = { # 外交アクションの対象国
            # 「自国(ROOT)と隣接している」 OR 「沿岸州である」かつ「自国が未請求・未中核・未所有」の州をランダムに1つ選ぶ
            random_owned_state = {
                limit = {
                    # この州は自国の中核州ではない
                    NOT = { is_core_of = ROOT }
                    # この州は自国が既に請求権を持っていない
                    NOT = { is_claimed_by = ROOT }
                    # そして、「自国と隣接している州」 OR 「沿岸州である」
                    OR = {
                        any_neighbor_state = { is_owned_by = ROOT }
                        is_coastal = yes
                    }
                }
                # 選ばれた州をイベントターゲットに保存
                save_event_target_as = bsm_target_state_for_claim_fabrication
                # デバッグ用ログ
                log = "LOG: [ROOT.GetNameDefCap] will attempt to fabricate claim on [THIS.GetName] (owned by [event_target:bsm_fabricate_claim_target.GetNameDefCap])"
            }
        }

        if = {
            limit = { has_event_target = bsm_target_state_for_claim_fabrication }
            # 捏造コストを算出（人口千人あたり1UC + 基本100UC）。descで表示するため immediate で計算する
            event_target:bsm_target_state_for_claim_fabrication = {
                set_temp_variable = { temp_pop_for_cost = state_population_k } # 千人単位の人口
                multiply_temp_variable = { temp_pop_for_cost = 1 }            # 千人あたり1UC
                ROOT = { set_variable = { bsm_fabrication_actual_cost = PREV.temp_pop_for_cost } }
            }
            add_to_variable = { bsm_fabrication_actual_cost = 100 } # 基本コスト加算
        }
        else = {
            # 有効な州が見つからなかった場合（外交アクションのトリガーで弾かれるはずだが念のため）
            log = "ERROR: No valid state found for claim fabrication for [ROOT.GetNameDefCap] against [event_target:bsm_fabricate_claim_target.GetNameDefCap]"
            set_variable = { bsm_fabrication_actual_cost = 0 }
        }
    }

    option = {
        name = "BSM_FABRICATE_CLAIM_EVENT_1_OPT_A" # "実行せよ！"
        trigger = {
            # 決定された対象州が存在すること
            has_event_target = bsm_target_state_for_claim_fabrication
        }
        ai_chance = { factor = 100 }

        if = {
            # UCが足りない場合は実行せず、UC不足を通知する
            limit = { check_variable = { Unified_Currency < bsm_fabrication_actual_cost } }
            custom_effect_tooltip = BSM_FABRICATE_CLAIM_NOT_ENOUGH_UC_TT
            hidden_effect = { country_event = { id = bsm_fabricate_claim_events.uc_shortage days = 1 } }
        }
        else = {
            # UC消費 (add_value にコストの負値を入れて ucs_add_Unified_Currency を呼ぶ)
            set_temp_variable = { add_value = bsm_fabrication_actual_cost }
            multiply_temp_variable = { add_value = -1 }
            ucs_add_Unified_Currency = yes

            # 対象州に自国の請求権を付与
            event_target:bsm_target_state_for_claim_fabrication = {
                add_claim_by = ROOT
            }

            # 国際緊張度の上昇
            add_named_threat = {
                threat = 0.003
                name = "LOC_BSM_FABRICATE_CLAIM_THREAT_JP"
            }

            # 対象国との関係悪化
            event_target:bsm_fabricate_claim_target = {
                add_opinion_modifier = {
                    target = ROOT
                    modifier = fabricated_claim_bsm_negative_opinion
                }
            }
        }
    }

    option = {
        name = "BSM_FABRICATE_CLAIM_EVENT_1_OPT_B" # "中止する"
        ai_chance = { factor = 0 }
    }
}

# (任意) UC不足通知イベント
country_event = {
    id = bsm_fabricate_claim_events.uc_shortage
    title = "BSM_FABRICATE_CLAIM_UC_SHORTAGE_TITLE"
    desc = "BSM_FABRICATE_CLAIM_UC_SHORTAGE_DESC"
    picture = GFX_report_event_generic_contract_cancelled # 仮
    is_triggered_only = yes
    option = { name = OKAY }
}
