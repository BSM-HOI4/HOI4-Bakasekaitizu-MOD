country_event = {
	id = bsm_ind.20
	title = bsm_ind.20.t
	desc = bsm_ind.20.d
	picture = GFX_report_event_RAJ_political

	is_triggered_only = yes

	immediate = {
		# パキスタンが存在しない場合、独立させる
		if = {
			limit = { PAK = { exists = no } }
			release = PAK
		}
		if = {
			limit = { PAK = { exists = yes } }
			PAK = {
				set_politics = {
					ruling_party = neutrality_ideology
					last_election = "1947.8.14"
					elections_allowed = yes
				}
				add_political_power = 100
				if = {
					limit = { NOT = { has_idea = RAJ_marginalized_muslim_community_happy } }
					add_ideas = RAJ_marginalized_muslim_community_happy
				}
			}
		}
	}

	option = {
		name = bsm_ind.20.a
		# 分割を受け入れ、平和的な移行を目指す
		add_stability = 0.05
		add_political_power = 50
		if = {
			limit = { PAK = { exists = yes } }
			PAK = { add_opinion_modifier = { target = IND modifier = small_increase } }
			add_opinion_modifier = { target = PAK modifier = small_increase }
		}
		news_event = { id = bsm_ind.20b days = 1 }
	}
}
