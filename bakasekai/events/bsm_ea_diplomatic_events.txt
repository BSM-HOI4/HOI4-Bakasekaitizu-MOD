# ----------------------------------------------------------------------

	id = bsm_ea_diplomatic.1
	title = bsm_ea_diplomatic.1.t
	desc = bsm_ea_diplomatic.1.d
	picture = GFX_report_event_generic_read_write

	is_triggered_only = yes

	# Accept
	option = {
		name = bsm_ea_diplomatic.1.a
		ai_chance = {
			base = 50
			modifier = {
				add = 50
				has_opinion = { target = FROM value > 50 }
			}
		}
		# Add FROM to our alliance
		FROM = {
			set_temp_variable = { bsm_ea_token = ROOT.bsm_ea_sphere }
			bsm_ea_join_sphere = yes
			country_event = { id = bsm_ea_diplomatic.11 }
		}
		bsm_update_ea_gui_data = yes
	}

	# Reject
	option = {
		name = bsm_ea_diplomatic.1.b
		ai_chance = {
			base = 50
			modifier = {
				add = -50
				has_opinion = { target = FROM value > 50 }
			}
		}
		FROM = {
			country_event = { id = bsm_ea_diplomatic.12 }
		}
	}
}
