# 1. Join Request (Sent to Leader)
country_event = {
	id = bsm_economic_alliance_actions.1
	title = bsm_economic_alliance_actions.1.t
	desc = bsm_economic_alliance_actions.1.d
	is_triggered_only = yes
	option = {
		name = bsm_economic_alliance_actions.1.a		# Accept
		ai_chance = {
			base = 30
			modifier = {
				factor = 0
				has_opinion = {
					target = FROM
					value < -20
				}
			}
			modifier = {
				add = 10
				has_opinion = {
					target = FROM
					value > 20
				}
			}
			modifier = {
				add = 100
				is_in_faction_with = FROM
			}
		}
		# Set Targets
		save_global_event_target_as = bsm_ea_leader
		FROM = {
			save_global_event_target_as = bsm_ea_joiner
		}
		# Notify Joiner
		FROM = {
			country_event = {
				id = bsm_economic_alliance_actions.11
			}
		}
		# Execute Join Effect
		event_target:bsm_ea_joiner = {
			bsm_join_economic_alliance_effect = yes
		}
		# News Event
		news_event = {
			id = bsm_economic_alliance_news.1
		}
	}
	option = {
		name = bsm_economic_alliance_actions.1.b		# Decline
		ai_chance = {
			base = 10
			modifier = {
				add = 20
				has_opinion = {
					target = FROM
					value < 0
				}
			}
		}
		FROM = {
			country_event = {
				id = bsm_economic_alliance_actions.12
			}
		}
	}
}
