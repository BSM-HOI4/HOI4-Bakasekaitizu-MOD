add_namespace = nc

# --- 通貨危機イベント ---
country_event = {
	id = nc.1
	title = nc.1.t
	desc = nc.1.d
	picture = GFX_report_event_generic_rally

	is_triggered_only = yes

	option = {
		name = nc.1.a
		add_to_variable = { bsm_nc_rate = 50 }
		subtract_from_variable = { bsm_nc_stability = 30 }
		clamp_variable = { var = bsm_nc_stability min = 0 max = 100 }
	}
}
