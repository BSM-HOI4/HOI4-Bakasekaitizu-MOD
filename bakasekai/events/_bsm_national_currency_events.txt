# ==========================================
# EA 通貨危機伝播イベント (Phase 2)
# ==========================================

# --- EA通貨危機の警告 (bsm_ea_crisis_pressure >= 75) ---
country_event = {
	id = nc.10
	title = nc.10.t
	desc = nc.10.d
	picture = GFX_news_event_generic_diplomacy

	is_triggered_only = yes

	option = {
		name = nc.10.a
		ai_chance = { factor = 60 }
		# 共同対応: 準備金を投入して危機を緩和
		set_temp_variable = { bsm_ea_token = bsm_ea_sphere }
		bsm_ea_spend_reserve_for_crisis = yes
	}
	option = {
		name = nc.10.b
		ai_chance = { factor = 30 }
		# 資本規制のみ (軽減策なし)
		custom_effect_tooltip = nc.10_b_tt
	}
	option = {
		name = nc.10.c
		ai_chance = {
			factor = 10
			modifier = {
				factor = 2
				check_variable = { bsm_ea_crisis_pressure > 89 }
			}
		}
		# 放置: 離脱圧力上昇
		add_to_variable = { bsm_nc_union_exit_pressure = 10 }
		clamp_variable = { var = bsm_nc_union_exit_pressure min = 0 max = 100 }
	}
}
