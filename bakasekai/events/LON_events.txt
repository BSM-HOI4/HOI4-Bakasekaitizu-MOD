# ロンドン橋の定期的な損耗イベント
country_event = {
  id = LON.15
  title = LON.15.t
  desc = LON.15.desc
  picture = GFX_event_empty
  trigger = {
    tag = LON
    has_country_flag = LON_london_bridge_exists
    check_variable = {
      var = LON_london_bridge_condition
      compare = less_than
      value = 100
    }
  }
  mean_time_to_happen = {
    days = 180
    modifier = {
      factor = 0.5
      check_variable = {
        var = LON_london_bridge_condition
        compare = greater_than
        value = 50
      }
    }
    modifier = {
      factor = 0.3
      check_variable = {
        var = LON_london_bridge_condition
        compare = greater_than
        value = 80
      }
    }
  }
  immediate = {
    set_temp_variable = {
      add_damage = 5
    }
    random = {
      chance = 50
      add_to_temp_variable = {
        add_damage = 5
      }
    }
    add_to_variable = {
      LON_london_bridge_condition = temp:add_damage
    }
    clamp_variable = {
      var = LON_london_bridge_condition
      max = 100
    }
    if = {
      limit = {
        check_variable = {
          var = LON_london_bridge_condition
          value = 100
          compare = greater_than_or_equals
        }
      }
      country_event = {
        id = LON.14
        hours = 1
      }
    }
    if = {
      limit = {
        check_variable = {
          var = LON_london_bridge_condition
          compare = greater_than
          value = 70
        }
        check_variable = {
          var = LON_london_bridge_condition
          compare = less_than
          value = 100
        }
      }
      country_event = {
        id = LON.16
        hours = 1
      }
    }
  }
  option = {
    name = EVT_OK
  }
}
