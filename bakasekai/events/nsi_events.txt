	desc = nsi_events.3.d

	trigger = {
		tag = NSI
		has_idea = nsi_isolation_policy
		any_country = {
			is_neighbor_of = NSI
			NOT = { tag = NSI }
		}
	}
	
	mean_time_to_happen = {

	}
	
	option = {
		name = nsi_events.3.a

		every_neighbor_country = {
			add_opinion_modifier = {
				target = NSI

			}
		}
	}
}
